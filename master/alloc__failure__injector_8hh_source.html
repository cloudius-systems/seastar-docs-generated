48class alloc_failure_injector {
49 uint64_t _alloc_count = 0;
50 uint64_t _fail_at = std::numeric_limits<uint64_t>::max();
51 noncopyable_function<void()> _on_alloc_failure = [] { throw std::bad_alloc(); };
52 bool _failed = false;
53private:
54 void fail();
55public:
57 void on_alloc_point() {
58 if (is_critical_alloc_section()) {
59 return;
60 }
61 if (_alloc_count >= _fail_at) {
62 fail();
63 }
64 ++_alloc_count;
65 }
66
68 uint64_t alloc_count() const {
69 return _alloc_count;
70 }
71
73 void fail_after(uint64_t count) {
74 _fail_at = _alloc_count + count;
75 _failed = false;
76 }
77
79 void cancel() {
80 _fail_at = std::numeric_limits<uint64_t>::max();
81 }
82

86 }
