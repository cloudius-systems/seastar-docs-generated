47// Get a format_info's format string as a string_view.
48//
49// fmt 12 deprecates the implicit basic_format_string -> basic_string_view
50// conversion in favour of basic_format_string::get(), which was only added
51// in fmt 10. Keeping the version check in here means the call sites need no
52// #ifdef of their own.
53//
54// The format string is a deduced template parameter so one template serves
55// both the compile-time fmt::format_string<Args...> and the runtime
56// std::string_view without the caller needing to know which is in play.
57template <typename FormatString>
58fmt::string_view format_string_view(const FormatString& format) noexcept {
59#if defined(SEASTAR_LOGGER_COMPILE_TIME_FMT) && SEASTAR_FMT_VERSION >= 100000
60 return format.get();
61#else
62 return fmt::string_view(format);
63#endif
64}

276 template <typename... Args>
277 void log(log_level level, format_info_t<Args...> fmt, Args&&... args) noexcept {
278 if (is_enabled(level)) {
279 try {
280 lambda_log_writer writer([&] (internal::log_buf::inserter_iterator it) {
281#ifdef SEASTAR_LOGGER_COMPILE_TIME_FMT
282 return fmt::format_to(it, fmt.format, std::forward<Args>(args)...);
283#else
284 return fmt::format_to(it, fmt::runtime(fmt.format), std::forward<Args>(args)...);
285#endif
286 });
287 do_log(level, writer);

308 template <typename... Args>
309 void log(log_level level, rate_limit& rl, format_info_t<Args...> fmt, Args&&... args) noexcept {
310 if (is_enabled(level) && !rl.rate_limited()) {
311 try {
312 lambda_log_writer writer([&] (internal::log_buf::inserter_iterator it) {
313 if (rl.has_dropped_messages()) {
314 it = fmt::format_to(it, "(rate limiting dropped {} similar messages) ", rl.get_and_reset_dropped_messages());
315 }
316 return fmt::format_to(it, fmt::runtime(internal::format_string_view(fmt.format)), std::forward<Args>(args)...);
317 });

321 }

609// Seastar has no business defining a {fmt} formatter for std::exception_ptr,
610// a type it does not own; that is for the standard library or {fmt} to do (see
611// https://github.com/fmtlib/fmt/issues/4808). Until then we provide one, but
612// deprecate it in favour of seastar::formattable(), which wraps the pointer in
613// a Seastar-owned type. The deprecation is on parse() (which {fmt} odr-uses)
614// rather than on the specialization, so that it is diagnosed at the call site.
615template <>
616struct fmt::formatter<std::exception_ptr> {
617 [[deprecated("Use seastar::formattable(eptr) instead of formatting a std::exception_ptr directly")]]
618 constexpr auto parse(format_parse_context& ctx) { return ctx.begin(); }
619 auto format(const std::exception_ptr& eptr, fmt::format_context& ctx) const -> decltype(ctx.out()) {
620 return fmt::format_to(ctx.out(), "{}", seastar::formattable(eptr));
621 }
