24// Buffered input and output streams
25//
26// Two abstract classes (data_source and data_sink) provide means
27// to acquire bulk data from, or push bulk data to, some provider.
28// These could be tied to a TCP connection, a disk file, or a memory
29// buffer.
30//
31// Two concrete classes (input_stream and output_stream) buffer data
32// from data_source and data_sink and provide easier means to process

111class data_sink_impl {
112#ifdef SEASTAR_DATA_SINK_PUT_DEBUG
113 // Set while a put() issued through data_sink has not resolved yet.
114 // Implementations are allowed to hold per-sink state for the duration of a
115 // put, which makes two things errors: destroying the sink while this is set,
116 // which leaves the operation and the continuation that cleans up after it
117 // pointing at freed memory, and issuing a second put while it is set, which
118 // makes the two puts share that state.
119 bool _put_in_flight = false;
120
121 friend class data_sink;
122#endif
123public:
124 virtual ~data_sink_impl() {
125#ifdef SEASTAR_DATA_SINK_PUT_DEBUG
126 SEASTAR_DEBUG_ASSERT(!_put_in_flight);
127#endif
128 }
129 virtual temporary_buffer<char> allocate_buffer(size_t size) {
130 return temporary_buffer<char>(size);
131 }
132#if SEASTAR_API_LEVEL >= 9
133 // The caller assumes that the storage that backs this span can be released
134 // once this method returns, so implementations should move the buffers into
135 // stable storage on their own early, before the returned future resolves.
136 //
137 // The caller will not issue another put() until the returned future resolves,
138 // and will not destroy the sink before then, so implementations may keep
139 // per-put state in the sink for the duration of the operation.
140 virtual future<> put(std::span<temporary_buffer<char>> data) = 0;
141#else
142 virtual future<> put(net::packet data) = 0;
143 virtual future<> put(std::vector<temporary_buffer<char>> data) {
144 net::packet p;
145 p.reserve(data.size());
146 for (auto& buf : data) {
147 p = net::packet(std::move(p), net::fragment{buf.get_write(), buf.size()}, buf.release());
148 }
149 return put(std::move(p));
150 }
151 virtual future<> put(temporary_buffer<char> buf) {
152 return put(net::packet(net::fragment{buf.get_write(), buf.size()}, buf.release()));
153 }
154#endif
155 virtual future<> flush() {
156 return make_ready_future<>();
157 }
158 virtual future<> close() = 0;
159
160 // The method should return the maximum buffer size that's acceptable by
161 // the sink. It's used when the output stream is constructed without any
162 // specific buffer size. In this case the stream accepts this value as its
163 // buffer size and doesn't put larger buffers (see trim_to_size).
164 virtual size_t buffer_size() const noexcept {
165 SEASTAR_ASSERT(false && "Data sink must have the buffer_size() method overload");
166 return 0;
167 }
168
169 // In order to support flushes batching (output_stream_options.batch_flushes)
170 // the sink mush handle flush errors that may happen in the background by
171 // overriding the on_batch_flush_error() method. If the sink doesn't do it,
172 // turning on batch_flushes would have no effect
173 virtual bool can_batch_flushes() const noexcept {
174 return false;
175 }
176
177 virtual void on_batch_flush_error() noexcept {
178 SEASTAR_ASSERT(false && "Data sink must implement on_batch_flush_error() method");
179 }
180
181protected:
182#if SEASTAR_API_LEVEL >= 9
183 // A helper function that class that inhrerit from data_sink_impl
184 // can use to create a future chain holding buffers from the span
185 // to sequentially put them with the help of fn function
186 template <typename Fn>
187 requires std::is_invocable_r_v<future<>, Fn, temporary_buffer<char>&&>
188 static future<> fallback_put(std::span<temporary_buffer<char>> bufs, Fn fn) {
189 if (bufs.size() == 1) [[likely]] {
190 return fn(std::move(bufs.front()));
191 }
192
193 auto f = fn(std::move(bufs.front()));
194 for (auto&& buf : bufs.subspan(1)) {
195 f = std::move(f).then([fn, buf = std::move(buf)] () mutable {
196 return fn(std::move(buf));
197 });
198 }
199 return f;
200 }
201#else
202 // This is a helper function that classes that inherit from data_sink_impl
203 // can use to implement the put overload for net::packet.
204 // Unfortunately, we currently cannot define this function as
205 // 'virtual future<> put(net::packet)', because we would get infinite
206 // recursion between this function and
207 // 'virtual future<> put(temporary_buffer<char>)'.
208 future<> fallback_put(net::packet data) {
209 auto buffers = data.release();
210 for (temporary_buffer<char>& buf : buffers) {
211 co_await this->put(std::move(buf));
212 }
213 }
214#endif
215};

348// Consumer concept, for consume() method
349
350// The consumer should operate on the data given to it, and
351// return a future "consumption result", which can be
352// - continue_consuming, if the consumer has consumed all the input given
353// to it and is ready for more
354// - stop_consuming, when the consumer is done (and in that case
355// the contained buffer is the unconsumed part of the last data buffer - this
356// can also happen to be empty).
357// - skip_bytes, when the consumer has consumed all the input given to it
358// and wants to skip before processing the next chunk
359//
360// For backward compatibility reasons, we also support the deprecated return value
361// of type "unconsumed remainder" which can be
362// - empty optional, if the consumer consumed all the input given to it
363// and is ready for more
364// - non-empty optional, when the consumer is done (and in that case
365// the value is the unconsumed part of the last data buffer - this
366// can also happen to be empty).
367
368template <typename Consumer, typename CharType>
369concept InputStreamConsumer = requires (Consumer c) {
370 { c(temporary_buffer<CharType>{}) } -> std::same_as<future<consumption_result<CharType>>>;
371};
372
373template <typename Consumer, typename CharType>
374concept ObsoleteInputStreamConsumer = requires (Consumer c) {
375 { c(temporary_buffer<CharType>{}) } -> std::same_as<future<std::optional<temporary_buffer<CharType>>>>;
376};

503template <typename CharType>
504class output_stream final {
505 static_assert(sizeof(CharType) == 1, "must buffer stream of bytes");
506 data_sink _fd;
507 // Buffered writes accumulate into _buf[0.._end). Once _buf is full it is
508 // put() to the sink and a fresh buffer is allocated.
509 //
510 // Zero-copy writes bypass _buf and go directly into _zc_bufs. If buffered
511 // data is pending when a zero-copy write arrives, the filled prefix of _buf
512 // is shared into _zc_bufs and _buf is trim_front()'d so its tail can still
513 // be reused for future buffered writes after the zero-copy sequence.
514 //
515 // _zc_bufs is flushed to the sink (via zero_copy_put or
516 // zero_copy_split_and_put) when _zc_len >= _buffer_size, or on flush().
517 // After a flush _buf, _zc_bufs, _end and _zc_len are all reset to empty/0.
