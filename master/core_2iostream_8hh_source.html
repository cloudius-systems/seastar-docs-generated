23//
24// Buffered input and output streams
25//
26// Two abstract classes (data_source and data_sink) provide means
27// to acquire bulk data from, or push bulk data to, some provider.
28// These could be tied to a TCP connection, a disk file, or a memory
29// buffer.
30//
31// Two concrete classes (input_stream and output_stream) buffer data
32// from data_source and data_sink and provide easier means to process
33// it.
34//

106class data_sink_impl {
107public:
108 virtual ~data_sink_impl() {}
109 virtual temporary_buffer<char> allocate_buffer(size_t size) {
110 return temporary_buffer<char>(size);
111 }
112#if SEASTAR_API_LEVEL >= 9
113 // The caller assumes that the storage that backs this span can be released
114 // once this method returns, so implementations should move the buffers into
115 // stable storage on their own early, before the returned future resolves.
116 virtual future<> put(std::span<temporary_buffer<char>> data) = 0;
117#else
118 virtual future<> put(net::packet data) = 0;
119 virtual future<> put(std::vector<temporary_buffer<char>> data) {
120 net::packet p;
121 p.reserve(data.size());
122 for (auto& buf : data) {
123 p = net::packet(std::move(p), net::fragment{buf.get_write(), buf.size()}, buf.release());
124 }
125 return put(std::move(p));
126 }
127 virtual future<> put(temporary_buffer<char> buf) {
128 return put(net::packet(net::fragment{buf.get_write(), buf.size()}, buf.release()));
129 }
130#endif
131 virtual future<> flush() {
132 return make_ready_future<>();
133 }
134 virtual future<> close() = 0;
135
136 // The method should return the maximum buffer size that's acceptable by
137 // the sink. It's used when the output stream is constructed without any
138 // specific buffer size. In this case the stream accepts this value as its
139 // buffer size and doesn't put larger buffers (see trim_to_size).
140 virtual size_t buffer_size() const noexcept {
141 SEASTAR_ASSERT(false && "Data sink must have the buffer_size() method overload");
142 return 0;
143 }
144
145 // In order to support flushes batching (output_stream_options.batch_flushes)
146 // the sink mush handle flush errors that may happen in the background by
147 // overriding the on_batch_flush_error() method. If the sink doesn't do it,
148 // turning on batch_flushes would have no effect
149 virtual bool can_batch_flushes() const noexcept {
150 return false;
151 }
152
153 virtual void on_batch_flush_error() noexcept {
154 SEASTAR_ASSERT(false && "Data sink must implement on_batch_flush_error() method");
155 }
156
157protected:
158#if SEASTAR_API_LEVEL >= 9
159 // A helper function that class that inhrerit from data_sink_impl
160 // can use to create a future chain holding buffers from the span
161 // to sequentially put them with the help of fn function
162 template <typename Fn>
163 requires std::is_invocable_r_v<future<>, Fn, temporary_buffer<char>&&>
164 static future<> fallback_put(std::span<temporary_buffer<char>> bufs, Fn fn) {
165 if (bufs.size() == 1) [[likely]] {
166 return fn(std::move(bufs.front()));
167 }
168
169 auto f = fn(std::move(bufs.front()));
170 for (auto&& buf : bufs.subspan(1)) {
171 f = std::move(f).then([fn, buf = std::move(buf)] () mutable {
172 return fn(std::move(buf));
173 });
174 }
175 return f;
176 }
177#else
178 // This is a helper function that classes that inherit from data_sink_impl
179 // can use to implement the put overload for net::packet.
180 // Unfortunately, we currently cannot define this function as
181 // 'virtual future<> put(net::packet)', because we would get infinite
182 // recursion between this function and
183 // 'virtual future<> put(temporary_buffer<char>)'.
184 future<> fallback_put(net::packet data) {
185 auto buffers = data.release();
186 for (temporary_buffer<char>& buf : buffers) {
187 co_await this->put(std::move(buf));
188 }
189 }
190#endif
191};

306// Consumer concept, for consume() method
307
308// The consumer should operate on the data given to it, and
309// return a future "consumption result", which can be
310// - continue_consuming, if the consumer has consumed all the input given
311// to it and is ready for more
312// - stop_consuming, when the consumer is done (and in that case
313// the contained buffer is the unconsumed part of the last data buffer - this
314// can also happen to be empty).
315// - skip_bytes, when the consumer has consumed all the input given to it
316// and wants to skip before processing the next chunk
317//
318// For backward compatibility reasons, we also support the deprecated return value
319// of type "unconsumed remainder" which can be
320// - empty optional, if the consumer consumed all the input given to it
321// and is ready for more
322// - non-empty optional, when the consumer is done (and in that case
323// the value is the unconsumed part of the last data buffer - this
324// can also happen to be empty).
325
326template <typename Consumer, typename CharType>
327concept InputStreamConsumer = requires (Consumer c) {
328 { c(temporary_buffer<CharType>{}) } -> std::same_as<future<consumption_result<CharType>>>;
329};
330
331template <typename Consumer, typename CharType>
332concept ObsoleteInputStreamConsumer = requires (Consumer c) {
333 { c(temporary_buffer<CharType>{}) } -> std::same_as<future<std::optional<temporary_buffer<CharType>>>>;
334};

465 // Buffered writes accumulate into _buf[0.._end). Once _buf is full it is
466 // put() to the sink and a fresh buffer is allocated.
467 //
468 // Zero-copy writes bypass _buf and go directly into _zc_bufs. If buffered
469 // data is pending when a zero-copy write arrives, the filled prefix of _buf
470 // is shared into _zc_bufs and _buf is trim_front()'d so its tail can still
471 // be reused for future buffered writes after the zero-copy sequence.

473 // _zc_bufs is flushed to the sink (via zero_copy_put or
474 // zero_copy_split_and_put) when _zc_len >= _buffer_size, or on flush().
