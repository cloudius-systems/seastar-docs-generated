35template<typename T>
36struct dummy_expiry {
37 void operator()(T&) noexcept {};
38};
39
40template<typename... T>
41struct promise_expiry {
42 void operator()(promise<T...>& pr) noexcept {
43 pr.set_exception(std::make_exception_ptr(timed_out_error()));
44 };
45};

55template <typename T, typename OnExpiry = dummy_expiry<T>, typename Clock = lowres_clock>
56class expiring_fifo {
57public:
58 using clock = Clock;
59 using time_point = typename Clock::time_point;
60private:
61 struct entry {
62 std::optional<T> payload; // disengaged means that it's expired
63 timer<Clock> tr;
64 entry(T&& payload_) : payload(std::move(payload_)) {}
65 entry(const T& payload_) : payload(payload_) {}
66 entry(T payload_, expiring_fifo& ef, time_point timeout)
67 : payload(std::move(payload_))
68 , tr([this, &ef] {
69 ef._on_expiry(*payload);
70 payload = std::nullopt;
71 --ef._size;
72 ef.drop_expired_front();
73 })
74 {
75 tr.arm(timeout);
76 }
77 entry(entry&& x) = delete;
78 entry(const entry& x) = delete;
79 };
80
81 // If engaged, represents the first element.
82 // This is to avoid large allocations done by chunked_fifo for single-element cases.
83 // expiring_fifo is used to implement wait lists in synchronization primitives
84 // and in some uses it's common to have at most one waiter.
85 std::unique_ptr<entry> _front;
86
87 // There is an invariant that the front element is never expired.
88 chunked_fifo<entry> _list;
89 OnExpiry _on_expiry;
90 size_t _size = 0;
91
92 // Ensures that front() is not expired by dropping expired elements from the front.
93 void drop_expired_front() noexcept {
94 while (!_list.empty() && !_list.front().payload) {
95 _list.pop_front();
96 }
97 if (_front && !_front->payload) {
98 _front.reset();
99 }
100 }
101public:
102 expiring_fifo() noexcept = default;
103 expiring_fifo(OnExpiry on_expiry) noexcept(std::is_nothrow_move_constructible_v<OnExpiry>) : _on_expiry(std::move(on_expiry)) {}
104
105 expiring_fifo(expiring_fifo&& o) noexcept
106 : expiring_fifo(std::move(o._on_expiry)) {
107 // entry objects hold a reference to this so non-empty containers cannot be moved.
108 SEASTAR_ASSERT(o._size == 0);
109 }
110
111 expiring_fifo& operator=(expiring_fifo&& o) noexcept {
112 if (this != &o) {
113 this->~expiring_fifo();
114 new (this) expiring_fifo(std::move(o));
115 }
116 return *this;
117 }

168 void push_back(const T& payload) {
169 if (_size == 0) {
170 _front = std::make_unique<entry>(payload);
171 } else {
172 _list.emplace_back(payload);
173 }
174 ++_size;
175 }
176
179 void push_back(T&& payload) {
180 if (_size == 0) {
181 _front = std::make_unique<entry>(std::move(payload));
182 } else {
183 _list.emplace_back(std::move(payload));
184 }
185 ++_size;
186 }
187
191 void push_back(T&& payload, time_point timeout) {
192 if (timeout == time_point::max()) {
193 push_back(std::move(payload));
194 return;
195 }
196 if (_size == 0) {
197 _front = std::make_unique<entry>(std::move(payload), *this, timeout);
198 } else {
199 _list.emplace_back(std::move(payload), *this, timeout);
200 }
201 ++_size;
202 }
203
206 void pop_front() noexcept {
207 if (_front) {
208 _front.reset();
209 } else {
210 _list.pop_front();
211 }
212 --_size;
213 drop_expired_front();
214 }
215};
