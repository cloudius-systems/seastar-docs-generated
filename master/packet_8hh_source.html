61// Zero-copy friendly packet class
62//
63// For implementing zero-copy, we need a flexible destructor that can
64// destroy packet data in different ways: decrementing a reference count,
65// or calling a free()-like function.
66//
67// Moreover, we need different destructors for each set of fragments within
68// a single fragment. For example, a header and trailer might need delete[]
69// to be called, while the internal data needs a reference count to be
70// released. Matters are complicated in that fragments can be split
71// (due to virtual/physical translation).
72//
73// To implement this, we associate each packet with a single destructor,
74// but allow composing a packet from another packet plus a fragment to
75// be added, with its own destructor, causing the destructors to be chained.
76//
77// The downside is that the data needed for the destructor is duplicated,
78// if it is already available in the fragment itself.
79//
80// As an optimization, when we allocate small fragments, we allocate some
81// extra space, so prepending to the packet does not require extra
82// allocations. This is useful when adding headers.
83//
84class packet final {
85 // enough for lots of headers, not quite two cache lines:
86 static constexpr size_t internal_data_size = 128 - 16;
87 static constexpr size_t default_nr_frags = 4;

160 bool using_internal_data() const noexcept {
161 return _nr_frags

166 void unuse_internal_data() {
167 if (!using_internal_data()) {
168 return;
169 }
170 auto buf = static_cast<char*>(::malloc(_frags[0].size));
171 if (!buf) {
172 throw std::bad_alloc();
173 }
174 deleter d = make_free_deleter(buf);
175 std::copy(_frags[0].base, _frags[0].base + _frags[0].size, buf);
176 _frags[0].base = buf;
177 d.append(std::move(_deleter));
178 _deleter = std::move(d);
179 _headroom = internal_data_size;
180 }
181 void copy_internal_fragment_to(impl* to) noexcept {
182 if (!using_internal_data()) {
183 return;
184 }
185 to->_frags[0].base = to->_data + _headroom;
186 std::copy(_frags[0].base, _frags[0].base + _frags[0].size,
187 to->_frags[0].base);
188 }

290 // Call `func` for each fragment, avoiding data copies when possible
291 // `func` is called with a temporary_buffer<char> parameter
292 template <typename Func>
293 void release_into(Func&& func) {
294 unsigned idx = 0;
295 if (_impl->using_internal_data()) {
296 auto&& f = frag(idx++);
297 func(temporary_buffer<char>(f.base, f.size));
298 }
299 while (idx < nr_frags()) {
300 auto&& f = frag(idx++);
301 func(temporary_buffer<char>(f.base, f.size, _impl->_deleter.share()));
302 }
303 }
304 std::vector<temporary_buffer<char>> release() {
305 std::vector<temporary_buffer<char>> ret;
306 ret.reserve(_impl->_nr_frags);
307 release_into([&ret] (temporary_buffer<char>&& frag) {
308 ret.push_back(std::move(frag));
309 });
310 return ret;
311 }

340inline
341packet::impl::impl(fragment frag, size_t nr_frags)
342 : _len(frag.size), _allocated_frags(nr_frags) {
343 SEASTAR_ASSERT(_allocated_frags > _nr_frags);
344 if (frag.size <= internal_data_size) {
345 _headroom -= frag.size;
346 _frags[0] = { _data + _headroom, frag.size };
347 } else {
348 auto buf = static_cast<char*>(::malloc(frag.size));
349 if (!buf) {
350 throw std::bad_alloc();
351 }
352 deleter d = make_free_deleter(buf);
353 _frags[0] = { buf, frag.size };
354 _deleter.append(std::move(d));
355 }
356 std::copy(frag.base, frag.base + frag.size, _frags[0].base);
357 ++_nr_frags;
358}

437bool
438packet::allocate_headroom(size_t size) {
439 if (_impl->_headroom >= size) {
440 _impl->_len += size;
441 if (!_impl->using_internal_data()) {
442 _impl = impl::allocate_if_needed(std::move(_impl), 1);
443 std::copy_backward(_impl->_frags, _impl->_frags + _impl->_nr_frags,
444 _impl->_frags + _impl->_nr_frags + 1);
445 _impl->_frags[0] = { _impl->_data + internal_data_size, 0 };
446 ++_impl->_nr_frags;
447 }
448 _impl->_headroom -= size;
449 _impl->_frags[0].base -= size;
450 _impl->_frags[0].size += size;
451 return true;
452 } else {
453 return false;
454 }
455}

521inline
522char* packet::get_header(size_t offset, size_t size) {
523 if (offset + size > _impl->_len) {
524 return nullptr;
525 }
526 size_t i = 0;
527 while (i != _impl->_nr_frags && offset >= _impl->_frags[i].size) {
528 offset -= _impl->_frags[i++].size;
529 }
530 if (i == _impl->_nr_frags) {
531 return nullptr;
532 }
533 if (offset + size > _impl->_frags[i].size) {
534 linearize(i, offset + size);
535 }
536 return _impl->_frags[i].base + offset;
537}

545inline
546void packet::trim_front(size_t how_much) noexcept {
547 SEASTAR_ASSERT(how_much <= _impl->_len);
548 _impl->_len -= how_much;
549 size_t i = 0;
550 while (how_much && how_much >= _impl->_frags[i].size) {
551 how_much -= _impl->_frags[i++].size;
552 }
553 std::copy(_impl->_frags + i, _impl->_frags + _impl->_nr_frags, _impl->_frags);
554 _impl->_nr_frags -= i;
555 if (!_impl->using_internal_data()) {
556 _impl->_headroom = internal_data_size;
557 }
558 if (how_much) {
559 if (_impl->using_internal_data()) {
560 _impl->_headroom += how_much;
561 }
562 _impl->_frags[0].base += how_much;
563 _impl->_frags[0].size -= how_much;
564 }
565}

567inline
568void packet::trim_back(size_t how_much) noexcept {
569 SEASTAR_ASSERT(how_much <= _impl->_len);
570 _impl->_len -= how_much;
571 size_t i = _impl->_nr_frags - 1;
572 while (how_much && how_much >= _impl->_frags[i].size) {
573 how_much -= _impl->_frags[i--].size;
574 }
575 _impl->_nr_frags = i + 1;
576 if (how_much) {
577 _impl->_frags[i].size -= how_much;
578 if (i == 0 && _impl->using_internal_data()) {
579 _impl->_headroom += how_much;
580 }

619inline
620packet packet::share(size_t offset, size_t len) {
621 _impl->unuse_internal_data(); // FIXME: eliminate?
622 packet n;
623 n._impl = impl::allocate_if_needed(std::move(n._impl), _impl->_nr_frags);
624 size_t idx = 0;
625 while (offset > 0 && offset >= _impl->_frags[idx].size) {
626 offset -= _impl->_frags[idx++].size;
627 }
628 while (n._impl->_len < len) {
629 auto& f = _impl->_frags[idx++];
630 auto fsize = std::min(len - n._impl->_len, f.size - offset);
631 n._impl->_frags[n._impl->_nr_frags++] = { f.base + offset, fsize };
632 n._impl->_len += fsize;
633 offset = 0;
634 }
635 n._impl->_offload_info = _impl->_offload_info;
636 SEASTAR_ASSERT(!n._impl->_deleter);
637 n._impl->_deleter = _impl->_deleter.share();
638 return n;
639}
