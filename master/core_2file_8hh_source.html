142public:
144 file_mapping() noexcept = default;
145 file_mapping(const file_mapping&) = delete;
146 file_mapping& operator=(const file_mapping&) = delete;
147
148 file_mapping(file_mapping&& other) noexcept
149 : _addr(std::exchange(other._addr, nullptr))
150 , _length(std::exchange(other._length, 0)) {}
151
152 file_mapping& operator=(file_mapping&& other) noexcept {
153 if (this != &other) {
154 SEASTAR_ASSERT(!_addr && !_length && "file_mapping assigned to without unmapping");
155 _addr = std::exchange(other._addr, nullptr);
156 _length = std::exchange(other._length, 0);
157 }
158 return *this;
159 }
160
161 ~file_mapping() noexcept;
162
164 void* get() const noexcept { return _addr; }
165
167 size_t size() const noexcept { return _length; }

700template <std::invocable<file&> Func>
701futurize_t<std::invoke_result_t<Func, file&>> with_file(future<file> file_fut, Func func) noexcept {
702 auto f = co_await std::move(file_fut);
703 // If f.close() fails, return that as nested exception.
704 co_return co_await futurize_invoke(func, f).finally([&f] {
705 return f.close();
706 });
707}

723template <std::invocable<file&> Func>
724futurize_t<std::invoke_result_t<Func, file&>> with_file_close_on_failure(future<file> file_fut, Func func) noexcept {
725 auto f = co_await std::move(file_fut);
726 auto fut = co_await coroutine::as_future(futurize_invoke(func, f));
727 // If f.close() fails, return that as nested exception.
728 if (fut.failed()) {
729 fut = fut.finally([&f] {
730 return f.close();
731 });
732 }
733 co_return co_await std::move(fut);
734}
