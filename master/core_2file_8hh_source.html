224 // O_DIRECT reading requires that buffer, offset, and read length, are
225 // all aligned. Alignment of 4096 was necessary in the past, but no longer

228 // In addition, if we start reading in things outside page boundaries,
229 // we will end up with various pages around, some of them with

597template <std::invocable<file&> Func>
598futurize_t<std::invoke_result_t<Func, file&>> with_file(future<file> file_fut, Func func) noexcept {
599 auto f = co_await std::move(file_fut);
600 // If f.close() fails, return that as nested exception.
601 co_return co_await futurize_invoke(func, f).finally([&f] {
602 return f.close();
603 });
604}
605
620template <std::invocable<file&> Func>
621futurize_t<std::invoke_result_t<Func, file&>> with_file_close_on_failure(future<file> file_fut, Func func) noexcept {
622 auto f = co_await std::move(file_fut);
623 auto fut = co_await coroutine::as_future(futurize_invoke(func, f));
624 // If f.close() fails, return that as nested exception.
625 if (fut.failed()) {
626 fut = fut.finally([&f] {
627 return f.close();
628 });
629 }
630 co_return co_await std::move(fut);
631}
