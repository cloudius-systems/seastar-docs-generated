183struct awaiter {
184 seastar::future<T> _future;
185public:
186 explicit awaiter(seastar::future<T>&& f) noexcept : _future(std::move(f)) { }
187
188 awaiter(const awaiter&) = delete;
189 awaiter(awaiter&&) = delete;
190
191 bool await_ready() const noexcept {
192 return _future.available() && (!CheckPreempt || !need_preempt());
193 }
194
195 template<typename U>
196 void await_suspend(std::coroutine_handle<U> hndl SEASTAR_COROUTINE_LOC_PARAM) noexcept {
197 SEASTAR_COROUTINE_LOC_STORE(hndl.promise());
198 if (!CheckPreempt || !_future.available()) {
199 _future.set_coroutine(hndl.promise());
200 } else {
201 schedule(&hndl.promise());
202 }
203 }
204
205 T await_resume() { return _future.get(); }
206};
