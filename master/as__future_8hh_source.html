31template <bool CheckPreempt, typename T>
32class [[nodiscard]] as_future_awaiter {
33 seastar::future<T> _future;
34
35public:
36 explicit as_future_awaiter(seastar::future<T>&& f) noexcept : _future(std::move(f)) {}
37
38 as_future_awaiter(const as_future_awaiter&) = delete;
39 as_future_awaiter(as_future_awaiter&&) = delete;
40
41 bool await_ready() const noexcept {
42 return _future.available() && (!CheckPreempt || !need_preempt());
43 }
44
45 template<typename U>
46 void await_suspend(std::coroutine_handle<U> hndl SEASTAR_COROUTINE_LOC_PARAM) noexcept {
47 SEASTAR_COROUTINE_LOC_STORE(hndl.promise());
48 if (!CheckPreempt || !_future.available()) {
49 _future.set_coroutine(hndl.promise());
50 } else {
51 schedule(&hndl.promise());
52 }
53 }
54
55 seastar::future<T> await_resume() {
56 return std::move(_future);
57 }
58};
