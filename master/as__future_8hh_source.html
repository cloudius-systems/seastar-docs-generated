31// This makes calls to the factories below safe elide contexts, so their
32// annotated arguments can propagate to the wrapped coroutine.
33template <bool CheckPreempt, typename T>
34class [[nodiscard]] SEASTAR_CORO_AWAIT_ELIDABLE as_future_awaiter {
35 seastar::future<T> _future;
36
37public:
38 explicit as_future_awaiter(seastar::future<T>&& f) noexcept : _future(std::move(f)) {}
39
40 as_future_awaiter(const as_future_awaiter&) = delete;
41 as_future_awaiter(as_future_awaiter&&) = delete;
42
43 bool await_ready() const noexcept {
44 return _future.available() && (!CheckPreempt || !need_preempt());
45 }
46
47 template<typename U>
48 void await_suspend(std::coroutine_handle<U> hndl SEASTAR_COROUTINE_LOC_PARAM) noexcept {
49 SEASTAR_COROUTINE_LOC_STORE(hndl.promise());
50 if (!CheckPreempt || !_future.available()) {
51 _future.set_coroutine(hndl.promise());
52 } else {
53 schedule(&hndl.promise());
54 }
55 }
56
57 seastar::future<T> await_resume() {
58 return std::move(_future);
59 }
60};
61
62} // namespace seastar::internal
