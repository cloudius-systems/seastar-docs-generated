251// Takes a vector of buffers and detaches some buffers from its front so
252// that the total size of the detached part equals the given value. The
253// detached buffers are returned, the original vector is updated to hold
254// the remainder.
255//
256// Buffers content is not copied. If it happens that one buffer must be
257// split, it is shared between the returned and the original vectors.
258
259template <typename CharType>
260inline std::vector<temporary_buffer<CharType>> detach_front(std::vector<temporary_buffer<CharType>>& bufs, size_t length) {
261 std::vector<temporary_buffer<CharType>> ret;
262 auto it = bufs.begin();
263 while (it != bufs.end()) {
264 if (it->size() <= length) {
265 length -= it->size();
266 ret.emplace_back(std::move(*it));
267 it++;
268 continue;
269 } else {
270 if (length > 0) {
271 auto b = it->share();
272 b.trim(length);
273 it->trim_front(length);
274 ret.emplace_back(std::move(b));
275 length = 0;
276 }
277 break;
278 }
279 }
280 bufs.erase(bufs.begin(), it);
281 SEASTAR_ASSERT(length == 0);
282 return ret;
283}
