132// Permissions for files/directories
133enum class file_permissions {
134 user_read = S_IRUSR, // Read by owner
135 user_write = S_IWUSR, // Write by owner
136 user_execute = S_IXUSR, // Execute by owner
137
138 group_read = S_IRGRP, // Read by group
139 group_write = S_IWGRP, // Write by group
140 group_execute = S_IXGRP, // Execute by group
141
142 others_read = S_IROTH, // Read by others
143 others_write = S_IWOTH, // Write by others
144 others_execute = S_IXOTH, // Execute by others
145
146 user_permissions = user_read | user_write | user_execute,
147 group_permissions = group_read | group_write | group_execute,
148 others_permissions = others_read | others_write | others_execute,
149 all_permissions = user_permissions | group_permissions | others_permissions,
150
151 default_file_permissions = user_read | user_write | group_read | group_write | others_read | others_write, // 0666
152 default_dir_permissions = all_permissions, // 0777
153};
154
