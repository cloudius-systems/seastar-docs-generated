55class shared_mutex {
56 unsigned _readers = 0;
57 bool _writer = false;
58 struct waiter {
59 waiter(promise<>&& pr, bool for_write) : pr(std::move(pr)), for_write(for_write) {}
60 promise<> pr;
61 bool for_write;
62 };
63 chunked_fifo<waiter> _waiters;
64public:
65 shared_mutex() = default;
66 shared_mutex(shared_mutex&&) = default;
67 shared_mutex& operator=(shared_mutex&&) = default;
68 shared_mutex(const shared_mutex&) = delete;
69 void operator=(const shared_mutex&) = delete;
74 future<> lock_shared() noexcept {
75 if (try_lock_shared()) {
76 return make_ready_future<>();
77 }
78 try {
79 _waiters.emplace_back(promise<>(), false);
80 return _waiters.back().pr.get_future();
81 } catch (...) {
82 return current_exception_as_future();
83 }
84 }
88 bool try_lock_shared() noexcept {
89 if (!_writer && _waiters.empty()) {
90 ++_readers;
91 return true;
92 }
93 return false;
94 }
96 void unlock_shared() noexcept {
97 SEASTAR_ASSERT(_readers > 0);
98 --_readers;
99 wake();
100 }
105 future<> lock() noexcept {
106 if (try_lock()) {
107 return make_ready_future<>();
108 }
109 try {
110 _waiters.emplace_back(promise<>(), true);
111 return _waiters.back().pr.get_future();
112 } catch (...) {
113 return current_exception_as_future();
114 }
115 }
119 bool try_lock() noexcept {
120 if (!_readers && !_writer) {
121 _writer = true;
122 return true;
123 }
124 return false;
125 }
127 void unlock() noexcept {
128 SEASTAR_ASSERT(_writer);
129 _writer = false;
130 wake();
131 }
132private:
133 void wake() noexcept {
134 while (!_waiters.empty()) {
135 auto& w = _waiters.front();
136 // note: _writer == false in wake()
137 if (w.for_write) {
138 if (!_readers) {
139 _writer = true;
140 w.pr.set_value();
141 _waiters.pop_front();
142 }
143 break;
144 } else { // for read
145 ++_readers;
146 w.pr.set_value();
147 _waiters.pop_front();
148 }
149 }
150 }
151};
