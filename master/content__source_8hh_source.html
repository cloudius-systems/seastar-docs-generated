39/*
40 * An input_stream wrapper that allows to read only "length" bytes
41 * from it, used to handle requests with large bodies.
42 * */
43class content_length_source_impl : public data_source_impl {
44 input_stream<char>& _inp;
45 /*
46 * The _remaining_bytes references the counter of yet unread body bytes.
47 * The external counter variable is provided by users that want to track
48 * this value. If it's not the case and the external counter is not provided,
49 * this reference points to _builtin_remaining_bytes member.
50 */
51 size_t _builtin_remaining_bytes = 0;
52 size_t& _remaining_bytes;
53 size_t _length;
54public:
55 content_length_source_impl(input_stream<char>& inp, size_t length, size_t& remaining)
56 : _inp(inp), _remaining_bytes(remaining), _length(length)
57 {
58 _remaining_bytes = length;
59 }
60
61 content_length_source_impl(input_stream<char>& inp, size_t length)
62 : _inp(inp), _builtin_remaining_bytes(length), _remaining_bytes(_builtin_remaining_bytes), _length(length) {
63 }
64
65 virtual future<temporary_buffer<char>> get() override {
66 if (_remaining_bytes == 0) {
67 return make_ready_future<temporary_buffer<char>>();
68 }
69 return _inp.read_up_to(_remaining_bytes).then([this] (temporary_buffer<char> tmp_buf) {
70 if (tmp_buf.empty()) {
71 // read_up_to() asked for a non-zero amount, so an empty buffer is
72 // end of stream and the body is not over yet. Answering it as a
73 // clean end of stream hands the reader a silently short body.
74 return make_exception_future<temporary_buffer<char>>(std::system_error(
75 std::make_error_code(std::errc::protocol_error),
76 format("Body ended with {} of its {} bytes undelivered", _remaining_bytes, _length)));
77 }
78 _remaining_bytes -= tmp_buf.size();
79 return make_ready_future<temporary_buffer<char>>(std::move(tmp_buf));
80 });
81 }
82
83 virtual future<temporary_buffer<char>> skip(uint64_t n) override {
84 uint64_t skip_bytes = std::min(n, _remaining_bytes);
85 _remaining_bytes -= skip_bytes;
86 return _inp.skip(skip_bytes).then([] {
87 return temporary_buffer<char>();
88 });
89 }
90
91 virtual future<> close() override {
92 return make_ready_future<>();
93 }
94};

96/*
97 * An input_stream wrapper that decodes a request body
98 * with "chunked" encoding.
99 * */
100class chunked_source_impl : public data_source_impl {
101 class chunk_parser {
102 enum class parsing_state
103 : uint8_t {
104 size_and_ext,
105 body,
106 trailer_part
107 };
108 http_chunk_size_and_ext_parser _size_and_ext_parser;
109 http_chunk_trailer_parser _trailer_parser;
110
111 temporary_buffer<char> _buf;
112 size_t _current_chunk_bytes_read = 0;
113 size_t _current_chunk_length;
114 parsing_state _ps = parsing_state::size_and_ext;
115 bool _end_of_request = false;
116 // references to fields in the request structure
117 std::unordered_map<sstring, sstring>& _chunk_extensions;
118 std::unordered_map<sstring, sstring>& _trailing_headers;
119 /*
120 * Reference to the counter of yet unread body bytes, similar to the one
121 * from content_length_source_impl. Similarly, when the external counter is
122 * not provided it points to the _builtin_remaining_bytes, which sits on the
123 * chunked_source_impl itself, not in the chunk_parser.
124 *
125 * This counter differs from the content_length_source_impl one -- since body
126 * size is not known for chunked body, this counter has only two values. It
127 * starts with uint64_t::max and remains such up until all the body is read.
128 * Once end-of-body is met, the counter is reset to zero.
129 */
130 size_t& _remaining_bytes;
131 using consumption_result_type = consumption_result<char>;
132 public:
133 chunk_parser(std::unordered_map<sstring, sstring>& chunk_extensions, std::unordered_map<sstring, sstring>& trailing_headers, size_t& remaining)
134 : _chunk_extensions(chunk_extensions), _trailing_headers(trailing_headers), _remaining_bytes(remaining) {
135 _remaining_bytes = std::numeric_limits<size_t>::max();
136 _size_and_ext_parser.init();
137 }
138 temporary_buffer<char> buf() {
139 _current_chunk_bytes_read += _buf.size();
140 return std::move(_buf);
141 }
142
143 future<consumption_result_type> operator()(temporary_buffer<char> data) {
144 if (_buf.size() || _end_of_request) {
145 // return if we have already read some content (_buf.size()), or we have
146 // already reached the end of the chunked request (_end_of_request)
147 return make_ready_future<consumption_result_type>(stop_consuming(std::move(data)));
148 }
149 if (data.empty()) {
150 // consume() only hands over an empty buffer at end of stream, and the
151 // terminating chunk has not been seen, so the body was cut short.
152 return make_exception_future<consumption_result_type>(std::system_error(
153 std::make_error_code(std::errc::protocol_error), "Chunked body ended before its last chunk"));
154 }
155 switch (_ps) {
156 // "data" buffer is non-empty
157 case parsing_state::size_and_ext:
158 return _size_and_ext_parser(std::move(data)).then([this] (consumption_result_type res) {
159 if (auto* stop = std::get_if<stop_consuming<char>>(&res.get())) {
160 if (_size_and_ext_parser.failed()) {
161 return make_exception_future<consumption_result_type>(bad_request_exception("Can't parse chunk size and extensions"));
162 }
163 // save extensions
164 auto parsed_extensions = _size_and_ext_parser.get_parsed_extensions();
165 _chunk_extensions.merge(parsed_extensions);
166 for (auto& key_val : parsed_extensions) {
167 _chunk_extensions[key_val.first] += sstring(",") + key_val.second;
168 }
169
170 // save size
171 auto size_string = _size_and_ext_parser.get_size();
172 if (size_string.size() > 16) {
173 return make_exception_future<consumption_result_type>(bad_chunk_exception("Chunk length too big"));
174 }
175 _current_chunk_bytes_read = 0;
176 _current_chunk_length = strtol(size_string.c_str(), nullptr, 16);
177
178 if (_current_chunk_length == 0) {
179 _ps = parsing_state::trailer_part;
180 _trailer_parser.init();
181 } else {
182 _ps = parsing_state::body;
183 }
184 if (stop->get_buffer().empty()) {
185 return make_ready_future<consumption_result_type>(continue_consuming{});
186 }
187 return this->operator()(std::move(stop->get_buffer()));
188 } else {
189 return make_ready_future<consumption_result_type>(continue_consuming{});
190 }
191 });
192 case parsing_state::body:
193 // read the new data into _buf
194 if (_current_chunk_bytes_read < _current_chunk_length) {
195 size_t to_read = std::min(_current_chunk_length - _current_chunk_bytes_read, data.size());
196 if (_buf.empty()) {
197 _buf = data.share(0, to_read);
198 }
199 data.trim_front(to_read);
200 return make_ready_future<consumption_result_type>(stop_consuming(std::move(data)));
201 }
202
203 // chunk body is finished, we haven't entered the previous if, so "data" is still non-empty
204 if (_current_chunk_bytes_read == _current_chunk_length) {
205 // we haven't read \r yet
206 if (data.get()[0] != '\r') {
207 return make_exception_future<consumption_result_type>(bad_chunk_exception("The actual chunk length exceeds the specified length"));
208 } else {
209 _current_chunk_bytes_read++;
210 data.trim_front(1);
211 if (data.empty()) {
212 return make_ready_future<consumption_result_type>(continue_consuming{});
213 }
214 }
215 }
216 if (_current_chunk_bytes_read == _current_chunk_length + 1) {
217 // we haven't read \n but have \r
218 if (data.get()[0] != '\n') {
219 return make_exception_future<consumption_result_type>(bad_chunk_exception("The actual chunk length exceeds the specified length"));
220 } else {
221 _ps = parsing_state::size_and_ext;
222 _size_and_ext_parser.init();
223 data.trim_front(1);
224 if (data.empty()) {
225 return make_ready_future<consumption_result_type>(continue_consuming{});
226 }
227 }
228 }
229 return this->operator()(std::move(data));
230 case parsing_state::trailer_part:
231 return _trailer_parser(std::move(data)).then([this] (consumption_result_type res) {
232 if (auto* stop = std::get_if<stop_consuming<char>>(&res.get())) {
233 if (_trailer_parser.failed()) {
234 return make_exception_future<consumption_result_type>(bad_request_exception("Can't parse chunked request trailer"));
235 }
236 // save trailing headers
237 _trailing_headers = _trailer_parser.get_parsed_headers();
238 _end_of_request = true;
239 _remaining_bytes = 0;
240 return make_ready_future<consumption_result_type>(std::move(*stop));
241 } else {
242 return make_ready_future<consumption_result_type>(continue_consuming{});
243 }
244 });
245 }
246 __builtin_unreachable();
247 }
248 };
249 input_stream<char>& _inp;
250 chunk_parser _chunk;
251 size_t _builtin_remaining_bytes = 0;
252
253public:
254 chunked_source_impl(input_stream<char>& inp, std::unordered_map<sstring, sstring>& chunk_extensions, std::unordered_map<sstring, sstring>& trailing_headers, size_t& remaining)
255 : _inp(inp), _chunk(chunk_extensions, trailing_headers, remaining) {
256 }
257
258 chunked_source_impl(input_stream<char>& inp, std::unordered_map<sstring, sstring>& chunk_extensions, std::unordered_map<sstring, sstring>& trailing_headers)
259 : _inp(inp), _chunk(chunk_extensions, trailing_headers, _builtin_remaining_bytes) {
260 }
261
262 virtual future<temporary_buffer<char>> get() override {
263 return _inp.consume(_chunk).then([this] () mutable {
264 return _chunk.buf();
265 });
266 }
267
268 virtual future<> close() override {
269 return make_ready_future<>();
270 }
271};
