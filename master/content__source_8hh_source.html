37/*
38 * An input_stream wrapper that allows to read only "length" bytes
39 * from it, used to handle requests with large bodies.
40 * */
41class content_length_source_impl : public data_source_impl {
42 input_stream<char>& _inp;
43 /*
44 * The _remaining_bytes references the counter of yet unread body bytes.
45 * The external counter variable is provided by users that want to track
46 * this value. If it's not the case and the external counter is not provided,
47 * this reference points to _builtin_remaining_bytes member.
48 */
49 size_t _builtin_remaining_bytes = 0;
50 size_t& _remaining_bytes;
51public:
52 content_length_source_impl(input_stream<char>& inp, size_t length, size_t& remaining)
53 : _inp(inp), _remaining_bytes(remaining)
54 {
55 _remaining_bytes = length;
56 }
57
58 content_length_source_impl(input_stream<char>& inp, size_t length)
59 : _inp(inp), _builtin_remaining_bytes(length), _remaining_bytes(_builtin_remaining_bytes) {
60 }
61
62 virtual future<temporary_buffer<char>> get() override {
63 if (_remaining_bytes == 0) {
64 return make_ready_future<temporary_buffer<char>>();
65 }
66 return _inp.read_up_to(_remaining_bytes).then([this] (temporary_buffer<char> tmp_buf) {
67 _remaining_bytes -= tmp_buf.size();
68 return tmp_buf;
69 });
70 }
71
72 virtual future<temporary_buffer<char>> skip(uint64_t n) override {
73 uint64_t skip_bytes = std::min(n, _remaining_bytes);
74 _remaining_bytes -= skip_bytes;
75 return _inp.skip(skip_bytes).then([] {
76 return temporary_buffer<char>();
77 });
78 }
79
80 virtual future<> close() override {
81 return make_ready_future<>();
82 }
83};

85/*
86 * An input_stream wrapper that decodes a request body
87 * with "chunked" encoding.
88 * */
89class chunked_source_impl : public data_source_impl {
90 class chunk_parser {
91 enum class parsing_state
92 : uint8_t {
93 size_and_ext,
94 body,
95 trailer_part
96 };
97 http_chunk_size_and_ext_parser _size_and_ext_parser;
98 http_chunk_trailer_parser _trailer_parser;
99
100 temporary_buffer<char> _buf;
101 size_t _current_chunk_bytes_read = 0;
102 size_t _current_chunk_length;
103 parsing_state _ps = parsing_state::size_and_ext;
104 bool _end_of_request = false;
105 // references to fields in the request structure
106 std::unordered_map<sstring, sstring>& _chunk_extensions;
107 std::unordered_map<sstring, sstring>& _trailing_headers;
108 /*
109 * Reference to the counter of yet unread body bytes, similar to the one
110 * from content_length_source_impl. Similarly, when the external counter is
111 * not provided it points to the _builtin_remaining_bytes, which sits on the
112 * chunked_source_impl itself, not in the chunk_parser.
113 *
114 * This counter differs from the content_length_source_impl one -- since body
115 * size is not known for chunked body, this counter has only two values. It
116 * starts with uint64_t::max and remains such up until all the body is read.
117 * Once end-of-body is met, the counter is reset to zero.
118 */
119 size_t& _remaining_bytes;
120 using consumption_result_type = consumption_result<char>;
121 public:
122 chunk_parser(std::unordered_map<sstring, sstring>& chunk_extensions, std::unordered_map<sstring, sstring>& trailing_headers, size_t& remaining)
123 : _chunk_extensions(chunk_extensions), _trailing_headers(trailing_headers), _remaining_bytes(remaining) {
124 _remaining_bytes = std::numeric_limits<size_t>::max();
125 _size_and_ext_parser.init();
126 }
127 temporary_buffer<char> buf() {
128 _current_chunk_bytes_read += _buf.size();
129 return std::move(_buf);
130 }
131
132 future<consumption_result_type> operator()(temporary_buffer<char> data) {
133 if (_buf.size() || _end_of_request || data.empty()) {
134 // return if we have already read some content (_buf.size()), we have already reached the end of the chunked request (_end_of_request),
135 // or the underlying stream reached eof (data.empty())
136 return make_ready_future<consumption_result_type>(stop_consuming(std::move(data)));
137 }
138 switch (_ps) {
139 // "data" buffer is non-empty
140 case parsing_state::size_and_ext:
141 return _size_and_ext_parser(std::move(data)).then([this] (std::optional<temporary_buffer<char>> res) {
142 if (res.has_value()) {
143 if (_size_and_ext_parser.failed()) {
144 return make_exception_future<consumption_result_type>(bad_request_exception("Can't parse chunk size and extensions"));
145 }
146 // save extensions
147 auto parsed_extensions = _size_and_ext_parser.get_parsed_extensions();
148 _chunk_extensions.merge(parsed_extensions);
149 for (auto& key_val : parsed_extensions) {
150 _chunk_extensions[key_val.first] += sstring(",") + key_val.second;
151 }
152
153 // save size
154 auto size_string = _size_and_ext_parser.get_size();
155 if (size_string.size() > 16) {
156 return make_exception_future<consumption_result_type>(bad_chunk_exception("Chunk length too big"));
157 }
158 _current_chunk_bytes_read = 0;
159 _current_chunk_length = strtol(size_string.c_str(), nullptr, 16);
160
161 if (_current_chunk_length == 0) {
162 _ps = parsing_state::trailer_part;
163 _trailer_parser.init();
164 } else {
165 _ps = parsing_state::body;
166 }
167 if (res->empty()) {
168 return make_ready_future<consumption_result_type>(continue_consuming{});
169 }
170 return this->operator()(std::move(res.value()));
171 } else {
172 return make_ready_future<consumption_result_type>(continue_consuming{});
173 }
174 });
175 case parsing_state::body:
176 // read the new data into _buf
177 if (_current_chunk_bytes_read < _current_chunk_length) {
178 size_t to_read = std::min(_current_chunk_length - _current_chunk_bytes_read, data.size());
179 if (_buf.empty()) {
180 _buf = data.share(0, to_read);
181 }
182 data.trim_front(to_read);
183 return make_ready_future<consumption_result_type>(stop_consuming(std::move(data)));
184 }
185
186 // chunk body is finished, we haven't entered the previous if, so "data" is still non-empty
187 if (_current_chunk_bytes_read == _current_chunk_length) {
188 // we haven't read \r yet
189 if (data.get()[0] != '\r') {
190 return make_exception_future<consumption_result_type>(bad_chunk_exception("The actual chunk length exceeds the specified length"));
191 } else {
192 _current_chunk_bytes_read++;
193 data.trim_front(1);
194 if (data.empty()) {
195 return make_ready_future<consumption_result_type>(continue_consuming{});
196 }
197 }
198 }
199 if (_current_chunk_bytes_read == _current_chunk_length + 1) {
200 // we haven't read \n but have \r
201 if (data.get()[0] != '\n') {
202 return make_exception_future<consumption_result_type>(bad_chunk_exception("The actual chunk length exceeds the specified length"));
203 } else {
204 _ps = parsing_state::size_and_ext;
205 _size_and_ext_parser.init();
206 data.trim_front(1);
207 if (data.empty()) {
208 return make_ready_future<consumption_result_type>(continue_consuming{});
209 }
210 }
211 }
212 return this->operator()(std::move(data));
213 case parsing_state::trailer_part:
214 return _trailer_parser(std::move(data)).then([this] (std::optional<temporary_buffer<char>> res) {
215 if (res.has_value()) {
216 if (_trailer_parser.failed()) {
217 return make_exception_future<consumption_result_type>(bad_request_exception("Can't parse chunked request trailer"));
218 }
219 // save trailing headers
220 _trailing_headers = _trailer_parser.get_parsed_headers();
221 _end_of_request = true;
222 _remaining_bytes = 0;
223 return make_ready_future<consumption_result_type>(stop_consuming(std::move(*res)));
224 } else {
225 return make_ready_future<consumption_result_type>(continue_consuming{});
226 }
227 });
228 }
229 __builtin_unreachable();
230 }
231 };
232 input_stream<char>& _inp;
233 chunk_parser _chunk;
234 size_t _builtin_remaining_bytes = 0;
235
236public:
237 chunked_source_impl(input_stream<char>& inp, std::unordered_map<sstring, sstring>& chunk_extensions, std::unordered_map<sstring, sstring>& trailing_headers, size_t& remaining)
238 : _inp(inp), _chunk(chunk_extensions, trailing_headers, remaining) {
239 }
240
241 chunked_source_impl(input_stream<char>& inp, std::unordered_map<sstring, sstring>& chunk_extensions, std::unordered_map<sstring, sstring>& trailing_headers)
242 : _inp(inp), _chunk(chunk_extensions, trailing_headers, _builtin_remaining_bytes) {
243 }
244
245 virtual future<temporary_buffer<char>> get() override {
246 return _inp.consume(_chunk).then([this] () mutable {
247 return _chunk.buf();
248 });
249 }
250
251 virtual future<> close() override {
252 return make_ready_future<>();
253 }
254};
