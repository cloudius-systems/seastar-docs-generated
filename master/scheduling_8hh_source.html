262template<typename ConstructorType, typename Tuple, size_t...Idx>
263void apply_constructor(void* pre_alocated_mem, Tuple args, std::index_sequence<Idx...>) {
264 new (pre_alocated_mem) ConstructorType(std::get<Idx>(args)...);
265}

281template <typename T, typename... ConstructorArgs>
282scheduling_group_key_config
283make_scheduling_group_key_config(ConstructorArgs... args) {
284 scheduling_group_key_config sgkc(typeid(T));
285 sgkc.allocation_size = sizeof(T);
286 sgkc.alignment = alignof(T);
287 sgkc.constructor = [args = std::make_tuple(args...)] (void* p) {
288 internal::apply_constructor<T>(p, args, std::make_index_sequence<sizeof...(ConstructorArgs)>());
289 };
290 sgkc.destructor = [] (void* p) {
291 static_cast<T*>(p)->~T();
292 };
293 if constexpr (requires(T key) { key.rename(); }) {
294 sgkc.rename = [] (void* p) {
295 static_cast<T*>(p)->rename();
296 };
297 }
298 return sgkc;
299}
