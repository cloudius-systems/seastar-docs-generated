122private:
123 using subscription_list_type = bi::list<subscription, bi::constant_time_size<false>>;
124 subscription_list_type _subscriptions;

127 void do_request_abort(std::optional<std::exception_ptr> ex) noexcept {
128 if (_ex) {
129 return;
130 }
131 _ex = ex.value_or(get_default_exception());
132 SEASTAR_ASSERT(_ex);
133 auto subs = std::move(_subscriptions);
134 while (!subs.empty()) {
135 subscription& s = subs.front();
136 s.on_abort(ex);
137 }
138 }

162 template <typename Func>
163 requires (std::is_nothrow_invocable_r_v<void, Func, const std::optional<std::exception_ptr>&> ||
164 std::is_nothrow_invocable_r_v<void, Func>)
165 [[nodiscard]]
166 optimized_optional<subscription> subscribe(Func&& f) {
167 if constexpr (std::is_invocable_v<Func, std::exception_ptr>) {
168 return { subscription(*this, std::forward<Func>(f)) };
169 } else {
170 return { subscription(subscription::naive_cb_tag{}, *this, std::forward<Func>(f)) };
171 }
172 }
