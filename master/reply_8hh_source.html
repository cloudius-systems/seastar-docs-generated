139 static constexpr status_class classify_status(status_type http_status) {
140 auto sc = static_cast<std::underlying_type_t<status_type>>(http_status) / 100;
141 if (sc < 1 || sc > 5) [[unlikely]] {
142 return status_class::unclassified;
143 }
144 return static_cast<status_class>(sc);

214 reply& set_content_type(std::optional<std::string_view> content_type) {
215 if (!content_type) {
216 return *this;
217 }
218 std::string_view ct = *content_type;
219 if (ct.find('/') == std::string_view::npos) {
220 ct = http::mime_types::extension_to_type(ct);
221 }
222 _headers["Content-Type"] = sstring(ct);
223 return *this;
