136 static constexpr status_class classify_status(status_type http_status) {
137 auto sc = static_cast<std::underlying_type_t<status_type>>(http_status) / 100;
138 if (sc < 1 || sc > 5) [[unlikely]] {
139 return status_class::unclassified;
140 }
141 return static_cast<status_class>(sc);
