29template <template <typename T> class Container, typename Element>
30concept CollectionOf = requires (Container<Element> col) {
31 { col.begin() };
32 { col.end() };
33 { *col.begin() } -> std::same_as<Element&>;
34};

36template <template <typename T> class Container>
37requires (CollectionOf<Container, temporary_buffer<char>> && std::is_nothrow_move_constructible_v<Container<temporary_buffer<char>>>)
38class basic_memory_data_source final : public data_source_impl {
39 using real_container = Container<temporary_buffer<char>>;
40 real_container _bufs;
41 real_container::iterator _cur;
42
43public:
44 explicit basic_memory_data_source(real_container&& b) noexcept
45 : _bufs(std::move(b))
46 , _cur(_bufs.begin())
47 {}
48
49 virtual future<temporary_buffer<char>> get() override {
50 return make_ready_future<temporary_buffer<char>>(_cur != _bufs.end() ? std::move(*_cur++) : temporary_buffer<char>());
51 }
52};

54using memory_data_source = basic_memory_data_source<std::vector>;

56template <template <typename T> class C>
57requires requires (C<temporary_buffer<char>> t) { basic_memory_data_source(std::move(t)); }
58inline input_stream<char> as_input_stream(C<temporary_buffer<char>>&& bufs) {
