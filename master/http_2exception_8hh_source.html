68class redirect_exception : public base_exception {
69public:
70 using header_list = std::initializer_list<std::pair<std::string, std::string>>;
71
72 redirect_exception(const std::string& url,
73 http::reply::status_type status = http::reply::status_type::moved_permanently,
74 header_list extra_headers = {})
75 : base_exception("", status), url(url), _extra_headers(extra_headers) {
76 }
77
79 http::reply to_reply() const {
80 http::reply reply{};
81 reply.add_header("Location", url);
82 for (const auto& [name, value] : _extra_headers) {
83 reply.add_header(name, value);
84 }
85 reply.set_status(status());
86 return reply;
87 }
88
89 std::string url;
90private:
91 std::vector<std::pair<std::string, std::string>> _extra_headers;
92};
