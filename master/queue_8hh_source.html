128 void abort(std::exception_ptr ex) noexcept {
129 // std::queue::empty() and pop() doesn't throw
130 // since it just calls seastar::circular_buffer::pop_front
131 // that is specified as noexcept.
132 while (!_q.empty()) {
133 _q.pop();
134 }
135 _ex = ex;
136 if (_not_full) {
137 _not_full->set_exception(ex);
138 _not_full= std::nullopt;
139 }
140 if (_not_empty) {
141 _not_empty->set_exception(std::move(ex));
142 _not_empty = std::nullopt;
143 }
144 }

202template <typename T>
203requires std::is_nothrow_move_constructible_v<T>
204inline
205T queue<T>::pop() noexcept {
206 if (_q.size() == _max) {
207 notify_not_full();
208 }
209 // popping the front element must not throw
210 // as T is required to be nothrow_move_constructible
211 // and std::queue::pop won't throw since it uses
212 // seastar::circular_beffer::pop_front.
213 SEASTAR_ASSERT(!_q.empty());
214 T data = std::move(_q.front());
215 _q.pop();
216 return data;
217}

219template <typename T>
220// seastar allows only nothrow_move_constructible types
221// to be returned as future<T>
222requires std::is_nothrow_move_constructible_v<T>
223inline
224future<T> queue<T>::pop_eventually() noexcept {
225 if (_ex) {
226 return make_exception_future<T>(_ex);
227 }
228 if (empty()) {
229 return not_empty().then([this] {
230 if (_ex) {
231 return make_exception_future<T>(_ex);
232 } else {
233 return make_ready_future<T>(pop());
234 }
235 });
236 } else {
237 return make_ready_future<T>(pop());
238 }
239}

241template <typename T>
242requires std::is_nothrow_move_constructible_v<T>
243inline
244future<> queue<T>::push_eventually(T&& data) noexcept {
245 if (_ex) {
246 return make_exception_future<>(_ex);
247 }
248 if (full()) {
249 return not_full().then([this, data = std::move(data)] () mutable {
250 _q.push(std::move(data));
251 notify_not_empty();
252 });
253 } else {
254 try {
255 _q.push(std::move(data));
256 notify_not_empty();
257 return make_ready_future<>();
258 } catch (...) {
259 return current_exception_as_future();
260 }

264template <typename T>

266template <typename Func>
267inline
268bool queue<T>::consume(Func&& func) {
269 if (_ex) {
270 std::rethrow_exception(_ex);
271 }
272 bool running = true;
273 while (!_q.empty() && running) {
274 running = func(std::move(_q.front()));
275 _q.pop();
276 }
277 if (!full()) {
278 notify_not_full();
279 }
280 return running;
