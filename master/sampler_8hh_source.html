45class sampler {
46public:
53 constexpr sampler() = default;
55 void set_sampling_interval(uint64_t sampling_interval) {
56 sampling_interval_ = sampling_interval;
57 if (sampling_interval_ == 0) {
58 // Set the interval very large. This means in practice we will
59 // likely never get this below zero and hence it's unlikely we will
60 // ever have to run the reset path with sampling off
61 interval_to_next_sample_ = std::numeric_limits<int64_t>::max();
62 return;
63 }
64 sampling_rate_ = 1.0 / static_cast<double>(sampling_interval_);
65 interval_to_next_sample_ = next_sampling_interval();
66 }
67
73 [[gnu::always_inline]]
74 bool maybe_sample(size_t alloc_size) {
75 return (interval_to_next_sample_ -= alloc_size) < 0;
76 }
77
81 bool definitely_sample(size_t alloc_size) {
82 // this will hold if maybe_sample returned false for this allocation
83 if (interval_to_next_sample_ >= 0) {
84 return false;
85 }
86 reset_interval_to_next_sample(alloc_size);
87 return sampling_interval_ != 0; // sampling interval 0 means off
88 }
89
90 uint64_t sampling_interval() const { return sampling_interval_; }
91
93 size_t sample_size(size_t allocation_size) const {
94 return std::max(allocation_size, sampling_interval_);
95 }

98 struct disable_sampling_temporarily {
99 disable_sampling_temporarily() = default;
100 disable_sampling_temporarily(sampler& sampler)
101 : sampler_(&sampler)
102 , previous_sampling_interval_(sampler_->sampling_interval_)
103 , previous_sampling_rate_(sampler_->sampling_rate_)
104 , previous_interval_to_next_sample_(sampler_->interval_to_next_sample_) {
105 sampler_->set_sampling_interval(0);
106 }
107
108 ~disable_sampling_temporarily() {
109 if (sampler_) {
110 sampler_->sampling_interval_ = previous_sampling_interval_;
111 sampler_->sampling_rate_ = previous_sampling_rate_;
112 sampler_->interval_to_next_sample_ = previous_interval_to_next_sample_;
113 }
114 }

116 private:
117 sampler* sampler_ = nullptr;
118 uint64_t previous_sampling_interval_ = 0; // sampling interval before pausing
119 double previous_sampling_rate_ = 0; // sampling rate before pausing
120 int64_t previous_interval_to_next_sample_ = 0; // interval to next sample before pausing
121 };

129 disable_sampling_temporarily pause_sampling() {
130 return disable_sampling_temporarily(*this);
131 }

133private:
137 void reset_interval_to_next_sample(size_t alloc_size)
138 {
139 if (sampling_interval_ == 0) { // sampling is off
140 interval_to_next_sample_ = std::numeric_limits<int64_t>::max();
141 }
142 else {
143 // Large allocations we will just consider in whole. This avoids
144 // having to sample the distribution too many times if a large alloc
145 // took us very negative we just add the alloc size back on
146 if (alloc_size > sampling_interval_) {
147 interval_to_next_sample_ += alloc_size;
148 }
149 else {
150 while (interval_to_next_sample_ < 0) {
151 interval_to_next_sample_ += next_sampling_interval();
152 }
153 }
154 }
155 }
156
157 int64_t next_sampling_interval() {
158 if (!random_gen_) {
159 // Seeded on first use: std::random_device and std::mt19937_64 are
160 // not constant-initializable, and sampling is usually never enabled.
161 random_gen_.emplace(std::random_device()());
162 }
163 std::exponential_distribution<double> dist(sampling_rate_);
164 int64_t next = static_cast<int64_t>(dist(*random_gen_));
165 // We approximate the geometric distribution using an exponential
166 // distribution.
167 return next;
168 }
169
170 uint64_t sampling_interval_ = 0; // Sample every N bytes ; 0 means off
171 double sampling_rate_ = 0; // 1 / sampling_interval_ ; used by the exp distribution
172 // How many bytes remain to be allocated before we take a sample.
173 // Specifically, if this member has value N, a sample will be taken of the allocation
174 // that allocates the Nth+1 byte.
175 //
176 // Sampling is off initially, so this is set very large: see
177 // set_sampling_interval().
178 int64_t interval_to_next_sample_ = std::numeric_limits<int64_t>::max();
179 std::optional<std::mt19937_64> random_gen_; // seeded on first use
180};
