126// RPC call that passes stream connection id as a parameter
127// may arrive to a different shard from where the stream connection
128// was opened, so the connection id is not known to a server that handles
129// the RPC call. The shard that the stream connection belong to is know
130// since it is a part of connection id, but this is not enough to locate
131// a server instance the connection belongs to if there are more than one
132// server on the shard. Stream domain parameter is here to help with that.
133// Different servers on all shards logically belonging to the same service should
134// belong to the same streaming domain. Only one server on each shard can belong to
135// a particulr streaming domain.
136class streaming_domain_type {
137 uint64_t _id;
138public:
139 explicit streaming_domain_type(uint64_t id) : _id(id) {}
140 bool operator==(const streaming_domain_type& o) const {
141 return _id == o._id;
142 }
143 friend struct std::hash<streaming_domain_type>;
144 friend std::ostream& operator<<(std::ostream&, const streaming_domain_type&);
145};

411template <typename T>
412requires std::is_base_of_v<bi::slist_base_hook<>, T>
413class batched_queue {
414 using list_type = boost::intrusive::slist<T, boost::intrusive::cache_last<true>, boost::intrusive::constant_time_size<false>>;
415
416 std::function<future<>(T*)> _process_func;
417 shard_id _processing_shard;
418 list_type _queue;
419 list_type _cur_batch;
420 future<> _process_fut = make_ready_future();
421
422public:
423 batched_queue(std::function<future<>(T*)> process_func, shard_id processing_shard)
424 : _process_func(std::move(process_func))
425 , _processing_shard(processing_shard)
426 {}
427
428 ~batched_queue() {
429 assert(_process_fut.available());
430 }
431
432 future<> stop() noexcept {
433 return std::exchange(_process_fut, make_ready_future());
434 }
435
436 void enqueue(T* buf) noexcept {
437 _queue.push_back(*buf);
438 if (_process_fut.available()) {
439 _process_fut = process_loop();
440 }
441 }
442
443 future<> process_loop() {
444 return seastar::do_until([this] { return _queue.empty(); }, [this] {
445 _cur_batch = std::exchange(_queue, list_type());
446 return smp::submit_to(_processing_shard, [this] {
447 return seastar::do_until([this] { return _cur_batch.empty(); }, [this] {
448 auto* buf = &_cur_batch.front();
449 _cur_batch.pop_front();
450 return _process_func(buf);
451 });
452 });
453 });
454 }
455};
