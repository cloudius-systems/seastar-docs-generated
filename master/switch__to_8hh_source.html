58 task* _task = nullptr;
59public:

63 { }

68 bool await_ready() const noexcept {

72 template<typename T>
73 void await_suspend(std::coroutine_handle<T> hndl SEASTAR_COROUTINE_LOC_PARAM) noexcept {
74 SEASTAR_COROUTINE_LOC_STORE(hndl.promise());
75 auto& t = hndl.promise();
76 t.set_scheduling_group(_switch_to_sg);
77 _task = &t;
