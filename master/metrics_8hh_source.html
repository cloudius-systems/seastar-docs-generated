325class metric_value {
326public:
327 std::variant<double, histogram> u;
328 data_type _type;
329 data_type type() const {
330 return _type;
331 }
332
333 double d() const {
334 return std::get<double>(u);
335 }
336
337 uint64_t ui() const {
338 auto d = std::get<double>(u);
339 if (d >= 0 && d <= double(std::numeric_limits<long>::max())) {
340 return lround(d);
341 } else {
342 // double value is out of range or NaN or Inf
343 ulong_conversion_error(d);
344 return 0;
345 }
346 }
347
348 int64_t i() const {
349 auto d = std::get<double>(u);
350 if (d >= double(std::numeric_limits<long>::min()) && d <= double(std::numeric_limits<long>::max())) {
351 return lround(d);
352 } else {
353 // double value is out of range or NaN or Inf
354 ulong_conversion_error(d);
355 return 0;
356 }
357 }
358
359 metric_value()
360 : _type(data_type::GAUGE) {
361 }
362
363 metric_value(histogram&& h, data_type t = data_type::HISTOGRAM) :
364 u(std::move(h)), _type(t) {
365 }
366 metric_value(const histogram& h, data_type t = data_type::HISTOGRAM) :
367 u(h), _type(t) {
368 }
369
370 metric_value(double d, data_type t)
371 : u(d), _type(t) {
372 }
373
374 metric_value& operator+=(const metric_value& c);
375
376 metric_value operator+(const metric_value& c) {
377 return metric_value(*this) += c;
378 }
379
380 const histogram& get_histogram() const {
381 return std::get<histogram>(u);
382 }
383
390 bool is_empty() const noexcept {
391 return ((_type == data_type::HISTOGRAM || _type == data_type::SUMMARY) && get_histogram().sample_count == 0) ||
392 ((_type == data_type::COUNTER || _type == data_type::REAL_COUNTER) && d() == 0);
393 }
394private:
395 static void ulong_conversion_error(double d);
396};
