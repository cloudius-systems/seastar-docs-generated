406// Helper to convert lvalue reference functions to rvalue.
407// This is used early in then()-alike entry points to convert lvalue references into
408// rvalues (which are used to recursively call back into the same entry point) to
409// avoid having to handle both lvalues and rvalues downstream: we convert them all
410// to rvalues. This helper handles both references to callable objects such as lambdas,
411// copying them, and references to functions, decaying them to function pointers.
412template <typename Func>
413static constexpr auto func_to_rvalue(Func&& func) {
414 static_assert(std::is_lvalue_reference_v<Func>, "call with lvalue reference");
415 if constexpr (std::is_function_v<std::remove_reference_t<Func>>) {
416 // return decayed function pointer
417 return +func;
418 } else {
419 // return a copy (used as an rvalue by the called)
420 return std::forward<Func>(func);
421 }
422}

425//
426// A future/promise pair maintain one logical value (a future_state).
427// There are up to three places that can store it, but only one is
428// active at any time.
429//
430// - in the promise _local_state member variable
431//
432// This is necessary because a promise is created first and there
433// would be nowhere else to put the value.
434//
435// - in the future _state variable
436//
437// This is used anytime a future exists and then has not been called
438// yet. This guarantees a simple access to the value for any code
439// that already has a future.
440//
441// - in the task associated with the .then() clause (after .then() is called,
442// if a value was not set)
443//
444//
445// The promise maintains a pointer to the state, which is modified as
446// the state moves to a new location due to events (such as .then() or
447// get_future being called) or due to the promise or future being
448// moved around.
449//
450
451// non templated base class to reduce code duplication
452struct future_state_base {
453 static_assert(sizeof(std::exception_ptr) == sizeof(void*), "exception_ptr not a pointer");
454 enum class state : uintptr_t {
455 invalid = 0,
456 future = 1,
457 // the substate is intended to decouple the run-time prevention
458 // for duplicative result extraction (calling e.g. then() twice
459 // ends up in abandoned()) from the wrapped object's destruction
460 // handling which is orchestrated by future_state. Instead of
461 // creating a temporary future_state just for the sake of setting
462 // the "invalid" in the source instance, result_unavailable can
463 // be set to ensure future_state_base::available() returns false.
464 result_unavailable = 2,
465 result = 3,
466 exception_min = 4, // or anything greater
467 };
468 union any {
469 any() noexcept { st = state::future; }
470 any(state s) noexcept { st = s; }
471 void set_exception(std::exception_ptr&& e) noexcept {
472 new (&ex) std::exception_ptr(std::move(e));
473 assert(st >= state::exception_min);
474 }
475 any(std::exception_ptr&& e) noexcept {
476 set_exception(std::move(e));
477 }
478 // From a users' perspective, a result_unavailable is not valid
479 bool valid() const noexcept { return st != state::invalid && st != state::result_unavailable; }
480 bool available() const noexcept { return st == state::result || st >= state::exception_min; }
481 bool failed() const noexcept { return __builtin_expect(st >= state::exception_min, false); }
482 void check_failure() noexcept;
483 ~any() noexcept { }
484 std::exception_ptr take_exception() noexcept {
485 std::exception_ptr ret(std::move(ex));
486 // Unfortunately in libstdc++ ~exception_ptr is defined out of line. We know that it does nothing for
487 // moved out values, so we omit calling it. This is critical for the code quality produced for this
488 // function. Without the out of line call, gcc can figure out that both sides of the if produce
489 // identical code and merges them.if
490 // We don't make any assumptions about other c++ libraries.
491 // There is request with gcc to define it inline: https://gcc.gnu.org/bugzilla/show_bug.cgi?id=90295
492#ifndef __GLIBCXX__
493 ex.~exception_ptr();
494#endif
495 st = state::invalid;
496 return ret;
497 }
498 void move_it(any&& x) noexcept {
499#ifdef __GLIBCXX__
500 // Unfortunally gcc cannot fully optimize the regular
501 // implementation:
502 // https://gcc.gnu.org/bugzilla/show_bug.cgi?id=95014
503 // Given what we know about the libstdc++ implementation
504 // (see the comment in take_exception), we can just
505 // memmove and zero x. We use memmove to guarantee
506 // vaild results if &x == this.
507 memmove(static_cast<void*>(this), &x, sizeof(any));
508 x.st = state::invalid;
509#else
510 if (x.st < state::exception_min) {
511 st = x.st;
512 x.st = state::invalid;
513 } else {
514 new (&ex) std::exception_ptr(x.take_exception());
515 }
516#endif
517 }
518 any(any&& x) noexcept {
519 move_it(std::move(x));
520 }
521 any& operator=(any&& x) noexcept {
522 check_failure();
523 // If this is a self move assignment, check_failure
524 // guarantees that we don't have an exception and calling
525 // move_it is safe.
526 move_it(std::move(x));
527 return *this;
528 }
529 bool has_result() const noexcept {
530 return st == state::result || st == state::result_unavailable;
531 }
532 state st;
533 std::exception_ptr ex;
534 } _u;
535
536 future_state_base() noexcept = default;
537 future_state_base(state st) noexcept : _u(st) { }
538 future_state_base(std::exception_ptr&& ex) noexcept : _u(std::move(ex)) { }
539 future_state_base(future_state_base&& x) noexcept : _u(std::move(x._u)) { }

542 // protected instead of virtual.
543protected:

549 ~future_state_base() noexcept = default;

554public:
555
556 bool valid() const noexcept { return _u.valid(); }
557 bool available() const noexcept { return _u.available(); }
558 bool failed() const noexcept { return _u.failed(); }
559
560 void ignore() noexcept;
561
562 void set_exception(std::exception_ptr&& ex) noexcept {
563 assert(_u.st == state::future);
564 _u.set_exception(std::move(ex));
565 }
566 future_state_base& operator=(future_state_base&& x) noexcept = default;
567 void set_exception(future_state_base&& state) noexcept {

577 assert(_u.st >= state::exception_min);
578 return _u.ex;

612template <typename T>
613struct future_state : public future_state_base, private internal::uninitialized_wrapper<T> {
614 static constexpr bool copy_noexcept = std::is_nothrow_copy_constructible_v<T>;
615 static constexpr bool has_trivial_move_and_destroy = internal::is_trivially_move_constructible_and_destructible<T>::value;
616 static_assert(std::is_nothrow_move_constructible_v<T>,
617 "Types must be no-throw move constructible");
618 static_assert(std::is_nothrow_destructible_v<T>,
619 "Types must be no-throw destructible");
620 future_state() noexcept = default;
621 void move_it(future_state&& x) noexcept {
622 if constexpr (has_trivial_move_and_destroy) {
623#pragma GCC diagnostic push
624 // This function may copy uninitialized memory, such as when
625 // creating an uninitialized promise and calling get_future()
626 // on it. Gcc 12 started to catch some simple cases of this
627 // at compile time, so we need to tell it that it's fine.
628#pragma GCC diagnostic ignored "-Wuninitialized"
629 memmove(reinterpret_cast<char*>(&this->uninitialized_get()),
630 &x.uninitialized_get(),
631 internal::used_size<internal::maybe_wrap_ref<T>>::value);
632#pragma GCC diagnostic pop
633 } else if (_u.has_result()) {
634 this->uninitialized_set(std::move(x.uninitialized_get()));
635 std::destroy_at(&x.uninitialized_get());
636 }
637 }

642 }

645 if (_u.has_result()) {
646 std::destroy_at(&this->uninitialized_get());
647 } else {
648 _u.check_failure();
649 }

652 ~future_state() noexcept {
653 clear();

659 // safe to call move_it.

813template <typename Promise, typename Func, typename Wrapper, typename T = void>
814struct continuation final : continuation_base_with_promise<Promise, T> {
815 // Func is the original function passed to then/then_wrapped. The
816 // Wrapper is a helper function that implements the specific logic
817 // needed by then/then_wrapped. We call the wrapper passing it the
818 // original function, promise and state.
819 // Note that if Func's move constructor throws, this will call
820 // std::unexpected. We could try to require Func to be nothrow
821 // move constructible, but that will cause a lot of churn. Since
822 // we can't support a failure to create a continuation, calling
823 // std::unexpected as close to the failure as possible is the best
824 // we can do.

826 : continuation_base_with_promise<Promise, T>(std::move(pr))
827 , _func(std::move(func))

830 try {
831 _wrapper(std::move(this->_pr), std::move(_func), std::move(this->_state));
832 } catch (...) {
833 this->_pr.set_to_current_exception();
834 }
835 delete this;
836 }
837 Func _func;
