362// Helper to convert lvalue reference functions to rvalue.
363// This is used early in then()-alike entry points to convert lvalue references into
364// rvalues (which are used to recursively call back into the same entry point) to
365// avoid having to handle both lvalues and rvalues downstream: we convert them all
366// to rvalues. This helper handles both references to callable objects such as lambdas,
367// copying them, and references to functions, decaying them to function pointers.
368template <typename Func>
369static constexpr auto func_to_rvalue(Func&& func) {
370 static_assert(std::is_lvalue_reference_v<Func>, "call with lvalue reference");
371 if constexpr (std::is_function_v<std::remove_reference_t<Func>>) {
372 // return decayed function pointer
373 return +func;
374 } else {
375 // return a copy (used as an rvalue by the called)
376 return std::forward<Func>(func);
377 }
378}

381//
382// A future/promise pair maintain one logical value (a future_state).
383// There are up to three places that can store it, but only one is
384// active at any time.
385//
386// - in the promise _local_state member variable
387//
388// This is necessary because a promise is created first and there
389// would be nowhere else to put the value.
390//
391// - in the future _state variable
392//
393// This is used anytime a future exists and then has not been called
394// yet. This guarantees a simple access to the value for any code
395// that already has a future.
396//
397// - in the task associated with the .then() clause (after .then() is called,
398// if a value was not set)
399//
400//
401// The promise maintains a pointer to the state, which is modified as
402// the state moves to a new location due to events (such as .then() or
403// get_future being called) or due to the promise or future being
404// moved around.
405//
406
407// non templated base class to reduce code duplication
408struct future_state_base {
409 static_assert(sizeof(std::exception_ptr) == sizeof(void*), "exception_ptr not a pointer");
410 enum class state : uintptr_t {
411 invalid = 0,
412 future = 1,
413 // the substate is intended to decouple the run-time prevention
414 // for duplicative result extraction (calling e.g. then() twice
415 // ends up in abandoned()) from the wrapped object's destruction
416 // handling which is orchestrated by future_state. Instead of
417 // creating a temporary future_state just for the sake of setting
418 // the "invalid" in the source instance, result_unavailable can
419 // be set to ensure future_state_base::available() returns false.
420 result_unavailable = 2,
421 result = 3,
422 exception_min = 4, // or anything greater
423 };
424 union any {
425 any() noexcept { st = state::future; }
426 any(state s) noexcept { st = s; }
427 void set_exception(std::exception_ptr&& e) noexcept {
428 new (&ex) std::exception_ptr(std::move(e));
429 assert(st >= state::exception_min);
430 }
431 any(std::exception_ptr&& e) noexcept {
432 set_exception(std::move(e));
433 }
434 // From a users' perspective, a result_unavailable is not valid
435 bool valid() const noexcept { return st != state::invalid && st != state::result_unavailable; }
436 bool available() const noexcept { return st == state::result || st >= state::exception_min; }
437 bool failed() const noexcept { return __builtin_expect(st >= state::exception_min, false); }
438 void check_failure() noexcept;
439 ~any() noexcept { }
440 std::exception_ptr take_exception() noexcept {
441 std::exception_ptr ret(std::move(ex));
442 // Unfortunately in libstdc++ ~exception_ptr is defined out of line. We know that it does nothing for
443 // moved out values, so we omit calling it. This is critical for the code quality produced for this
444 // function. Without the out of line call, gcc can figure out that both sides of the if produce
445 // identical code and merges them.if
446 // We don't make any assumptions about other c++ libraries.
447 // There is request with gcc to define it inline: https://gcc.gnu.org/bugzilla/show_bug.cgi?id=90295
448#ifndef __GLIBCXX__
449 ex.~exception_ptr();
450#endif
451 st = state::invalid;
452 return ret;
453 }
454 void move_it(any&& x) noexcept {
455#ifdef __GLIBCXX__
456 // Unfortunally gcc cannot fully optimize the regular
457 // implementation:
458 // https://gcc.gnu.org/bugzilla/show_bug.cgi?id=95014
459 // Given what we know about the libstdc++ implementation
460 // (see the comment in take_exception), we can just
461 // memmove and zero x. We use memmove to guarantee
462 // vaild results if &x == this.
463 memmove(static_cast<void*>(this), &x, sizeof(any));
464 x.st = state::invalid;
465#else
466 if (x.st < state::exception_min) {
467 st = x.st;
468 x.st = state::invalid;
469 } else {
470 new (&ex) std::exception_ptr(x.take_exception());
471 }
472#endif
473 }
474 any(any&& x) noexcept {
475 move_it(std::move(x));
476 }
477 any& operator=(any&& x) noexcept {
478 check_failure();
479 // If this is a self move assignment, check_failure
480 // guarantees that we don't have an exception and calling
481 // move_it is safe.
482 move_it(std::move(x));
483 return *this;
484 }
485 bool has_result() const noexcept {
486 return st == state::result || st == state::result_unavailable;
487 }
488 state st;
489 std::exception_ptr ex;
490 } _u;
491
492 future_state_base() noexcept = default;
493 future_state_base(state st) noexcept : _u(st) { }
494 future_state_base(std::exception_ptr&& ex) noexcept : _u(std::move(ex)) { }
495 future_state_base(future_state_base&& x) noexcept : _u(std::move(x._u)) { }
496
497 // We never need to destruct this polymorphicly, so we can make it
498 // protected instead of virtual.
499protected:

505 ~future_state_base() noexcept = default;

507 void rethrow_exception() &&;
508 void rethrow_exception() const&;

510public:
511
512 bool valid() const noexcept { return _u.valid(); }
513 bool available() const noexcept { return _u.available(); }
514 bool failed() const noexcept { return _u.failed(); }
515
516 void ignore() noexcept;
517
518 void set_exception(std::exception_ptr&& ex) noexcept {
519 assert(_u.st == state::future);
520 _u.set_exception(std::move(ex));
521 }
522 future_state_base& operator=(future_state_base&& x) noexcept = default;
523 void set_exception(future_state_base&& state) noexcept {
524 assert(_u.st == state::future);

530 return _u.take_exception();
531 }
532 const std::exception_ptr& get_exception() const& noexcept {
533 assert(_u.st >= state::exception_min);
534 return _u.ex;

564template <typename T>
565struct future_state : public future_state_base, private internal::uninitialized_wrapper<T> {
566 static constexpr bool copy_noexcept = std::is_nothrow_copy_constructible_v<T>;
567 static constexpr bool has_trivial_move_and_destroy = internal::is_trivially_move_constructible_and_destructible<T>::value;
568 static_assert(std::is_nothrow_move_constructible_v<T>,
569 "Types must be no-throw move constructible");
570 static_assert(std::is_nothrow_destructible_v<T>,
571 "Types must be no-throw destructible");
572 future_state() noexcept = default;
573 void move_it(future_state&& x) noexcept {
574 if constexpr (has_trivial_move_and_destroy) {
575#pragma GCC diagnostic push
576 // This function may copy uninitialized memory, such as when
577 // creating an uninitialized promise and calling get_future()
578 // on it. Gcc 12 started to catch some simple cases of this
579 // at compile time, so we need to tell it that it's fine.
580#pragma GCC diagnostic ignored "-Wuninitialized"
581 memmove(reinterpret_cast<char*>(&this->uninitialized_get()),
582 &x.uninitialized_get(),
583 internal::used_size<internal::maybe_wrap_ref<T>>::value);
584#pragma GCC diagnostic pop
585 } else if (_u.has_result()) {
586 this->uninitialized_set(std::move(x.uninitialized_get()));
587 std::destroy_at(&x.uninitialized_get());
588 }
589 }
590
591 [[gnu::always_inline]]
592 future_state(future_state&& x) noexcept : future_state_base(std::move(x)) {
593 move_it(std::move(x));
594 }
595
596 void clear() noexcept {
597 if (_u.has_result()) {
598 std::destroy_at(&this->uninitialized_get());
599 } else {
600 _u.check_failure();
601 }
602 }
603 __attribute__((always_inline))
604 ~future_state() noexcept {
605 clear();
606 }
607 future_state& operator=(future_state&& x) noexcept {
608 clear();
609 future_state_base::operator=(std::move(x));
610 // If &x == this, _u.st is now state::invalid and so it is
611 // safe to call move_it.
612 move_it(std::move(x));
613 return *this;
614 }
615 template <typename... A>
616 future_state(ready_future_marker, A&&... a) noexcept : future_state_base(state::result) {
617 try {
618 this->uninitialized_set(std::forward<A>(a)...);
619 } catch (...) {
620 new (this) future_state(current_exception_future_marker());
621 }
622 }
623 template <typename... A>
624 void set(A&&... a) noexcept {
625 assert(_u.st == state::future);
626 new (this) future_state(ready_future_marker(), std::forward<A>(a)...);
627 }
628 future_state(exception_future_marker, std::exception_ptr&& ex) noexcept : future_state_base(std::move(ex)) { }
629 future_state(exception_future_marker, future_state_base&& state) noexcept : future_state_base(std::move(state)) { }
630 future_state(current_exception_future_marker m) noexcept : future_state_base(m) { }
631 future_state(nested_exception_marker m, future_state_base&& old) noexcept : future_state_base(m, std::move(old)) { }
632 future_state(nested_exception_marker m, future_state_base&& n, future_state_base&& old) noexcept : future_state_base(m, std::move(n), std::move(old)) { }
633 T&& get_value() && noexcept {
634 assert(_u.st == state::result);
635 return static_cast<T&&>(this->uninitialized_get());
636 }
637 T&& take_value() && noexcept {
638 assert(_u.st == state::result);
639 _u.st = state::result_unavailable;
640 return static_cast<T&&>(this->uninitialized_get());
641 }
642 template<std::copy_constructible U = T>
643 const U& get_value() const& noexcept(copy_noexcept) {
644 assert(_u.st == state::result);
645 return this->uninitialized_get();
646 }
647 T&& take() && {
648 assert(available());
649 if (_u.st >= state::exception_min) {
650 std::move(*this).rethrow_exception();
651 }
652 _u.st = state::result_unavailable;
653 return static_cast<T&&>(this->uninitialized_get());
654 }
655 T&& get() && {
656 assert(available());
657 if (_u.st >= state::exception_min) {
658 std::move(*this).rethrow_exception();
659 }
660 return static_cast<T&&>(this->uninitialized_get());
661 }
662 const T& get() const& {
663 assert(available());
664 if (_u.st >= state::exception_min) {
665 rethrow_exception();
666 }
667 return this->uninitialized_get();
668 }
669 using get0_return_type = typename internal::get0_return_type<T>::type;
670 static get0_return_type get0(T&& x) {
671 return internal::get0_return_type<T>::get0(std::move(x));

724template <typename Promise, typename Func, typename Wrapper, typename T = void>
725struct continuation final : continuation_base_with_promise<Promise, T> {
726 // Func is the original function passed to then/then_wrapped. The
727 // Wrapper is a helper function that implements the specific logic
728 // needed by then/then_wrapped. We call the wrapper passing it the
729 // original function, promise and state.
730 // Note that if Func's move constructor throws, this will call
731 // std::unexpected. We could try to require Func to be nothrow
732 // move constructible, but that will cause a lot of churn. Since
733 // we can't support a failure to create a continuation, calling
734 // std::unexpected as close to the failure as possible is the best
735 // we can do.
736 continuation(Promise&& pr, Func&& func, Wrapper&& wrapper) noexcept
737 : continuation_base_with_promise<Promise, T>(std::move(pr))
738 , _func(std::move(func))
739 , _wrapper(std::move(wrapper)) {}
740 virtual void run_and_dispose() noexcept override {
741 try {
742 _wrapper(std::move(this->_pr), std::move(_func), std::move(this->_state));
743 } catch (...) {
744 this->_pr.set_to_current_exception();
745 }
746 delete this;
747 }
748 Func _func;
749 [[no_unique_address]] Wrapper _wrapper;
750};
