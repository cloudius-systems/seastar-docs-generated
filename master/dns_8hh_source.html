53struct hostent {

60 // https://datatracker.ietf.org/doc/html/rfc1035#section-2.3.4
61 // https://datatracker.ietf.org/doc/html/rfc1035#section-4.1.3
62 // https://datatracker.ietf.org/doc/html/rfc2181#section-8
63 std::chrono::seconds ttl{std::numeric_limits<signed int>::max()};
64 };
