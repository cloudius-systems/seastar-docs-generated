100class datagram final {
101private:
102 std::unique_ptr<datagram_impl> _impl;
103 // The get_data() below will need to tell
104 // - _p wasn't initialized from get_buffers() span
105 // - _p was initialized, but was then release()-d by caller
106 // from each other. thus std::optional
107 std::optional<net::packet> _p;
108public:
109 datagram(std::unique_ptr<datagram_impl>&& impl) noexcept : _impl(std::move(impl)) {};
110 socket_address get_src() { return _impl->get_src(); }
111 socket_address get_dst() { return _impl->get_dst(); }
112 uint16_t get_dst_port() { return _impl->get_dst_port(); }
113 [[deprecated("Use get_buf() instead")]]
114 packet& get_data() {
115 if (!_p) {
116 _p.emplace(_impl->get_buffers());
117 }
118 return _p.value();
119 }
120 std::span<temporary_buffer<char>> get_buffers() { return _impl->get_buffers(); }
121};

348class server_socket {
349 std::unique_ptr<net::server_socket_impl> _ssi;
350 bool _aborted = false;
351public:
352 enum class load_balancing_algorithm {
353 // This algorithm tries to distribute all connections equally between all shards.
354 // It does this by sending new connections to a shard with smallest amount of connections.
355 connection_distribution,
356 // This algorithm distributes new connection based on peer's tcp port. Destination shard
357 // is calculated as a port number modulo number of shards. This allows a client to connect
358 // to a specific shard in a server given it knows how many shards server has by choosing
359 // src port number accordingly.
360 port,
361 // This algorithm distributes all new connections to listen_options::fixed_cpu shard only.
362 fixed,
363 default_ = connection_distribution
364 };
366 server_socket() noexcept;
368 explicit server_socket(std::unique_ptr<net::server_socket_impl> ssi) noexcept;
371 server_socket(server_socket&& ss) noexcept;
372 ~server_socket();
374 server_socket& operator=(server_socket&& cs) noexcept;
375

384
389 void abort_accept();
390

399
407 explicit operator bool() const noexcept {
408 return static_cast<bool>(_ssi);
409 }
