96class datagram final {
97private:
98 std::unique_ptr<datagram_impl> _impl;
99 // The get_data() below will need to tell
100 // - _p wasn't initialized from get_buffers() span
101 // - _p was initialized, but was then release()-d by caller
102 // from each other. thus std::optional
103 std::optional<net::packet> _p;
104public:
105 datagram(std::unique_ptr<datagram_impl>&& impl) noexcept : _impl(std::move(impl)) {};
106 socket_address get_src() { return _impl->get_src(); }
107 socket_address get_dst() { return _impl->get_dst(); }
108 uint16_t get_dst_port() { return _impl->get_dst_port(); }
109 [[deprecated("Use get_buf() instead")]]
110 packet& get_data() {
111 if (!_p) {
112 _p.emplace(_impl->get_buffers());
113 }
114 return _p.value();
115 }
116 std::span<temporary_buffer<char>> get_buffers() { return _impl->get_buffers(); }
117};

346 bool _aborted = false;
347public:
348 enum class load_balancing_algorithm {
349 // This algorithm tries to distribute all connections equally between all shards.
350 // It does this by sending new connections to a shard with smallest amount of connections.
351 connection_distribution,
352 // This algorithm distributes new connection based on peer's tcp port. Destination shard
353 // is calculated as a port number modulo number of shards. This allows a client to connect
354 // to a specific shard in a server given it knows how many shards server has by choosing
355 // src port number accordingly.
356 port,
357 // This algorithm distributes all new connections to listen_options::fixed_cpu shard only.
358 fixed,
359 default_ = connection_distribution
360 };
