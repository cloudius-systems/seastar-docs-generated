57class pollable_fd_state {
58 unsigned _refs = 0;
59public:
60 struct speculation {
61 int events = 0;
62 explicit speculation(int epoll_events_guessed = 0) : events(epoll_events_guessed) {}
63 };
64 pollable_fd_state(const pollable_fd_state&) = delete;
65 void operator=(const pollable_fd_state&) = delete;
73 void speculate_epoll(int events) { events_known |= events; }
78 bool take_speculation(int events) {
79 // invalidate the speculation set by the last speculate_epoll() call,
80 if (events_known & events) {
81 events_known &= ~events;
82 return true;
83 }
84 return false;
85 }
86 file_desc fd;
87 bool events_rw = false; // single consumer for both read and write (accept())
88 unsigned shutdown_mask = 0; // For udp, there is no shutdown indication from the kernel
89 int events_requested = 0; // wanted by pollin/pollout promises
90 int events_epoll = 0; // installed in epoll
91 int events_known = 0; // returned from epoll
