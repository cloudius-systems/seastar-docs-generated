42template<typename Timer, boost::intrusive::list_member_hook<> Timer::*link>
43class timer_set {
44public:
45 using time_point = typename Timer::time_point;
46 using timer_list_t = boost::intrusive::list<Timer, boost::intrusive::member_hook<Timer, boost::intrusive::list_member_hook<>, link>>;
47private:
48 using duration = typename Timer::duration;
49 using timestamp_t = typename Timer::duration::rep;
50
51 static constexpr timestamp_t max_timestamp = std::numeric_limits<timestamp_t>::max();
52 static constexpr int timestamp_bits = std::numeric_limits<timestamp_t>::digits;
53
54 // The last bucket is reserved for active timers with timeout <= _last.
55 static constexpr int n_buckets = timestamp_bits + 1;
56
57 std::array<timer_list_t, n_buckets> _buckets;
58 timestamp_t _last;
59 timestamp_t _next;
60
61 std::bitset<n_buckets> _non_empty_buckets;
62private:
63 static timestamp_t get_timestamp(time_point _time_point) noexcept
64 {
65 return _time_point.time_since_epoch().count();
66 }
67
68 static timestamp_t get_timestamp(Timer& timer) noexcept
69 {
70 return get_timestamp(timer.get_timeout());
71 }

73 int get_index(timestamp_t timestamp) const noexcept
74 {
75 if (timestamp <= _last) {
76 return n_buckets - 1;
77 }
78
79 auto index = bitsets::count_leading_zeros(timestamp ^ _last);
80 SEASTAR_ASSERT(index < n_buckets - 1);
81 return index;
82 }
83
84 int get_index(Timer& timer) const noexcept
85 {
86 return get_index(get_timestamp(timer));

89 int get_last_non_empty_bucket() const noexcept

102 ~timer_set() {

128 bool insert(Timer& timer) noexcept
129 {
130 auto timestamp = get_timestamp(timer);
131 auto index = get_index(timestamp);
132
133 _buckets[index].push_back(timer);
134 _non_empty_buckets[index] = true;
135
136 if (timestamp < _next) {
137 _next = timestamp;
138 return true;
139 }
140 return false;

154 void remove(Timer& timer) noexcept
155 {

191 timer_list_t expire(time_point now) noexcept
192 {
193 timer_list_t exp;
194 auto timestamp = get_timestamp(now);
195
196 if (timestamp < _last) {
197 abort();
198 }
199
200 auto index = get_index(timestamp);
201
202 for (int i : bitsets::for_each_set(_non_empty_buckets, index + 1)) {
203 exp.splice(exp.end(), _buckets[i]);
204 _non_empty_buckets[i] = false;
205 }
206
207 _last = timestamp;
208 _next = max_timestamp;
209
210 auto& list = _buckets[index];
211 while (!list.empty()) {
212 auto& timer = *list.begin();
213 list.pop_front();
214 if (timer.get_timeout() <= now) {
215 exp.push_back(timer);
216 } else {
217 insert(timer);
218 }
219 }
220
221 _non_empty_buckets[index] = !list.empty();
222
223 if (_next == max_timestamp && _non_empty_buckets.any()) {
224 for (auto& timer : _buckets[get_last_non_empty_bucket()]) {
225 _next = std::min(_next, get_timestamp(timer));
226 }
227 }
228 return exp;
229 }

231 void complete(timer_list_t& expired_timers) noexcept {
232 expired_timers = expire(this->now());
233 for (auto& t : expired_timers) {
234 t._expired = true;
235 }
236 const auto prev_sg = current_scheduling_group();
237 while (!expired_timers.empty()) {
238 auto t = &*expired_timers.begin();
239 expired_timers.pop_front();
240 t->_queued = false;
241 if (t->_armed) {
242 t->_armed = false;
243 if (t->_period) {
244 t->readd_periodic();
245 }
246 try {
247 *internal::current_scheduling_group_ptr() = t->_sg;
248 t->_callback();
249 } catch (...) {
250 internal::log_timer_callback_exception(std::current_exception());
251 }
252 }
253 }
254 // complete_timers() can be called from the context of run_tasks()
255 // as well so we need to restore the previous scheduling group (set by run_tasks()).
256 *internal::current_scheduling_group_ptr() = prev_sg;
257 }
