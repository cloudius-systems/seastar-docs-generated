201// Determines when reclaimer can be invoked
202enum class reclaimer_scope {
203 //
204 // Reclaimer is only invoked in its own fiber. That fiber will be
205 // given higher priority than regular application fibers.
206 //
207 async,
208
209 //
210 // Reclaimer may be invoked synchronously with allocation.
211 // It may also be invoked in async scope.
212 //
213 // Reclaimer may invoke allocation, though it is discouraged because
214 // the system may be low on memory and such allocations may fail.
215 // Reclaimers which allocate should be prepared for re-entry.
216 //
217 sync
218};

220class reclaimer {
221public:
222 struct request {
223 // The number of bytes which is needed to be released.
224 // The reclaimer can release a different amount.
225 // If less is released then the reclaimer may be invoked again.
226 size_t bytes_to_reclaim;
227 };
228 using reclaim_fn = std::function<reclaiming_result ()>;
229private:
230 std::function<reclaiming_result (request)> _reclaim;
231 reclaimer_scope _scope;
232public:
233 // Installs new reclaimer which will be invoked when system is falling
234 // low on memory. 'scope' determines when reclaimer can be executed.
235 reclaimer(std::function<reclaiming_result ()> reclaim, reclaimer_scope scope = reclaimer_scope::async);
236 reclaimer(std::function<reclaiming_result (request)> reclaim, reclaimer_scope scope = reclaimer_scope::async);
237 ~reclaimer();
238 reclaiming_result do_reclaim(size_t bytes_to_reclaim) { return _reclaim(request{bytes_to_reclaim}); }
239 reclaimer_scope scope() const { return _scope; }
240};

429struct allocation_site {
430 mutable size_t count = 0;
431 mutable size_t size = 0;
432 simple_backtrace backtrace;
433
434 // All allocation sites are linked to each other. This can be used for easy
435 // iteration across them in gdb scripts where it's difficult to work with
436 // the C++ data structures.
437 mutable const allocation_site* next = nullptr; // next allocation site in the chain
438 mutable const allocation_site* prev = nullptr; // previous allocation site in the chain
439
440 bool operator==(const allocation_site& o) const {
441 return backtrace == o.backtrace;
442 }
443
444 bool operator!=(const allocation_site& o) const {
445 return !(*this == o);
446 }
447};
