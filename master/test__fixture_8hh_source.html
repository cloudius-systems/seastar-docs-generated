37 template<typename T>
38future<> conditional_invoke_setup(T& t) {
39 if constexpr (boost::unit_test::impl_fixture::has_setup<T>::value) {
40 return futurize_invoke(std::mem_fn(&T::setup), t);
41 }
42 return make_ready_future<>();
43}
44
45template<typename T>
46future<> conditional_invoke_teardown(T& t) {
47 if constexpr (boost::unit_test::impl_fixture::has_teardown<T>::value) {
48 return futurize_invoke(std::mem_fn(&T::teardown), t);
49 }
50 return make_ready_future<>();
51}
52void warn_teardown_exception(const sstring&, std::exception_ptr);
53
54template<typename Func, typename T>
55requires std::is_invocable_v<Func, T>
56future<> do_fixture_test(Func&& f, T& t, const sstring& name) {
57 return conditional_invoke_setup(t).then([f = std::forward<Func>(f), &t]() mutable {
58 return futurize_invoke(std::forward<Func>(f), t);
59 }).finally([&]() {
60 return conditional_invoke_teardown(t).handle_exception([&](auto ep) {
61 warn_teardown_exception(name, ep);
62 });
63 });
64}

67// holder for decorator based fixture using class
68template<typename F>
69class async_class_based_fixture : public boost::unit_test::test_unit_fixture {
70public:
71 template<typename... Args>
72 requires std::is_constructible_v<F, const Args&...>
73 async_class_based_fixture(Args&& ...args)
74 : _create([args = std::make_tuple(std::forward<Args>(args)...)] {
75 return std::apply([](auto&& ...args) {
76 return std::make_unique<F>(args...);
77 }, args);
78 })
79 {}
80private:
81 // Fixture interface
82 void setup() override {
83 // create and possibly init the fixture object in reactor, on the
84 // test thread.
85 global_test_runner().run_sync([this] {
86 _inst = _create();
87 return detail::conditional_invoke_setup(*_inst);
88 });
89 }
90 void teardown() override {
91 // possibly de-init and destroy the fixture object in reactor, on the
92 // test thread.
93 global_test_runner().run_sync([this] {
94 auto inst = std::exchange(_inst, {});
95 return detail::conditional_invoke_teardown(*inst).finally([inst = std::move(inst)] {});
96 });
97 }
98 // Data members
99 std::unique_ptr<F> _inst;
100 std::function<std::unique_ptr<F>()> _create;
101};

105// holder for decorator based fixture using functions
106class async_function_based_fixture : public boost::unit_test::test_unit_fixture {
107public:
108 async_function_based_fixture(fixture_function setup, fixture_function teardown)
109 : _setup(std::move(setup))
110 , _teardown(std::move(teardown))

112private:
113 // Fixture interface

116 // run in reactor thread

123 if (_teardown) {
124 // run in reactor thread

126 return _teardown();
127 });
128 }
129 }
130 fixture_function _setup, _teardown;
131};
