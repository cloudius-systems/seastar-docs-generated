75// Execution wraps lreferences in reference_wrapper so that the caller is forced
76// to use seastar::ref(). Then when the function is actually called the
77// reference is unwrapped. However, we need to distinguish between functions
78// which argument is lvalue reference and functions that take
79// reference_wrapper<> as an argument and not unwrap the latter. To solve this
80// issue reference_wrapper_for_es type is used for wrappings done automatically
81// by execution stage.
82template<typename T>
83struct reference_wrapper_for_es : reference_wrapper<T> {
84 reference_wrapper_for_es(reference_wrapper <T> rw) noexcept
85 : reference_wrapper<T>(std::move(rw)) {}
86};
87
88template<typename T>
89struct wrap_for_es {
90 using type = T;
91};
92
93template<typename T>
94struct wrap_for_es<T&> {
95 using type = reference_wrapper_for_es<T>;
96};
97
98template<typename T>
99struct wrap_for_es<T&&> {
100 using type = T;
101};
102
103template<typename T>
104decltype(auto) unwrap_for_es(T&& object) {
105 return std::forward<T>(object);
106}
107
108template<typename T>
109std::reference_wrapper<T> unwrap_for_es(reference_wrapper_for_es<T> ref) {
110 return std::reference_wrapper<T>(ref.get());
111}

206template<typename ReturnType, typename... Args>
207requires std::is_nothrow_move_constructible_v<std::tuple<Args...>>
208class concrete_execution_stage_base : public execution_stage {
209 using args_tuple = std::tuple<Args...>;
210 static_assert(std::is_nothrow_move_constructible_v<args_tuple>,
211 "Function arguments need to be nothrow move constructible");
212
213 static constexpr size_t flush_threshold = 128;
214 static constexpr size_t max_queue_length = 1024;
215
216 using return_type = futurize_t<ReturnType>;
217 using promise_type = typename return_type::promise_type;
218 using input_type = typename tuple_map_types<internal::wrap_for_es, args_tuple>::type;
219
220 struct work_item {
221 input_type _in;
222 promise_type _ready;
223
224 work_item(typename internal::wrap_for_es<Args>::type... args) : _in(std::move(args)...) { }
225
226 work_item(work_item&& other) = delete;
227 work_item(const work_item&) = delete;
228 work_item(work_item&) = delete;
229 };
230 chunked_fifo<work_item, flush_threshold> _queue;
231
232 noncopyable_function<ReturnType (Args...)> _function;
233private:
234 auto unwrap(input_type&& in) {
235 return tuple_map(std::move(in), [] (auto&& obj) {
236 return internal::unwrap_for_es(std::forward<decltype(obj)>(obj));
237 });
238 }
239
240 virtual void do_flush() noexcept override {
241 while (!_queue.empty()) {
242 auto& wi = _queue.front();
243 auto wi_in = std::move(wi._in);
244 auto wi_ready = std::move(wi._ready);
245 _queue.pop_front();
246 futurize<ReturnType>::apply(_function, unwrap(std::move(wi_in))).forward_to(std::move(wi_ready));
247 _stats.function_calls_executed++;
248
249 if (internal::scheduler_need_preempt()) {
250 _stats.tasks_preempted++;
251 break;
252 }
253 }
254 _empty = _queue.empty();
255 }
256public:
257 explicit concrete_execution_stage_base(const sstring& name, scheduling_group sg, noncopyable_function<ReturnType (Args...)> f)
258 : execution_stage(name, sg)
259 , _function(std::move(f))
260 {
261 _queue.reserve(flush_threshold);
262 }
263 explicit concrete_execution_stage_base(const sstring& name, noncopyable_function<ReturnType (Args...)> f)
264 : concrete_execution_stage_base(name, scheduling_group(), std::move(f)) {
265 }
266
288 return_type operator()(typename internal::wrap_for_es<Args>::type... args) {
289 if (_queue.size() >= max_queue_length) {
290 do_flush();
291 }
292 _queue.emplace_back(std::move(args)...);
293 _empty = false;
294 _stats.function_calls_enqueued++;
295 auto f = _queue.back()._ready.get_future();
296 flush();
297 return f;
298 }
299};

343 static sstring format_name(const sstring& name, scheduling_group sg) {
344 return fmt::format("{}.{}", name, sg.name());
345 }
346 sstring _base_name;
347 public:
348 per_group_stage_type(const sstring& base_name, scheduling_group sg, noncopyable_function<ReturnType (Args...)> f)
349 : internal::concrete_execution_stage_base<ReturnType, Args...>(per_group_stage_type::format_name(base_name, sg), sg, std::move(f))
350 , _base_name(base_name)

353 per_group_stage_type(const sstring& name, noncopyable_function<ReturnType (Args...)> f)

358 this->_name = per_group_stage_type::format_name(_base_name, this->_sg);

362 static_assert(std::is_nothrow_move_constructible_v<args_tuple>,
363 "Function arguments need to be nothrow move constructible");
364
365 sstring _name;
366 noncopyable_function<ReturnType (Args...)> _function;
367 std::vector<std::optional<per_group_stage_type>> _stage_for_group{max_scheduling_groups()};
368private:
369 per_group_stage_type make_stage_for_group(scheduling_group sg) {
370 // We can't use std::ref(function), because reference_wrapper decays to noncopyable_function& and
371 // that selects the noncopyable_function copy constructor. Use a lambda instead.
372 auto wrapped_function = [&_function = _function] (Args... args) {
373 return _function(std::forward<Args>(args)...);
374 };
375 return per_group_stage_type(_name, sg, wrapped_function);
376 }
377public:

385 : _name(std::move(name)),_function(std::move(f)) {
386 }

408 return_type operator()(typename internal::wrap_for_es<Args>::type... args) {
409 auto sg = current_scheduling_group();
410 auto sg_id = internal::scheduling_group_index(sg);
411 auto& slot = _stage_for_group[sg_id];
412 if (!slot) {
413 slot.emplace(make_stage_for_group(sg));
414 }
415 return (*slot)(std::move(args)...);
416 }

425 inheriting_execution_stage::stats get_stats() const noexcept {
426 inheriting_execution_stage::stats summary;
427 for (unsigned sg_id = 0; sg_id != _stage_for_group.size(); ++sg_id) {
428 auto sg = internal::scheduling_group_from_index(sg_id);
429 if (_stage_for_group[sg_id]) {
430 summary.push_back({sg, _stage_for_group[sg_id]->get_stats()});
431 }
432 }
433 return summary;
434 }
