29template <typename T, typename U>
30void try_future_resume_or_destroy_coroutine(seastar::future<T>& fut, seastar::task& coroutine_task) {
31 auto promise_ptr = static_cast<U*>(&coroutine_task);
32 auto hndl = std::coroutine_handle<U>::from_promise(*promise_ptr);
33
34 if (fut.failed()) {
35 hndl.promise().set_exception(std::move(fut).get_exception());
36 hndl.destroy();
37 } else {
38 set_current_task(promise_ptr);
39 hndl.resume();
40 }
41}

43template <bool CheckPreempt, typename T>
44class [[nodiscard]] try_future_awaiter : public seastar::task {
45 seastar::future<T> _future;
46 void (*_resume_or_destroy)(seastar::future<T>&, seastar::task&){};
47 seastar::task* _coroutine_task{};
48 seastar::task* _waiting_task{};
49
50public:
51 explicit try_future_awaiter(seastar::future<T>&& f) noexcept : _future(std::move(f)) {}
52
53 try_future_awaiter(const try_future_awaiter&) = delete;
54 try_future_awaiter(try_future_awaiter&&) = delete;
55
56 bool await_ready() const noexcept {
57 // Will suspend+schedule for ready failed futures too.
58 return _future.available() && !_future.failed() && (!CheckPreempt || !need_preempt());
59 }

61 template<typename U>
62 void await_suspend(std::coroutine_handle<U> hndl) noexcept {
63 _resume_or_destroy = try_future_resume_or_destroy_coroutine<T, U>;
64 _coroutine_task = &hndl.promise();
65 _waiting_task = hndl.promise().waiting_task();
66
67 if (_future.available()) {
68 execute_involving_handle_destruction_in_await_suspend(this);
69 } else {
70 _future.set_coroutine(*this);
71 }
72 }
73
74 T await_resume() {
75 if constexpr (std::is_void_v<T>) {
76 _future.get();
77 } else {
78 return std::move(_future).get();
79 }
80 }
81
82 virtual void run_and_dispose() noexcept final override {
83 _resume_or_destroy(_future, *_coroutine_task);
84 }
85
86 virtual task* waiting_task() noexcept override {
87 return _waiting_task;
88 }
89};
