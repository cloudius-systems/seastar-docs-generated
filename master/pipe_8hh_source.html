83template <typename T>
84class pipe_buffer {
85private:
86 queue<std::optional<T>> _buf;
87 bool _read_open = true;
88 bool _write_open = true;
89public:
90 pipe_buffer(size_t size) : _buf(size) {}
91 future<std::optional<T>> read() {
92 return _buf.pop_eventually();
93 }
94 future<> write(T&& data) {
95 return _buf.push_eventually(std::move(data));
96 }
97 bool readable() const {
98 return _write_open || !_buf.empty();
99 }
100 bool writeable() const {
101 return _read_open;
102 }
103 bool close_read() {
104 // If a writer blocking (on a full queue), need to stop it.
105 if (_buf.full()) {
106 _buf.abort(std::make_exception_ptr(broken_pipe_exception()));
107 }
108 _read_open = false;
109 return !_write_open;
110 }
111 bool close_write() {
112 // If the queue is empty, write the EOF (disengaged optional) to the
113 // queue to wake a blocked reader. If the queue is not empty, there is
114 // no need to write the EOF to the queue - the reader will return an
115 // EOF when it sees that _write_open == false.
116 if (_buf.empty()) {
117 _buf.push({});
118 }
119 _write_open = false;
120 return !_read_open;
121 }
122};

134template <typename T>
135class pipe_reader {
136private:
137 internal::pipe_buffer<T> *_bufp;
138 std::optional<T> _unread;
139 pipe_reader(internal::pipe_buffer<T> *bufp) noexcept : _bufp(bufp) { }
140 friend class pipe<T>;
141public:
148 future<std::optional<T>> read() {
149 if (_unread) {
150 auto ret = std::move(*_unread);
151 _unread = {};
152 return make_ready_future<std::optional<T>>(std::move(ret));
153 }
154 if (_bufp->readable()) {
155 return _bufp->read();
156 } else {
157 return make_ready_future<std::optional<T>>();
158 }
159 }
169 void unread(T&& item) {
170 if (_unread) {
171 throw unread_overflow_exception();
172 }
173 _unread = std::move(item);
174 }
175 ~pipe_reader() {
176 if (_bufp && _bufp->close_read()) {
177 delete _bufp;
178 }
179 }
180 // Allow move, but not copy, of pipe_reader
181 pipe_reader(pipe_reader&& other) noexcept : _bufp(other._bufp) {
182 other._bufp = nullptr;
183 }
184 pipe_reader& operator=(pipe_reader&& other) noexcept {
185 std::swap(_bufp, other._bufp);
186 return *this;
187 }
188};
