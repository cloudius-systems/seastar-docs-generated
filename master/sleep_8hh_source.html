43template <typename Clock = steady_clock_type, typename Rep, typename Period>
44future<> sleep(std::chrono::duration<Rep, Period> dur) {
45 struct sleeper {
46 promise<> done;
47 timer<Clock> tmr;
48 sleeper(std::chrono::duration<Rep, Period> dur)
49 : tmr([this] { done.set_value(); })
50 {
51 tmr.arm(dur);
52 }
53 };
54 auto s = std::make_unique<sleeper>(dur);
55 future<> fut = s->done.get_future();
56 return fut.finally([s = std::move(s)] {});
