157template <class T, class Container>
158class json_list_template : public json_base_element {
159public:
160
165 void push(const T& element) {
166 _set = true;
167 _elements.push_back(element);
168 }
169
174 void push(T&& element) {
175 _set = true;
176 _elements.push_back(std::move(element));
177 }
178
179 virtual std::string to_string() override
180 {
181 return formatter::to_json(_elements);
182 }
183
188 template<class C>
189 json_list_template& operator=(const C& list) {
190 _elements.clear();
191 _elements.reserve(std::size(list));
192 for (auto i : list) {
193 push(i);
194 }
195 return *this;
196 }
197
198 template<class C>
199 json_list_template& operator=(json_list_template<T, Container>&& list) noexcept {
200 if (this != &list) {
201 _elements = std::move(list._elements);
202 }
203 return *this;
204 }
205
206 virtual future<> write(output_stream<char>& s) const override {
207 return formatter::write(s, _elements);
208 }

210 Container _elements;

364json_return_type::body_writer_type stream_range_as_array(Container val, Func fun) {
365 return [val = std::move(val), fun = std::move(fun)](output_stream<char>&& s) mutable {
366 return do_with(output_stream<char>(std::move(s)), Container(std::move(val)), Func(std::move(fun)), true, [](output_stream<char>& s, const Container& val, const Func& f, bool& first){
367 return s.write("[").then([&val, &s, &first, &f] () {
368 return do_for_each(val, [&s, &first, &f](const typename Container::value_type& v){
369 auto fut = first ? make_ready_future<>() : s.write(", ");
370 first = false;
371 return fut.then([&s, &f, &v]() {
372 return formatter::write(s, f(v));
373 });
374 });
375 }).then([&s](){
376 return s.write("]");
377 }).finally([&s] {
378 return s.close();
379 });
380 });
381 };
