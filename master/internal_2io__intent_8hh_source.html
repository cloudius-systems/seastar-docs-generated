37/*
38 * The tracker of cancellable sub-queue of requests.
39 *
40 * This queue is stuffed with requests that sit in the
41 * same IO queue for dispatching (there can be other requests
42 * as well) and ties them together for cancellation.
43 * This IO queue is the fair_queue's priority_class's one.
44 * Beware, if requests from different IO queues happen
45 * in the same cancellable queue the whole thing blows up.
46 */
47class cancellable_queue {
48public:
49 class link {
50 friend class cancellable_queue;
51
52 union {
53 cancellable_queue* _ref;
54 bi::slist_member_hook<> _hook;
55 };
56
57 public:
58 link() noexcept : _ref(nullptr) {}
59 ~link() { SEASTAR_ASSERT(_ref == nullptr); }
60
61 void enqueue(cancellable_queue& cq) noexcept {
62 cq.push_back(*this);
63 }
64
65 void maybe_dequeue() noexcept {
66 if (_ref != nullptr) {
67 _ref->pop_front();
68 }
69 }
70 };
71
72private:
73 static_assert(sizeof(link) == sizeof(void*), "cancellable_queue::link size is too big");
74 using list_of_links_t = bi::slist<link,
75 bi::constant_time_size<false>,
76 bi::cache_last<true>,
77 bi::member_hook<link, bi::slist_member_hook<>, &link::_hook>>;
78
79 link* _first;
80 list_of_links_t _rest;
81
82 void push_back(link& il) noexcept;
83 void pop_front() noexcept;
84
85public:
86 cancellable_queue() noexcept : _first(nullptr) {}
87 cancellable_queue(const cancellable_queue&) = delete;
88 cancellable_queue(cancellable_queue&& o) noexcept;
89 cancellable_queue& operator=(cancellable_queue&& o) noexcept;
90 ~cancellable_queue();
91};

93/*
94 * A "safe" reference on a intent. Safe here means that the original
95 * intent can be destroyed at any time and this reference will be
96 * updated not to point at it any longer.
97 * The retrieve() method brings the original intent back or throws
98 * and exception if the intent was cancelled.
99 */
100class intent_reference : public bi::list_base_hook<bi::link_mode<bi::auto_unlink>> {
101 friend class seastar::io_intent;
102 using container_type = bi::list<intent_reference, bi::constant_time_size<false>>;
103 static constexpr uintptr_t _cancelled_intent = 1;
104 io_intent* _intent;
105
106 void on_cancel() noexcept { _intent = reinterpret_cast<io_intent*>(_cancelled_intent); }
107 bool is_cancelled() const noexcept { return _intent == reinterpret_cast<io_intent*>(_cancelled_intent); }
108
109public:
110 intent_reference(io_intent* intent) noexcept;
111
112 intent_reference(intent_reference&& o) noexcept : _intent(std::exchange(o._intent, nullptr)) {
113 container_type::node_algorithms::swap_nodes(o.this_ptr(), this_ptr());
114 }
115
116 intent_reference& operator=(intent_reference&& o) noexcept {
117 if (this != &o) {
118 _intent = std::exchange(o._intent, nullptr);
119 unlink();
120 container_type::node_algorithms::swap_nodes(o.this_ptr(), this_ptr());
121 }
122 return *this;
123 }
124
125 io_intent* retrieve() const;
126};
