67class connection_id {
68 uint64_t _id;
69
70public:
71 uint64_t id() const {
72 return _id;
73 }
74 bool operator==(const connection_id& o) const {
75 return _id == o._id;
76 }
77 explicit operator bool() const {
78 return shard() != 0xffff;
79 }
80 size_t shard() const {
81 return size_t(_id & 0xffff);
82 }
83 constexpr static connection_id make_invalid_id(uint64_t _id = 0) {
84 return make_id(_id, 0xffff);
85 }
86 constexpr static connection_id make_id(uint64_t _id, uint16_t shard) {
87 return {_id << 16 | shard};
88 }
89 constexpr connection_id(uint64_t id) : _id(id) {}
90};

98struct client_info {
99 socket_address addr;
100 rpc::server& server;
101 connection_id conn_id;
102 std::unordered_map<sstring, std::any> user_data;
103 template <typename T>
104 void attach_auxiliary(const sstring& key, T&& object) {
105 user_data.emplace(key, std::any(std::forward<T>(object)));
106 }
107 template <typename T>
108 T& retrieve_auxiliary(const sstring& key) {
109 auto it = user_data.find(key);
110 SEASTAR_ASSERT(it != user_data.end());
111 return std::any_cast<T&>(it->second);
112 }
113 template <typename T>
114 std::add_const_t<T>& retrieve_auxiliary(const sstring& key) const {
115 return const_cast<client_info*>(this)->retrieve_auxiliary<std::add_const_t<T>>(key);
116 }
117 template <typename T>
118 T* retrieve_auxiliary_opt(const sstring& key) noexcept {
119 auto it = user_data.find(key);
120 if (it == user_data.end()) {
121 return nullptr;
122 }
123 return &std::any_cast<T&>(it->second);
124 }
125 template <typename T>
126 const T* retrieve_auxiliary_opt(const sstring& key) const noexcept {
127 auto it = user_data.find(key);
128 if (it == user_data.end()) {
129 return nullptr;
130 }
131 return &std::any_cast<const T&>(it->second);
132 }

205struct cancellable {
206 std::function<void()> cancel_send;
207 std::function<void()> cancel_wait;
208 cancellable** send_back_pointer = nullptr;
209 cancellable** wait_back_pointer = nullptr;
210 cancellable() = default;
211 cancellable(cancellable&& x) : cancel_send(std::move(x.cancel_send)), cancel_wait(std::move(x.cancel_wait)), send_back_pointer(x.send_back_pointer), wait_back_pointer(x.wait_back_pointer) {
212 if (send_back_pointer) {
213 *send_back_pointer = this;
214 x.send_back_pointer = nullptr;
215 }
216 if (wait_back_pointer) {
217 *wait_back_pointer = this;
218 x.wait_back_pointer = nullptr;
219 }
220 }
221 cancellable& operator=(cancellable&& x) {
222 if (&x != this) {
223 this->~cancellable();
224 new (this) cancellable(std::move(x));
225 }
226 return *this;
227 }
228 void cancel() {
229 if (cancel_send) {
230 cancel_send();
231 }
232 if (cancel_wait) {
233 cancel_wait();
234 }
235 }
236 ~cancellable() {
237 cancel();
238 }

273static inline memory_input_stream<rcv_buf::iterator> make_deserializer_stream(rcv_buf& input) {
274 auto* b = std::get_if<temporary_buffer<char>>(&input.bufs);
275 if (b) {
276 return memory_input_stream<rcv_buf::iterator>(memory_input_stream<rcv_buf::iterator>::simple(b->begin(), b->size()));

319// send data Out...
320template<typename... Out>
321class sink {
322public:
323 class impl {
324 protected:
325 xshard_connection_ptr _con;
326 semaphore _sem;
327 std::exception_ptr _ex;
328 impl(xshard_connection_ptr con) : _con(std::move(con)), _sem(max_stream_buffers_memory) {}
329 public:
330 virtual ~impl() {};
331 virtual future<> operator()(const Out&... args) = 0;
332 // Failures may be returned as an exceptional future
333 virtual future<> close() noexcept = 0;
334 virtual future<> flush() noexcept = 0;
335 friend sink;
336 };
337
338private:
339 shared_ptr<impl> _impl;
340
341public:
342 sink(shared_ptr<impl> impl) : _impl(std::move(impl)) {}
343 future<> operator()(const Out&... args) {
344 return _impl->operator()(args...);
345 }
346 // Failures may be returned as an exceptional future
347 future<> close() noexcept {
348 return _impl->close();
349 }
350 // Calling this function makes sure that any data buffered
351 // by the stream sink will be flushed to the network.
352 // It does not mean the data was received by the corresponding
353 // source.
354 future<> flush() noexcept {
355 return _impl->flush();
356 }
357 connection_id get_id() const;
358};
