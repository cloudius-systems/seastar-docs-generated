63class connection_id {
64 uint64_t _id;
65
66public:
67 uint64_t id() const {
68 return _id;
69 }
70 bool operator==(const connection_id& o) const {
71 return _id == o._id;
72 }
73 explicit operator bool() const {
74 return shard() != 0xffff;
75 }
76 size_t shard() const {
77 return size_t(_id & 0xffff);
78 }
79 constexpr static connection_id make_invalid_id(uint64_t _id = 0) {
80 return make_id(_id, 0xffff);
81 }
82 constexpr static connection_id make_id(uint64_t _id, uint16_t shard) {
83 return {_id << 16 | shard};
84 }
85 constexpr connection_id(uint64_t id) : _id(id) {}
86};

220struct cancellable {
221 std::function<void()> cancel_send;
222 std::function<void()> cancel_wait;
223 cancellable** send_back_pointer = nullptr;
224 cancellable** wait_back_pointer = nullptr;
225 cancellable() = default;
226 cancellable(cancellable&& x) : cancel_send(std::move(x.cancel_send)), cancel_wait(std::move(x.cancel_wait)), send_back_pointer(x.send_back_pointer), wait_back_pointer(x.wait_back_pointer) {
227 if (send_back_pointer) {
228 *send_back_pointer = this;
229 x.send_back_pointer = nullptr;
230 }
231 if (wait_back_pointer) {
232 *wait_back_pointer = this;
233 x.wait_back_pointer = nullptr;
234 }
235 }
236 cancellable& operator=(cancellable&& x) {
237 if (&x != this) {
238 this->~cancellable();
239 new (this) cancellable(std::move(x));
240 }
241 return *this;
242 }
243 void cancel() {
244 if (cancel_send) {
245 cancel_send();
246 }
247 if (cancel_wait) {
248 cancel_wait();
249 }
250 }
251 ~cancellable() {
252 cancel();
253 }

340// send data Out...
341template<typename... Out>
342class sink {
343public:
344 class impl {
345 protected:
346 xshard_connection_ptr _con;
347 semaphore _sem;
348 std::exception_ptr _ex;
349 impl(xshard_connection_ptr con) : _con(std::move(con)), _sem(max_stream_buffers_memory) {}
350 public:
351 virtual ~impl() {};
352 virtual future<> operator()(const Out&... args) = 0;
353 // Failures may be returned as an exceptional future
354 virtual future<> close() noexcept = 0;
355 virtual future<> flush() noexcept = 0;
356 friend sink;
357 };
358
359private:
360 shared_ptr<impl> _impl;
361
362public:
363 sink(shared_ptr<impl> impl) : _impl(std::move(impl)) {}
364 future<> operator()(const Out&... args) {
365 return _impl->operator()(args...);
366 }
367 // Failures may be returned as an exceptional future
368 future<> close() noexcept {
369 return _impl->close();
370 }
371 // Calling this function makes sure that any data buffered
372 // by the stream sink will be flushed to the network.
373 // It does not mean the data was received by the corresponding
374 // source.
375 future<> flush() noexcept {
376 return _impl->flush();
377 }
378 connection_id get_id() const;
379};
