53class forward_hash {
54 uint8_t data[64];
55 size_t end_idx = 0;
56public:
57 size_t size() const {
58 return end_idx;
59 }
60 void push_back(uint8_t b) {
61 SEASTAR_ASSERT(end_idx < sizeof(data));
62 data[end_idx++] = b;
63 }
64 void push_back(uint16_t b) {
65 push_back(uint8_t(b));
66 push_back(uint8_t(b >> 8));
67 }
68 void push_back(uint32_t b) {
69 push_back(uint16_t(b));
70 push_back(uint16_t(b >> 16));
71 }
72 const uint8_t& operator[](size_t idx) const {
73 return data[idx];
74 }
75};
