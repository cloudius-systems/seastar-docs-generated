77class gate {
78 size_t _count = 0;
79 std::optional<promise<>> _stopped;
80
81#ifdef SEASTAR_GATE_HOLDER_DEBUG
82 void assert_not_held_when_moved() const noexcept;
83 void assert_not_held_when_destroyed() const noexcept;
84#else // SEASTAR_GATE_HOLDER_DEBUG
85 void assert_not_held_when_moved() const noexcept {}
86 void assert_not_held_when_destroyed() const noexcept {}
87#endif // SEASTAR_GATE_HOLDER_DEBUG
88
89public:
90 // Implemented to force noexcept due to boost:intrusive::list
91 gate() noexcept {};
92 gate(const gate&) = delete;
93 gate(gate&& x) noexcept
94 : _count(std::exchange(x._count, 0)), _stopped(std::exchange(x._stopped, std::nullopt)) {
95 x.assert_not_held_when_moved();
96 }
97 gate& operator=(gate&& x) noexcept {
98 if (this != &x) {
99 SEASTAR_ASSERT(!_count && "gate reassigned with outstanding requests");
100 x.assert_not_held_when_moved();
101 _count = std::exchange(x._count, 0);
102 _stopped = std::exchange(x._stopped, std::nullopt);
103 }
104 return *this;
105 }
106 virtual ~gate() {
107 SEASTAR_ASSERT(!_count && "gate destroyed with outstanding requests");
108 assert_not_held_when_destroyed();
109 }
114 bool try_enter() noexcept {
115 bool opened = !_stopped;
116 if (opened) {
117 ++_count;
118 }
119 return opened;
120 }
125 void enter() {
126 if (!try_enter()) {
127 throw_closed_exception();
128 }
129 }
134 void leave() noexcept {
135 --_count;
136 if (!_count && _stopped) {
137 _stopped->set_value();
138 }
139 }
149 void check() const {
150 if (_stopped) {
151 throw_closed_exception();
152 }
153 }
159 future<> close() noexcept {
160 SEASTAR_ASSERT(!_stopped && "seastar::gate::close() cannot be called more than once");
161 _stopped = std::make_optional(promise<>());
162 if (!_count) {
163 _stopped->set_value();
164 }
165 return _stopped->get_future();
166 }
167
169 size_t get_count() const noexcept {
170 return _count;
171 }
172
174 bool is_closed() const noexcept {
175 return bool(_stopped);
176 }

187 class holder {
188 gate* _g;
189#ifdef SEASTAR_GATE_HOLDER_DEBUG
190 using member_hook_t = boost::intrusive::list_member_hook<boost::intrusive::link_mode<boost::intrusive::auto_unlink>>;
191 member_hook_t _link;
192
193 void debug_hold_gate() noexcept {
194 if (_g) {
195 _g->_holders.push_back(*this);
196 }
197 }
198
199 void debug_release_gate() noexcept {
200 _link.unlink();
201 }
202#else // SEASTAR_GATE_HOLDER_DEBUG
203 void debug_hold_gate() noexcept {}
204 void debug_release_gate() noexcept {}
205#endif // SEASTAR_GATE_HOLDER_DEBUG
206
207 // release the holder from the gate without leaving it.
208 // used for release and move.
209 gate* release_gate() noexcept {
210 auto* g = std::exchange(_g, nullptr);
211 if (g) {
212 debug_release_gate();
213 }
214 return g;
215 }
216
217 friend class gate;
218 public:
221 holder() noexcept : _g(nullptr) { }
222
225 explicit holder(gate& g) : _g(&g) {
226 _g->enter();
227 debug_hold_gate();
228 }
229
235 holder(const holder& x) noexcept : _g(x._g) {
236 if (_g) {
237 _g->_count++;
238 debug_hold_gate();
239 }
240 }
241
245 holder(holder&& x) noexcept : _g(std::move(x).release_gate()) {
246 debug_hold_gate();
247 }
248
250 ~holder() {
251 release();
252 }
253
260 holder& operator=(const holder& x) noexcept {
261 if (x._g != _g) {
262 release();
263 _g = x._g;
264 if (_g) {
265 _g->_count++;
266 debug_hold_gate();
267 }
268 }
269 return *this;
270 }
271
276 holder& operator=(holder&& x) noexcept {
277 if (&x != this) {
278 release();
279 _g = std::move(x).release_gate();
280 debug_hold_gate();
281 }
282 return *this;
283 }
284
286 void release() noexcept {
287 if (auto g = release_gate()) {
288 g->leave();
289 }
290 }
291 };
