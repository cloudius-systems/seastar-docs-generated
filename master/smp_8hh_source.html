219 template <typename Func>
220 struct async_work_item : work_item {
221 smp_message_queue& _queue;
222 Func _func;
223 using futurator = futurize<std::invoke_result_t<Func>>;
224 using future_type = typename futurator::type;
225 using value_type = typename future_type::value_type;
226 std::optional<value_type> _result;
227 std::exception_ptr _ex; // if !_result
228 typename futurator::promise_type _promise; // used on local side
229 async_work_item(smp_message_queue& queue, smp_service_group ssg, Func&& func) : work_item(ssg), _queue(queue), _func(std::move(func)) {}
230 virtual void fail_with(std::exception_ptr ex) override {
231 _promise.set_exception(std::move(ex));
232 }
233 virtual task* waiting_task() noexcept override {
234 // FIXME: waiting_tasking across shards is not implemented. Unsynchronized task access is unsafe.
235 return nullptr;
236 }
237 virtual void run_and_dispose() noexcept override {
238 // _queue.respond() below forwards the continuation chain back to the
239 // calling shard.
240 (void)futurator::invoke(this->_func).then_wrapped([this] (auto f) {
241 if (f.failed()) {
242 _ex = f.get_exception();
243 } else {
244 _result = f.get();
245 }
246 _queue.respond(this);
247 });
248 // We don't delete the task here as the creator of the work item will
249 // delete it on the origin shard.
250 }
251 virtual void complete() override {
252 if (_result) {
253 _promise.set_value(std::move(*_result));
254 } else {
255 // FIXME: _ex was allocated on another cpu
256 _promise.set_exception(std::move(_ex));
257 }
258 }
259 future_type get_future() { return _promise.get_future(); }
260 };

348 template <typename Func>
349 static futurize_t<std::invoke_result_t<Func>> submit_to(unsigned t, smp_submit_to_options options, Func&& func) noexcept {
350 using ret_type = std::invoke_result_t<Func>;
351 if (t == this_shard_id()) {
352 try {
353 if (!is_future<ret_type>::value) {
354 // Non-deferring function, so don't worry about func lifetime
355 return futurize<ret_type>::invoke(std::forward<Func>(func));
356 } else if (std::is_lvalue_reference_v<Func>) {
357 // func is an lvalue, so caller worries about its lifetime
358 return futurize<ret_type>::invoke(func);
359 } else {
360 // Deferring call on rvalue function, make sure to preserve it across call
361 auto w = std::make_unique<std::decay_t<Func>>(std::move(func));
362 auto ret = futurize<ret_type>::invoke(*w);
363 return ret.finally([w = std::move(w)] {});
364 }
365 } catch (...) {
366 // Consistently return a failed future rather than throwing, to simplify callers
367 return futurize<std::invoke_result_t<Func>>::make_exception_future(std::current_exception());
368 }
369 } else {
370 return _qs[t][this_shard_id()].submit(t, options, std::forward<Func>(func));
371 }
372 }
