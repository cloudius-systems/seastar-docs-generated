37template <typename Func>
38requires std::is_nothrow_move_constructible_v<Func>
39auto
40schedule_in_group(scheduling_group sg, Func func) noexcept {
41 static_assert(std::is_nothrow_move_constructible_v<Func>);
42 auto tsk = make_task(sg, std::move(func));
43 schedule_checked(tsk);
44 return tsk->get_future();
45}

60template <typename Func, typename... Args>
61requires std::is_nothrow_move_constructible_v<Func>
62inline
63auto
64with_scheduling_group(scheduling_group sg, Func func, Args&&... args) noexcept {
65 static_assert(std::is_nothrow_move_constructible_v<Func>);
66 using return_type = decltype(func(std::forward<Args>(args)...));
67 using futurator = futurize<return_type>;
68 if (sg.active()) {
69 return futurator::invoke(func, std::forward<Args>(args)...);
70 } else {
71 return internal::schedule_in_group(sg, [func = std::move(func), args = std::make_tuple(std::forward<Args>(args)...)] () mutable {
72 return futurator::apply(func, std::move(args));
73 });
74 }
75}
