176template<typename Test>
177class concrete_performance_test final : public performance_test {
178 std::optional<Test> _test;
179
180 using test_ret_type = decltype(_test->run());
181 // true iff the test method returns future<...>
182 static constexpr bool is_async_test = is_future<test_ret_type>::value;
183 // true iff the test returns the number of iterations run, otherwise it returns
184 // void and we consider each invocation to be 1 iteration
185 static constexpr bool is_iteration_returning = !(std::is_same_v<test_ret_type, future<>> || std::is_void_v<test_ret_type>);
186private:
187
188protected:
189 virtual void set_up() override {
190 _test.emplace();
191 }
192
193 virtual void tear_down() noexcept override {
194 _test = std::nullopt;
195 }
196
197 [[gnu::hot]]
198 virtual future<run_result> do_single_run() override {
199 run_hooks();
200 start_run();
201 while (!stop_iteration()) {
202 if constexpr (is_async_test) {
203 if constexpr (is_iteration_returning) {
204 auto f = _test->run();
205 next_iteration(f.available() ? std::move(f).get() : co_await std::move(f));
206 } else {
207 auto f = _test->run();
208 // The available() check is functionally redundant, but is significantly faster
209 // than invoking the co_await machinery on a future-returning function.
210 if (!f.available()) {
211 co_await std::move(f);
212 }
213 next_iteration(1);
214 }
215 } else {
216 if constexpr (is_iteration_returning) {
217 next_iteration(_test->run());
218 } else {
219 _test->run();
220 next_iteration(1);
221 }
222 }
223 }
224 co_return stop_run();
225 }
226public:
227 using performance_test::performance_test;
228};

265// PERF_TEST and PERF_TEST_F support both synchronous and asynchronous functions.
266// The former should return `void`, the latter `future<>`.
267// PERF_TEST_C executes a coroutine function, if enabled.
268// PERF_TEST_CN executes a coroutine function, if enabled, returning the number of inner-loops.
269//
270// Test cases may perform multiple operations in a single run, this may be desirable
271// if the cost of an individual operation is very small. This allows measuring either
272// the latency of throughput depending on how the test in written. In such cases,
273// the test function shall return either size_t or future<size_t> for synchronous and
274// asynchronous cases respectively. The returned value shall be the number of iterations
275// done in a single test run.
276
277#define PERF_TEST_F(test_group, test_case) \
278 struct test_##test_group##_##test_case : test_group { \
279 [[gnu::always_inline]] inline auto run(); \
280 }; \
281 static ::perf_tests::internal::test_registrar<test_##test_group##_##test_case> \
282 test_##test_group##_##test_case##_registrar(#test_group, #test_case); \
283 [[gnu::always_inline]] auto test_##test_group##_##test_case::run()
284
285#define PERF_TEST(test_group, test_case) \
286 struct test_##test_group##_##test_case { \
287 [[gnu::always_inline]] inline auto run(); \
288 }; \
289 static ::perf_tests::internal::test_registrar<test_##test_group##_##test_case> \
290 test_##test_group##_##test_case##_registrar(#test_group, #test_case); \
291 [[gnu::always_inline]] auto test_##test_group##_##test_case::run()
292
293
294#define PERF_TEST_C(test_group, test_case) \
295 struct test_##test_group##_##test_case : test_group { \
296 inline future<> run(); \
297 }; \
298 static ::perf_tests::internal::test_registrar<test_##test_group##_##test_case> \
299 test_##test_group##_##test_case##_registrar(#test_group, #test_case); \
300 future<> test_##test_group##_##test_case::run()
301
302#define PERF_TEST_CN(test_group, test_case) \
303 struct test_##test_group##_##test_case : test_group { \
304 inline future<size_t> run(); \
305 }; \
306 static ::perf_tests::internal::test_registrar<test_##test_group##_##test_case> \
307 test_##test_group##_##test_case##_registrar(#test_group, #test_case); \
308 future<size_t> test_##test_group##_##test_case::run()
