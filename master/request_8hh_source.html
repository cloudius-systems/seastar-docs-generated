247 bool should_keep_alive() const {
248 if (_version == "0.9") {
249 return false;
250 }
251
252 // TODO: handle HTTP/2.0 when it releases
253
254 auto it = _headers.find("Connection");
255 if (_version == "1.0") {
256 return it != _headers.end()
257 && seastar::internal::case_insensitive_cmp()(it->second, "keep-alive");
258 } else { // HTTP/1.1
259 return it == _headers.end() || !seastar::internal::case_insensitive_cmp()(it->second, "close");
260 }
