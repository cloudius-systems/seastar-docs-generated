447 constexpr auto parse(ParseContext& ctx) { return ctx.begin(); }
448 template <typename FormatContext>
449 auto format(const seastar::http::request& rq, FormatContext& ctx) const {
450 auto out = fmt::format_to(ctx.out(), "{} {}", rq._method, rq._url);
451 for (const auto& h : rq._headers) {
452 out = fmt::format_to(out, " {}:{}", h.first, h.second);
453 }
454 if (!rq.body_writer) {
455 const auto& body = seastar::http::internal::deprecated_content(rq);
456 if (!body.empty()) {
457 out = fmt::format_to(out, " {}", body);
458 }
459 }
460 return out;
