41template<typename T>
42class weak_ptr {
43 template<typename U>
44 friend class weakly_referencable;
45 template <typename U>
46 friend class weak_ptr;
47private:
48 using hook_type = boost::intrusive::list_member_hook<boost::intrusive::link_mode<boost::intrusive::auto_unlink>>;
49 hook_type _hook;
50 T* _ptr = nullptr;
51 weak_ptr(T* p) noexcept : _ptr(p) {}
52 void clear() noexcept {
53 _hook = {};
54 _ptr = nullptr;
55 }
56 void swap(weak_ptr&& o) noexcept {
57 _hook.swap_nodes(o._hook);
58 std::swap(_ptr, o._ptr);
59 }
60
61public:
62 template <typename U>
63 requires std::convertible_to<U*, T*>
64 weak_ptr(weak_ptr<U>&& o)
65 {
66 if (o._ptr) {
67 _ptr = std::exchange(o._ptr, nullptr);
68 _hook.swap_nodes(o._hook);
69 }
70 }
71
72 // Note: The default constructor's body is implemented as no-op
73 // rather than `noexcept = default` due to a bug with gcc 9.3.1
74 // that deletes the constructor since boost::intrusive::list_member_hook
75 // is not default_nothrow_constructible.
76 weak_ptr() noexcept {}
77 weak_ptr(std::nullptr_t) noexcept : weak_ptr() {}
78 weak_ptr(weak_ptr&& o) noexcept
79 {
80 swap(std::move(o));
81 }
82 weak_ptr(const weak_ptr& o) noexcept {
83 if (o._ptr) {
84 swap(o._ptr->weak_from_this());
85 }
86 }
87 weak_ptr& operator=(weak_ptr&& o) noexcept {
88 if (this != &o) {
89 clear();
90 swap(std::move(o));
91 }
92 return *this;
93 }
94 weak_ptr& operator=(const weak_ptr& o) noexcept {
95 if (this != &o) {
96 clear();
97 if (o._ptr) {
98 swap(o._ptr->weak_from_this());
99 }
100 }
101 return *this;
102 }
103 explicit operator bool() const noexcept { return _ptr != nullptr; }
104 T* operator->() const noexcept { return _ptr; }
105 T& operator*() const noexcept { return *_ptr; }
106 T* get() const noexcept { return _ptr; }
107 bool operator==(const weak_ptr& o) const noexcept { return _ptr == o._ptr; }
108 bool operator!=(const weak_ptr& o) const noexcept { return _ptr != o._ptr; }
109};

128template<typename T>
129class weakly_referencable {
130 boost::intrusive::list<weak_ptr<T>,
131 boost::intrusive::member_hook<weak_ptr<T>, typename weak_ptr<T>::hook_type, &weak_ptr<T>::_hook>,
132 boost::intrusive::constant_time_size<false>> _ptr_list;
133public:
134 // Note: The default constructor's body is implemented as no-op
135 // rather than `noexcept = default` due to a bug with gcc 9.3.1
136 // that deletes the constructor since boost::intrusive::member_hook
137 // is not default_nothrow_constructible.
138 weakly_referencable() noexcept {}
139 weakly_referencable(weakly_referencable&&) = delete; // pointer to this is captured and passed to weak_ptr
140 weakly_referencable(const weakly_referencable&) = delete;
141 ~weakly_referencable() noexcept {
142 _ptr_list.clear_and_dispose([] (weak_ptr<T>* wp) noexcept {
143 wp->_ptr = nullptr;
144 });
145 }
146 weak_ptr<T> weak_from_this() noexcept {
147 weak_ptr<T> ptr(static_cast<T*>(this));
148 _ptr_list.push_back(ptr);
149 return ptr;
150 }
151
152 weak_ptr<const T> weak_from_this() const noexcept {
153 return const_cast<weakly_referencable*>(this)->weak_from_this();
154 }
155};
