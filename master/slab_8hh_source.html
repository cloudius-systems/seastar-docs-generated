48/*
49 * slab_page_desc is 1:1 mapped to slab page.
50 * footprint: 80b for each slab page.
51 */
52struct slab_page_desc {
53private:
54 boost::intrusive::list_member_hook<> _lru_link;
55 boost::intrusive::list_member_hook<> _free_pages_link;
56 void *_slab_page;
57 std::vector<uintptr_t> _free_objects;
58 uint32_t _refcnt;
59 uint32_t _index; // index into slab page vector
60 uint16_t _magic;
61 uint8_t _slab_class_id;
62public:
63 slab_page_desc(void *slab_page, size_t objects, size_t object_size, uint8_t slab_class_id, uint32_t index)
64 : _slab_page(slab_page)
65 , _refcnt(0U)
66 , _index(index)
67 , _magic(SLAB_MAGIC_NUMBER)
68 , _slab_class_id(slab_class_id)
69 {
70 auto object = reinterpret_cast<uintptr_t>(slab_page);
71 _free_objects.reserve(objects - 1);
72 for (auto i = 1u; i < objects; i++) {
73 object += object_size;
74 _free_objects.push_back(object);
75 }
76 }
77
78 bool empty() const {
79 return _free_objects.empty();
80 }
81
82 size_t size() const {
83 return _free_objects.size();
84 }
85
86 uint32_t& refcnt() {
87 return _refcnt;
88 }
89
90 uint32_t index() const {
91 return _index;
92 }
93
94 uint16_t magic() const {
95 return _magic;
96 }
97
98 uint8_t slab_class_id() const {
99 return _slab_class_id;
100 }
101
102 void* slab_page() const {
103 return _slab_page;
104 }
105
106 std::vector<uintptr_t>& free_objects() {
107 return _free_objects;
108 }
109
110 void* allocate_object() {
111 SEASTAR_ASSERT(!_free_objects.empty());
112 auto object = reinterpret_cast<void*>(_free_objects.back());
113 _free_objects.pop_back();
114 return object;
115 }

117 void free_object(void *object) {
118 _free_objects.push_back(reinterpret_cast<uintptr_t>(object));

134template<typename Item>
135class slab_class {
136private:
137 boost::intrusive::list<slab_page_desc,
138 boost::intrusive::member_hook<slab_page_desc, boost::intrusive::list_member_hook<>,
139 &slab_page_desc::_free_pages_link>> _free_slab_pages;
140 boost::intrusive::list<slab_item_base,
141 boost::intrusive::member_hook<slab_item_base, boost::intrusive::list_member_hook<>,
142 &slab_item_base::_lru_link>> _lru;
143 size_t _size; // size of objects
144 uint8_t _slab_class_id;
145private:
146 template<typename... Args>
147 inline
148 Item* create_item(void *object, uint32_t slab_page_index, Args&&... args) {
149 Item *new_item = new(object) Item(slab_page_index, std::forward<Args>(args)...);
150 _lru.push_front(reinterpret_cast<slab_item_base&>(*new_item));
151 return new_item;
152 }

154 inline
155 std::pair<void *, uint32_t> evict_lru_item(std::function<void (Item& item_ref)>& erase_func) {
156 if (_lru.empty()) {
157 return { nullptr, 0U };
158 }
159
160 Item& victim = reinterpret_cast<Item&>(_lru.back());
161 uint32_t index = victim.get_slab_page_index();
162 SEASTAR_ASSERT(victim.is_unlocked());
163 _lru.erase(_lru.iterator_to(reinterpret_cast<slab_item_base&>(victim)));
164 // WARNING: You need to make sure that erase_func will not release victim back to slab.
165 erase_func(victim);
166
167 return { reinterpret_cast<void*>(&victim), index };
168 }
169public:
170 slab_class(size_t size, uint8_t slab_class_id)
171 : _size(size)
172 , _slab_class_id(slab_class_id)
173 {
174 }
175 slab_class(slab_class&&) = default;
176 ~slab_class() {
177 _free_slab_pages.clear();
178 _lru.clear();
179 }

182 return _size;

186 return _free_slab_pages.empty();
187 }

193 template<typename... Args>
194 Item *create(Args&&... args) {
195 SEASTAR_ASSERT(!_free_slab_pages.empty());
196 auto& desc = _free_slab_pages.back();
197 auto object = desc.allocate_object();
198 if (desc.empty()) {
199 // if empty, remove desc from the list of slab pages with free objects.
200 _free_slab_pages.erase(_free_slab_pages.iterator_to(desc));
201 }
202
203 return create_item(object, desc.index(), std::forward<Args>(args)...);
204 }

206 template<typename... Args>
207 Item *create_from_new_page(uint64_t max_object_size, uint32_t slab_page_index,
208 std::function<void (slab_page_desc& desc)> insert_slab_page_desc,
209 Args&&... args) {
210 // allocate slab page.
211 constexpr size_t alignment = std::alignment_of_v<Item>;
212 void *slab_page = aligned_alloc(alignment, max_object_size);
213 if (!slab_page) {
214 throw std::bad_alloc{};
215 }
216 // allocate descriptor to slab page.
217 slab_page_desc *desc = nullptr;
218 SEASTAR_ASSERT(_size % alignment == 0);
219 auto objects = max_object_size / _size;
220 try {
221 desc = new slab_page_desc(slab_page, objects, _size, _slab_class_id, slab_page_index);
222 } catch (const std::bad_alloc& e) {
223 ::free(slab_page);
224 throw std::bad_alloc{};
225 }
226
227 // Only add the page to the free list if it can hold more than one object.
228 // If it holds only one, it's already full since we are about to return it.
229 if (objects > 1) {
230 _free_slab_pages.push_front(*desc);
231 }
232 insert_slab_page_desc(*desc);
233
234 // first object from the allocated slab page is returned.
235 return create_item(slab_page, slab_page_index, std::forward<Args>(args)...);
236 }

238 template<typename... Args>
239 Item *create_from_lru(std::function<void (Item& item_ref)>& erase_func, Args&&... args) {
240 auto ret = evict_lru_item(erase_func);
241 if (!ret.first) {
242 throw std::bad_alloc{};
243 }
244 return create_item(ret.first, ret.second, std::forward<Args>(args)...);
245 }
246
247 void free_item(Item *item, slab_page_desc& desc) {
248 void *object = item;
249 _lru.erase(_lru.iterator_to(reinterpret_cast<slab_item_base&>(*item)));
250 desc.free_object(object);
251 if (desc.size() == 1) {
252 // push back desc into the list of slab pages with free objects.
253 _free_slab_pages.push_back(desc);
254 }
255 }
256
257 void touch_item(Item *item) {
258 auto& item_ref = reinterpret_cast<slab_item_base&>(*item);
259 _lru.erase(_lru.iterator_to(item_ref));
260 _lru.push_front(item_ref);
261 }
262
263 void remove_item_from_lru(Item *item) {
264 auto& item_ref = reinterpret_cast<slab_item_base&>(*item);
265 _lru.erase(_lru.iterator_to(item_ref));
266 }
267
268 void insert_item_into_lru(Item *item) {
269 auto& item_ref = reinterpret_cast<slab_item_base&>(*item);
270 _lru.push_front(item_ref);
271 }
272
273 void remove_desc_from_free_list(slab_page_desc& desc) {
274 SEASTAR_ASSERT(desc.slab_class_id() == _slab_class_id);
275 _free_slab_pages.erase(_free_slab_pages.iterator_to(desc));
276 }

279template<typename Item>
280class slab_allocator {
281private:
282 std::vector<size_t> _slab_class_sizes;
283 std::vector<slab_class<Item>> _slab_classes;
284 seastar::metrics::metric_groups _metrics;
285 // erase_func() is used to remove the item from the cache using slab.
286 std::function<void (Item& item_ref)> _erase_func;
287 std::vector<slab_page_desc*> _slab_pages_vector;
288 boost::intrusive::list<slab_page_desc,
289 boost::intrusive::member_hook<slab_page_desc, boost::intrusive::list_member_hook<>,
290 &slab_page_desc::_lru_link>> _slab_page_desc_lru;
291 uint64_t _max_object_size;
292 uint64_t _available_slab_pages;
293 struct collectd_stats {
294 uint64_t allocs;
295 uint64_t frees;
296 } _stats;
297 memory::reclaimer *_reclaimer = nullptr;
298 bool _reclaimed = false;
299private:
300 memory::reclaiming_result evict_lru_slab_page() {
301 if (_slab_page_desc_lru.empty()) {
302 // NOTE: Nothing to evict. If this happens, it implies that all
303 // slab pages in the slab are being used at the same time.
304 // That being said, this event is very unlikely to happen.
305 return memory::reclaiming_result::reclaimed_nothing;
306 }
307 // get descriptor of the least-recently-used slab page and related info.
308 auto& desc = _slab_page_desc_lru.back();
309 SEASTAR_ASSERT(desc.refcnt() == 0);
310 uint8_t slab_class_id = desc.slab_class_id();
311 auto slab_class = get_slab_class(slab_class_id);
312 void *slab_page = desc.slab_page();
313
314 auto& free_objects = desc.free_objects();
315 if (!desc.empty()) {
316 // if not empty, remove desc from the list of slab pages with free objects.
317 slab_class->remove_desc_from_free_list(desc);
318 // and sort the array of free objects for binary search later on.
319 std::sort(free_objects.begin(), free_objects.end());
320 }
321 // remove desc from the list of slab page descriptors.
322 _slab_page_desc_lru.erase(_slab_page_desc_lru.iterator_to(desc));
323 // remove desc from the slab page vector.
324 _slab_pages_vector[desc.index()] = nullptr;
325
326 // Iterate through objects in the slab page and if the object is an allocated
327 // item, the item should be removed from LRU and then erased.
328 uintptr_t object = reinterpret_cast<uintptr_t>(slab_page);
329 auto object_size = slab_class->size();
330 auto objects = _max_object_size / object_size;
331 for (auto i = 0u; i < objects; i++, object += object_size) {
332 if (!desc.empty()) {
333 // if binary_search returns true, it means that object at the current
334 // offset isn't an item.
335 if (std::binary_search(free_objects.begin(), free_objects.end(), object)) {
336 continue;
337 }
338 }
339 Item* item = reinterpret_cast<Item*>(object);
340 SEASTAR_ASSERT(item->is_unlocked());
341 slab_class->remove_item_from_lru(item);
342 _erase_func(*item);
343 _stats.frees++;
344 }
345#ifdef SEASTAR_DEBUG
346 printf("lru slab page eviction succeeded! desc_empty?=%d\n", desc.empty());
347#endif
348 ::free(slab_page); // free slab page object
349 delete &desc; // free its descriptor
350 return memory::reclaiming_result::reclaimed_something;
351 }
352
353 /*
354 * Reclaim the least recently used slab page that is unused.
355 */
356 memory::reclaiming_result reclaim() {
357 // once reclaimer was called, slab pages should no longer be allocated, as the
358 // memory used by slab is supposed to be calibrated.
359 _reclaimed = true;
360 // FIXME: Should reclaim() only evict a single slab page at a time?
361 return evict_lru_slab_page();
362 }

364 void initialize_slab_allocator(double growth_factor, uint64_t limit) {
365 constexpr size_t alignment = std::alignment_of_v<Item>;
366 constexpr size_t initial_size = 96;
367 size_t size = initial_size; // initial object size
368 uint8_t slab_class_id = 0U;
369
370 while (_max_object_size / size > 1) {
371 size = align_up(size, alignment);
372 _slab_class_sizes.push_back(size);
373 _slab_classes.emplace_back(size, slab_class_id);
374 size *= growth_factor;
375 SEASTAR_ASSERT(slab_class_id < std::numeric_limits<uint8_t>::max());
376 slab_class_id++;
377 }
378 _slab_class_sizes.push_back(_max_object_size);
379 _slab_classes.emplace_back(_max_object_size, slab_class_id);
380
381 // If slab limit is zero, enable reclaimer.
382 if (!limit) {
383 _reclaimer = new memory::reclaimer([this] { return reclaim(); });
384 } else {
385 _slab_pages_vector.reserve(_available_slab_pages);
386 }
387 }
388
389 slab_class<Item>* get_slab_class(const size_t size) {
390 // given a size, find slab class with binary search.
391 auto i = std::lower_bound(_slab_class_sizes.begin(), _slab_class_sizes.end(), size);
392 if (i == _slab_class_sizes.end()) {
393 return nullptr;
394 }
395 auto dist = std::distance(_slab_class_sizes.begin(), i);
396 return &_slab_classes[dist];
397 }
398
399 slab_class<Item>* get_slab_class(const uint8_t slab_class_id) {
400 SEASTAR_ASSERT(slab_class_id >= 0 && slab_class_id < _slab_classes.size());
401 return &_slab_classes[slab_class_id];
402 }

462 template<typename... Args>
463 Item* create(const size_t size, Args&&... args) {
464 auto slab_class = get_slab_class(size);
465 if (!slab_class) {
466 throw std::bad_alloc{};
467 }
468
469 Item *item = nullptr;
470 if (!slab_class->empty()) {
471 item = slab_class->create(std::forward<Args>(args)...);
472 _stats.allocs++;
473 } else {
474 if (can_allocate_page(*slab_class)) {
475 auto index_to_insert = _slab_pages_vector.size();
476 item = slab_class->create_from_new_page(_max_object_size, index_to_insert,
477 [this](slab_page_desc& desc) {
478 if (_reclaimer) {
479 // insert desc into the LRU list of slab page descriptors.
480 _slab_page_desc_lru.push_front(desc);
481 }
482 // insert desc into the slab page vector.
483 _slab_pages_vector.push_back(&desc);
484 },
485 std::forward<Args>(args)...);
486 if (_available_slab_pages > 0) {
487 _available_slab_pages--;
488 }
489 _stats.allocs++;
490 } else if (_erase_func) {
491 item = slab_class->create_from_lru(_erase_func, std::forward<Args>(args)...);
492 }
493 }
494 return item;
495 }

497 void lock_item(Item *item) {
498 auto& desc = get_slab_page_desc(item);
499 if (_reclaimer) {
500 auto& refcnt = desc.refcnt();
501
502 if (++refcnt == 1) {
503 // remove slab page descriptor from list of slab page descriptors.
504 _slab_page_desc_lru.erase(_slab_page_desc_lru.iterator_to(desc));
505 }
506 }
507 // remove item from the lru of its slab class.
508 auto slab_class = get_slab_class(desc.slab_class_id());
509 slab_class->remove_item_from_lru(item);
510 }
511
512 void unlock_item(Item *item) {
513 auto& desc = get_slab_page_desc(item);
514 if (_reclaimer) {
515 auto& refcnt = desc.refcnt();
516
517 if (--refcnt == 0) {
518 // insert slab page descriptor back into list of slab page descriptors.
519 _slab_page_desc_lru.push_front(desc);
520 }
521 }
522 // insert item into the lru of its slab class.
523 auto slab_class = get_slab_class(desc.slab_class_id());
524 slab_class->insert_item_into_lru(item);
525 }
