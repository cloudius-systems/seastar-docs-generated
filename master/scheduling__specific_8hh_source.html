38struct scheduling_group_specific_thread_local_data {
39 using val_ptr = std::unique_ptr<void, void (*)(void*) noexcept>;
40 using cfg_ptr = lw_shared_ptr<scheduling_group_key_config>;
41
42 struct specific_val {
43 val_ptr valp;
44 cfg_ptr cfg;
45
46 static void free(void* ptr) noexcept {
47 std::free(ptr);
48 }
49
50 specific_val() : valp(nullptr, &free), cfg(nullptr) {}
51
52 specific_val(val_ptr&& valp_, const cfg_ptr& cfg_) : valp(std::move(valp_)), cfg(cfg_) {
53 if (valp && cfg->constructor) {
54 cfg->constructor(valp.get());
55 }
56 }
57
58 ~specific_val() {
59 if (valp && cfg->destructor) {
60 cfg->destructor(valp.get());
61 }
62 }
63
64 specific_val(const specific_val& other) = delete;
65 specific_val& operator=(const specific_val& other) = delete;
66
67 specific_val(specific_val&& other) : valp(std::move(other.valp)), cfg(std::move(other.cfg)) {}
68
69 specific_val& operator=(specific_val&& other) {
70 if (this != &other) {
71 valp = std::move(other.valp);
72 cfg = std::move(other.cfg);
73 }
74 return *this;
75 }
76
77 void* get() { return valp.get(); }
78
79 void rename() {
80 if (valp && cfg->rename) {
81 cfg->rename(valp.get());
82 }
83 }
84 };

86 struct per_scheduling_group {
87 bool queue_is_initialized = false;
93 std::vector<specific_val> specific_vals;
94
95 void rename() {
96 for (auto& v : specific_vals) {
97 v.rename();
98 }
99 }
100 };
101 std::array<per_scheduling_group, max_scheduling_groups()> per_scheduling_group_data;
102 std::map<unsigned long, cfg_ptr> scheduling_group_key_configs;
103};

129template<typename T>
130T* scheduling_group_get_specific_ptr(scheduling_group sg, scheduling_group_key key) noexcept {
131 auto& data = internal::get_scheduling_group_specific_thread_local_data();
132#ifdef SEASTAR_DEBUG
133 SEASTAR_ASSERT(std::type_index(typeid(T)) == data.scheduling_group_key_configs[key.id()]->type_index);
134#endif
135 auto sg_id = internal::scheduling_group_index(sg);
136 if (__builtin_expect(sg_id < data.per_scheduling_group_data.size() &&
137 data.per_scheduling_group_data[sg_id].queue_is_initialized, true)) {
138 return reinterpret_cast<T*>(data.per_scheduling_group_data[sg_id].specific_vals[key.id()].get());
139 }
140 return nullptr;
141}

171template<typename T>
172T& scheduling_group_get_specific(scheduling_group_key key) noexcept {
173 // Unlike internal::scheduling_group_get_specific_ptr, this can
174 // return a reference to an element whose queue_is_initialized is
175 // false.
176 auto& data = internal::get_scheduling_group_specific_thread_local_data();
177 SEASTAR_ASSERT(std::type_index(typeid(T)) == data.scheduling_group_key_configs[key.id()]->type_index);
178 auto sg_id = internal::scheduling_group_index(current_scheduling_group());
179 return *reinterpret_cast<T*>(data.per_scheduling_group_data[sg_id].specific_vals[key.id()].get());
180}

201future<typename function_traits<Reducer>::return_type>
202map_reduce_scheduling_group_specific(Mapper mapper, Reducer reducer,
203 Initial initial_val, scheduling_group_key key) {
204 using per_scheduling_group = internal::scheduling_group_specific_thread_local_data::per_scheduling_group;
205 auto& data = internal::get_scheduling_group_specific_thread_local_data();
206 auto wrapped_mapper = [key, mapper] (per_scheduling_group& psg) {
207 auto id = internal::scheduling_group_key_id(key);
208 return make_ready_future<typename function_traits<Mapper>::return_type>
209 (mapper(*reinterpret_cast<SpecificValType*>(psg.specific_vals[id].get())));
210 };
211
212 return map_reduce(
213 data.per_scheduling_group_data
214 | std::views::filter(std::mem_fn(&per_scheduling_group::queue_is_initialized)),
215 wrapped_mapper, std::move(initial_val), reducer);
216}
