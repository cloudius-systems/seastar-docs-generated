32struct md5_hash {
33 std::array<uint8_t, 16> data;
34};

49class md5_hasher {
50public:
52 static constexpr size_t max_ctx_size = sizeof(void*);
53
55 struct ops {
56 void (*update)(unsigned char* ctx, const void* data, size_t len);
57 md5_hash (*finalize)(unsigned char* ctx);
58 void (*destroy)(unsigned char* ctx);
59 };
60
63 explicit md5_hasher(const ops* ops_table) noexcept : _ops(ops_table) {}
64
65 ~md5_hasher() {
66 if (_ops) {
67 _ops->destroy(_ctx);
68 }
69 }
70
71 md5_hasher(md5_hasher&& o) noexcept : _ops(o._ops) {
72 std::memcpy(_ctx, o._ctx, max_ctx_size);
73 o._ops = nullptr;
74 }
75
76 md5_hasher& operator=(md5_hasher&& o) noexcept {
77 if (this != &o) {
78 if (_ops) {
79 _ops->destroy(_ctx);
80 }
81 _ops = o._ops;
82 std::memcpy(_ctx, o._ctx, max_ctx_size);
83 o._ops = nullptr;
84 }
85 return *this;
86 }
87
88 md5_hasher(const md5_hasher&) = delete;
89 md5_hasher& operator=(const md5_hasher&) = delete;
90
92 void update(const void* data, size_t len) {
93 _ops->update(_ctx, data, len);
94 }
95
99 md5_hash finalize() {
100 auto result = _ops->finalize(_ctx);
101 _ops = nullptr;
102 return result;
103 }
104
106 unsigned char* ctx() noexcept { return _ctx; }
107
108private:
109 const ops* _ops;
110 alignas(void*) unsigned char _ctx[max_ctx_size];
111};
