64// If the future is ready, return true
65// if the future is not ready, chain a continuation to it, and return false
66using when_all_process_element_func = bool (*)(void* future, void* continuation, when_all_state_base* wasb) noexcept;
67
68struct when_all_process_element {
69 when_all_process_element_func func;
70 void* future;
71};
72
73class when_all_state_base {
74 size_t _nr_remain;
75 const when_all_process_element* _processors;
76 void* _continuation;
77public:
78 virtual ~when_all_state_base() {}
79 when_all_state_base(size_t nr_remain, const when_all_process_element* processors, void* continuation)
80 : _nr_remain(nr_remain), _processors(processors), _continuation(continuation) {
81 }
82 virtual task* waiting_task() = 0;
83 void complete_one() noexcept {
84 // We complete in reverse order; if the futures happen to complete
85 // in order, then waiting for the last one will find the rest ready
86 --_nr_remain;
87 while (_nr_remain) {
88 bool ready = process_one(_nr_remain - 1);
89 if (!ready) {
90 return;
91 }
92 --_nr_remain;
93 }
94 if (!_nr_remain) {
95 delete this;
96 }
97 }
98 void do_wait_all() noexcept {
99 ++_nr_remain; // fake pending completion for complete_one()
100 complete_one();
101 }
102 bool process_one(size_t idx) noexcept {
103 auto p = _processors[idx];
104 return p.func(p.future, _continuation, this);
105 }
106};

108template <typename Future>
109class when_all_state_component final : public continuation_base_from_future_t<Future> {
110 when_all_state_base* _base;
111 Future* _final_resting_place;
112public:
113 static bool process_element_func(void* future, void* continuation, when_all_state_base* wasb) noexcept {
114 auto f = reinterpret_cast<Future*>(future);
115 if (f->available()) {
116 return true;
117 } else {
118 auto c = new (continuation) when_all_state_component(wasb, f);
119 set_callback(std::move(*f), c);
120 return false;
121 }
122 }
123 when_all_state_component(when_all_state_base *base, Future* future) noexcept : _base(base), _final_resting_place(future) {}
124 task* waiting_task() noexcept override { return _base->waiting_task(); }
125 virtual void run_and_dispose() noexcept override {
126 using futurator = futurize<Future>;
127 if (__builtin_expect(this->_state.failed(), false)) {
128 *_final_resting_place = futurator::make_exception_future(std::move(this->_state).get_exception());
129 } else {
130 *_final_resting_place = futurator::from_tuple(std::move(this->_state).get_value());
131 }
132 auto base = _base;
133 this->~when_all_state_component();
134 base->complete_one();
135 }
136};

138template<typename ResolvedTupleTransform, typename... Futures>
139class when_all_state : public when_all_state_base {
140 static constexpr size_t nr = sizeof...(Futures);
141 using type = std::tuple<Futures...>;
142 type tuple;
143 // We only schedule one continuation at a time, and store it in _cont.
144 // This way, if while the future we wait for completes, some other futures
145 // also complete, we won't need to schedule continuations for them.
146 alignas(when_all_state_component<Futures>...) std::byte _cont[std::max({sizeof(when_all_state_component<Futures>)...})];
147 when_all_process_element _processors[nr];
148public:
149 typename ResolvedTupleTransform::promise_type p;
150 when_all_state(Futures&&... t) : when_all_state_base(nr, _processors, &_cont), tuple(std::make_tuple(std::move(t)...)) {
151 init_element_processors(std::make_index_sequence<nr>());
152 }
153 virtual ~when_all_state() {
154 ResolvedTupleTransform::set_promise(p, std::move(tuple));
155 }
156 task* waiting_task() noexcept override {
157 return p.waiting_task();
158 }
159private:
160 template <size_t... Idx>
161 void init_element_processors(std::index_sequence<Idx...>) {
162 auto ignore = {
163 0,
164 (_processors[Idx] = when_all_process_element{
165 when_all_state_component<std::tuple_element_t<Idx, type>>::process_element_func,
166 &std::get<Idx>(tuple)
167 }, 0)...
168 };
169 (void)ignore;
170 }
171public:
172 static typename ResolvedTupleTransform::future_type wait_all(Futures&&... futures) noexcept {
173 if ((futures.available() && ...)) {
174 return ResolvedTupleTransform::make_ready_future(std::make_tuple(std::move(futures)...));
175 }
176 auto state = [&] () noexcept {
177 memory::scoped_critical_alloc_section _;
178 return new when_all_state(std::move(futures)...);
179 }();
180 auto ret = state->p.get_future();
181 state->do_wait_all();
182 return ret;
183 }
184};

266// Internal function for when_all().
267template <typename ResolvedVectorTransform, typename Future>
268inline
269typename ResolvedVectorTransform::future_type
270complete_when_all(std::vector<Future>&& futures, typename std::vector<Future>::iterator pos) noexcept {
271 // If any futures are already ready, skip them.
272 while (pos != futures.end() && pos->available()) {
273 ++pos;
274 }
275 // Done?
276 if (pos == futures.end()) {
277 return ResolvedVectorTransform::run(std::move(futures));
278 }
279 // Wait for unready future, store, and continue.
280 return pos->then_wrapped([futures = std::move(futures), pos] (auto fut) mutable {
281 *pos++ = std::move(fut);
282 return complete_when_all<ResolvedVectorTransform>(std::move(futures), pos);
283 });
284}

286template<typename ResolvedVectorTransform, typename FutureIterator, typename Sentinel>
287inline auto
288do_when_all(FutureIterator begin, Sentinel end) noexcept {
289 using itraits = std::iterator_traits<FutureIterator>;
290 auto make_values_vector = [] (size_t size) noexcept {
291 memory::scoped_critical_alloc_section _;
292 std::vector<typename itraits::value_type> ret;
293 ret.reserve(size);
294 return ret;
295 };
296 std::vector<typename itraits::value_type> ret =
297 make_values_vector(iterator_range_estimate_vector_capacity(begin, end));
298 // Important to invoke the *begin here, in case it's a function iterator,
299 // so we launch all computation in parallel.
300 std::ranges::move(begin, end, std::back_inserter(ret));
301 return complete_when_all<ResolvedVectorTransform>(std::move(ret), ret.begin());
302}

361template<typename... Futures>
362class extract_values_from_futures_tuple {
363 static auto transform(std::tuple<Futures...> futures) noexcept {
364 auto prepare_result = [] (auto futures) noexcept {
365 auto fs = tuple_filter_by_type<internal::future_has_value>(std::move(futures));
366 return tuple_map(std::move(fs), [] (auto&& e) {
367 return e.get();
368 });
369 };
370
371 using tuple_futurizer = internal::tuple_to_future<decltype(prepare_result(std::move(futures)))>;
372
373 std::exception_ptr excp;
374 tuple_for_each(futures, [&excp] (auto& f) {
375 if (!excp) {
376 if (f.failed()) {
377 excp = f.get_exception();
378 }
379 } else {
380 f.ignore_ready_future();
381 }
382 });
383 if (excp) {
384 return tuple_futurizer::make_failed(std::move(excp));
385 }
386
387 return tuple_futurizer::make_ready(prepare_result(std::move(futures)));
388 }
389public:
390 using future_type = decltype(transform(std::declval<std::tuple<Futures...>>()));
391 using promise_type = typename future_type::promise_type;
392
393 static void set_promise(promise_type& p, std::tuple<Futures...> tuple) {
394 transform(std::move(tuple)).forward_to(std::move(p));
395 }
396
397 static future_type make_ready_future(std::tuple<Futures...> tuple) noexcept {
398 return transform(std::move(tuple));
399 }
400
401 static future_type current_exception_as_future() noexcept {
402 future_type (*type_deduct)() = current_exception_as_future;
403 return type_deduct();
404 }
405};

407template<typename Future>
408struct extract_values_from_futures_vector {
409 using value_type = decltype(untuple(std::declval<typename Future::tuple_type>()));
410
411 using future_type = future<std::vector<value_type>>;
412
413 static future_type run(std::vector<Future> futures) noexcept {
414 auto make_values_vector = [] (size_t size) noexcept {
415 memory::scoped_critical_alloc_section _;
416 std::vector<value_type> values;
417 values.reserve(size);
418 return values;
419 };
420 std::vector<value_type> values = make_values_vector(futures.size());
421
422 std::exception_ptr excp;
423 for (auto&& f : futures) {
424 if (!excp) {
425 if (f.failed()) {
426 excp = f.get_exception();
427 } else {
428 values.emplace_back(f.get());
429 }
430 } else {
431 f.ignore_ready_future();
432 }
433 }
434 if (excp) {
435 return seastar::make_exception_future<std::vector<value_type>>(std::move(excp));
436 }
437 return make_ready_future<std::vector<value_type>>(std::move(values));
438 }
439
440 static future_type current_exception_as_future() noexcept {
441 return seastar::current_exception_as_future<std::vector<value_type>>();
442 }
443};
