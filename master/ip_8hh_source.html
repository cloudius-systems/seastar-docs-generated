104template <typename InetTraits>
105struct l4connid {
106 using ipaddr = typename InetTraits::address_type;
107 using inet_type = typename InetTraits::inet_type;
108 struct connid_hash;
109
110 ipaddr local_ip;
111 ipaddr foreign_ip;
112 uint16_t local_port;
113 uint16_t foreign_port;
114
115 bool operator==(const l4connid& x) const {
116 return local_ip == x.local_ip
117 && foreign_ip == x.foreign_ip
118 && local_port == x.local_port
119 && foreign_port == x.foreign_port;
120 }
121
122 uint32_t hash(rss_key_type rss_key) {
123 forward_hash hash_data;
124 hash_data.push_back(hton(foreign_ip.ip));
125 hash_data.push_back(hton(local_ip.ip));
126 hash_data.push_back(hton(foreign_port));
127 hash_data.push_back(hton(local_port));
128 return toeplitz_hash(rss_key, hash_data);
129 }
130};

160public:
161 using ipaddr = ipv4_address;
162 using inet_type = ipv4_l4<ip_protocol_num::icmp>;
163 explicit icmp(inet_type& inet) : _inet(inet) {
164 _inet.register_packet_provider([this] {
165 std::optional<ipv4_traits::l4packet> l4p;
166 if (!_packetq.empty()) {
167 l4p = std::move(_packetq.front());
168 _packetq.pop_front();
169 _queue_space.signal(l4p.value().p.len());

171 return l4p;
172 });
173 }

175private:
176 inet_type& _inet;
177 circular_buffer<ipv4_traits::l4packet> _packetq;
178 semaphore _queue_space = {212992};

257struct ipv4_frag_id::hash : private std::hash<ipv4_address>,
258 private std::hash<uint16_t>, private std::hash<uint8_t> {
259 size_t operator()(const ipv4_frag_id& id) const noexcept {
260 using h1 = std::hash<ipv4_address>;
261 using h2 = std::hash<uint16_t>;
262 using h3 = std::hash<uint8_t>;
