30template <typename Func>
31requires std::invocable<Func>
32class lambda_task final : public task {
33 Func _func;
34 using futurator = futurize<std::invoke_result_t<Func>>;
35 typename futurator::promise_type _result;
36public:
37 lambda_task(scheduling_group sg, const Func& func) : task(sg), _func(func) {}
38 lambda_task(scheduling_group sg, Func&& func) : task(sg), _func(std::move(func)) {}
39 typename futurator::type get_future() noexcept { return _result.get_future(); }
40 virtual void run_and_dispose() noexcept override {
41 futurator::invoke(_func).forward_to(std::move(_result));
42 delete this;
43 }
44 virtual task* waiting_task() noexcept override {
45 return _result.waiting_task();
46 }
47};
