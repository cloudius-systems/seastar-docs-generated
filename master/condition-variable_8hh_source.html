85 struct promise_waiter : public waiter, public promise<> {
86 void signal() noexcept override {
87 set_value();
88 // note: we self-delete in either case we are woken
89 // up. See usage below: only the resulting future
90 // state is required once we've left the wait queue
91 delete this;
92 }
93 void set_exception(std::exception_ptr ep) noexcept override {
94 promise<>::set_exception(std::move(ep));
95 // see comment above
96 delete this;
97 }
98 };

100 struct [[nodiscard("must co_await a when() call")]] awaiter : public waiter {
101 condition_variable* _cv;
102 promise<> _p;
103
104 awaiter(condition_variable* cv)
105 : _cv(cv)
106 {}
107
108 void signal() noexcept override {
109 _p.set_value();
110 }
111 void set_exception(std::exception_ptr ep) noexcept override {
112 _p.set_exception(std::move(ep));
113 }
114 auto operator co_await() {
115 if (_cv->check_and_consume_signal()) {
116 return ::seastar::internal::awaiter<false, void>(make_ready_future<>());
117 }
118 _cv->add_waiter(*this);
119 return ::seastar::internal::awaiter<false, void>(_p.get_future());
120 }
121 };

123 template<typename Clock, typename Duration>
124 struct [[nodiscard("must co_await a when() call")]] timeout_awaiter : public awaiter, public timer<Clock> {
125 using my_type = timeout_awaiter<Clock, Duration>;
126 using time_point = std::chrono::time_point<Clock, Duration>;
127
128 time_point _timeout;
129
130 timeout_awaiter(condition_variable* cv, time_point timeout)
131 : awaiter(cv)
132 , _timeout(timeout)
133 {}
134 void signal() noexcept override {
135 this->cancel();
136 awaiter::signal();
137 }
138 void set_exception(std::exception_ptr ep) noexcept override {
139 this->cancel();
140 awaiter::set_exception(std::move(ep));
141 }
142 auto operator co_await() {
143 if (_cv->check_and_consume_signal()) {
144 return ::seastar::internal::awaiter<false, void>(make_ready_future<>());
145 }
146 this->set_callback(std::bind(&waiter::timeout, this));
147 this->arm(_timeout);
148 return awaiter::operator co_await();
149 }

152 template<typename Func, typename Base>
153 struct [[nodiscard("must co_await a when() call")]] predicate_awaiter : public Base {
154 Func _func;
155 template<typename... Args>
156 predicate_awaiter(Func func, Args&& ...args)
157 : Base(std::forward<Args>(args)...)
158 , _func(std::move(func))
159 {}
160 void signal() noexcept override {
161 try {
162 if (_func()) {
163 Base::signal();
164 } else {
165 // must re-enter waiter queue
166 // this maintains "wait" version
167 // semantics of moving to back of queue
168 // if predicate fails
169 Base::_cv->add_waiter(*this);
170 }
171 } catch(...) {
172 Base::set_exception(std::current_exception());
173 }
174 }
175 auto operator co_await() {
176 try {
177 if (_func()) {
178 return ::seastar::internal::awaiter<false, void>(make_ready_future<>());
179 } else {
180 Base::_cv->check_and_consume_signal(); // clear out any signal state
181 return Base::operator co_await();
182 }
183 } catch (...) {
184 return ::seastar::internal::awaiter<false, void>(make_exception_future(std::current_exception()));
185 }
186 }
187 };
