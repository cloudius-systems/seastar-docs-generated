561template<typename T>
562class value {
563public:
564 template<typename W>
565 struct wrap {
566 wrap(const W & v)
567 : _v(v) {
568 }
569 const W & operator()() const {
570 return _v;
571 }
572 const W & _v;
573 };
574
575 typedef std::remove_reference_t<T> value_type;
576 typedef std::conditional_t<
577 std::invocable<T>,
578 value_type, wrap<value_type> > stored_type;
579
580 value(const value_type & t)
581 : value<T>(data_type_for<value_type>::value, t) {
582 }
583 value(data_type type, const value_type & t)
584 : _type(type), _t(t) {
585 }
586 uint64_t operator()() const {
587 auto v = _t();
588 if (_type == data_type::GAUGE) {
589 return convert(double(v));
590 } else {
591 uint64_t u = v;
592 return convert(u);
593 }
594 }
595 operator uint64_t() const {
596 return (*this)();
597 }
598 operator data_type() const {
599 return _type;
600 }
601 data_type type() const {
602 return _type;
603 }
604private:
605 // not super quick value -> protocol endian 64-bit values.
606 template<typename _Iter>
607 void bpack(_Iter s, _Iter e, uint64_t v) const {
608 while (s != e) {
609 *s++ = (v & 0xff);
610 v >>= 8;
611 }
612 }
613 template<std::integral V>
614 uint64_t convert(
615 V v) const {
616 uint64_t i = v;
617 // network byte order
618 return ntohq(i);
619 }
620 template<std::floating_point V>
621 uint64_t convert(
622 V t) const {
623 union {
624 uint64_t i;
625 double v;
626 } v;
627 union {
628 uint64_t i;
629 uint8_t b[8];
630 } u;
631 v.v = t;
632 // intel byte order. could also obviously be faster.
633 // could be ignored if we just assume we're le (for now),
634 // but this is ok me thinks.
635 bpack(std::begin(u.b), std::end(u.b), v.i);
636 return u.i;
637 }
638 ;
639
640 const data_type _type;
641 const stored_type _t;
642};

684template<typename ... Args>
685class values_impl: public value_list {
686public:
687 static const size_t num_values = sizeof...(Args);
688
689 values_impl(description d, Args&& ...args)
690 : value_list(std::move(d))
691 , _values(std::forward<Args>(args)...)
692 {}
693
694 values_impl(values_impl<Args...>&& a) = default;
695 values_impl(const values_impl<Args...>& a) = default;
696
697 size_t size() const override {
698 return num_values;
699 }
700 void types(data_type * p) const override {
701 unpack(_values, [p](Args... args) {
702 std::initializer_list<data_type> tmp = { args... };
703 std::copy(tmp.begin(), tmp.end(), p);
704 });
705 }
706 void values(net::packed<uint64_t> * p) const override {
707 unpack(_values, [p](Args... args) {
708 std::initializer_list<uint64_t> tmp = { args... };
709 std::copy(tmp.begin(), tmp.end(), p);
710 });
711 }
712private:
713 template<typename _Op>
714 void unpack(const std::tuple<Args...>& t, _Op&& op) const {
715 do_unpack(t, std::index_sequence_for<Args...> {}, std::forward<_Op>(op));
716 }
717
718 template<size_t ...S, typename _Op>
719 void do_unpack(const std::tuple<Args...>& t, const std::index_sequence<S...> &, _Op&& op) const {
720 op(std::get<S>(t)...);
721 }
722
723 std::tuple < Args... > _values;
724};
