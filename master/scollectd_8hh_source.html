559template<typename T>
560class value {
561public:
562 template<typename W>
563 struct wrap {
564 wrap(const W & v)
565 : _v(v) {
566 }
567 const W & operator()() const {
568 return _v;
569 }
570 const W & _v;
571 };
572
573 typedef std::remove_reference_t<T> value_type;
574 typedef std::conditional_t<
575 std::invocable<T>,
576 value_type, wrap<value_type> > stored_type;
577
578 value(const value_type & t)
579 : value<T>(data_type_for<value_type>::value, t) {
580 }
581 value(data_type type, const value_type & t)
582 : _type(type), _t(t) {
583 }
584 uint64_t operator()() const {
585 auto v = _t();
586 if (_type == data_type::GAUGE) {
587 return convert(double(v));
588 } else {
589 uint64_t u = v;
590 return convert(u);
591 }
592 }
593 operator uint64_t() const {
594 return (*this)();
595 }
596 operator data_type() const {
597 return _type;
598 }
599 data_type type() const {
600 return _type;
601 }
602private:
603 // not super quick value -> protocol endian 64-bit values.
604 template<typename _Iter>
605 void bpack(_Iter s, _Iter e, uint64_t v) const {
606 while (s != e) {
607 *s++ = (v & 0xff);
608 v >>= 8;
609 }
610 }
611 template<std::integral V>
612 uint64_t convert(
613 V v) const {
614 uint64_t i = v;
615 // network byte order
616 return ntohq(i);
617 }
618 template<std::floating_point V>
619 uint64_t convert(
620 V t) const {
621 union {
622 uint64_t i;
623 double v;
624 } v;
625 union {
626 uint64_t i;
627 uint8_t b[8];
628 } u;
629 v.v = t;
630 // intel byte order. could also obviously be faster.
631 // could be ignored if we just assume we're le (for now),
632 // but this is ok me thinks.
633 bpack(std::begin(u.b), std::end(u.b), v.i);
634 return u.i;
635 }
636 ;
637
638 const data_type _type;
639 const stored_type _t;
640};

682template<typename ... Args>
683class values_impl: public value_list {
684public:
685 static const size_t num_values = sizeof...(Args);
686
687 values_impl(description d, Args&& ...args)
688 : value_list(std::move(d))
689 , _values(std::forward<Args>(args)...)
690 {}
691
692 values_impl(values_impl<Args...>&& a) = default;
693 values_impl(const values_impl<Args...>& a) = default;
694
695 size_t size() const override {
696 return num_values;
697 }
698 void types(data_type * p) const override {
699 unpack(_values, [p](Args... args) {
700 std::initializer_list<data_type> tmp = { args... };
701 std::copy(tmp.begin(), tmp.end(), p);
702 });
703 }
704 void values(net::packed<uint64_t> * p) const override {
705 unpack(_values, [p](Args... args) {
706 std::initializer_list<uint64_t> tmp = { args... };
707 std::copy(tmp.begin(), tmp.end(), p);
708 });
709 }
710private:
711 template<typename _Op>
712 void unpack(const std::tuple<Args...>& t, _Op&& op) const {
713 do_unpack(t, std::index_sequence_for<Args...> {}, std::forward<_Op>(op));
714 }
715
716 template<size_t ...S, typename _Op>
717 void do_unpack(const std::tuple<Args...>& t, const std::index_sequence<S...> &, _Op&& op) const {
718 op(std::get<S>(t)...);
719 }
