73struct listener_entry {
74 server_socket socket;
75 socket_address addr;
76 bool tls;
77 // The position this listener was added at. Only the deprecated do_accepts(int) and
78 // accept_loop(int) still name a listener this way; nothing renumbers.
79 size_t index;
80 future<> accept_loop = make_ready_future<>();
81 // Held by every connection accepted on this listener, so that removing it can
82 // wait for them. Closed before the socket is aborted, so that it also says the
83 // listener is stopping: the accept loop reads the abort as being asked to stop
84 // rather than as a failure. The loop otherwise runs until the server's task gate
85 // closes, which is how stopping the whole server stops it; taking one listener
86 // away leaves that gate open.
87 gate connections_gate;
88 // Resolved once this listener has stopped accepting and its connections have
89 // drained, whether that came from its own abort_source or from stopping the
90 // whole server. What serve() hands back to its caller. The flag is what keeps the
91 // two paths from resolving it twice.
92 shared_promise<> removed;
93 bool removal_reported = false;
94
95 listener_entry(server_socket socket_, socket_address addr_, bool tls_, size_t index_)
96 : socket(std::move(socket_)), addr(addr_), tls(tls_), index(index_) {}
97};
