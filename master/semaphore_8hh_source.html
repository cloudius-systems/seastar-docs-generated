164 std::exception_ptr get_timeout_exception() {
165 try {
166 return std::make_exception_ptr(this->timeout());
167 } catch (...) {
168 return std::make_exception_ptr(semaphore_timed_out());
169 }

171 std::exception_ptr get_aborted_exception() {
172 if constexpr (internal::has_aborted<exception_factory>::value) {
173 try {
174 return std::make_exception_ptr(this->aborted());
175 } catch (...) {
176 return std::make_exception_ptr(semaphore_aborted());

182 struct expiry_handler {
183 basic_semaphore& sem;
184 void operator()(entry& e, const std::optional<std::exception_ptr>& ex) noexcept {
185 if (e.timer) {
186 e.pr.set_exception(sem.get_timeout_exception());
187 } else if (ex) {
188 e.pr.set_exception(*ex);
189 } else if (sem._ex) {
190 e.pr.set_exception(sem._ex);
191 } else {
192 e.pr.set_exception(sem.get_aborted_exception());
193 }
194 }
195 // After a waiter left the wait list because it timed out or was
196 // aborted, the waiters that were behind it may now be served.
197 void after_abort() noexcept {
198 sem.wake_waiters();
199 }
200 };
201 internal::abortable_fifo<entry, expiry_handler> _wait_list;

231 bool has_available_units(size_t nr) const noexcept {
232 return _count >= 0 && (static_cast<size_t>(_count) >= nr);
233 }
234 bool may_proceed(size_t nr) const noexcept {
235 return has_available_units(nr) && _wait_list.empty();
236 }
237 // Serves the waiters at the front of the wait list, in order, while the
238 // available units are enough for them.
239 void wake_waiters() noexcept {
240 while (!_wait_list.empty() && has_available_units(_wait_list.front().nr)) {
241 auto& x = _wait_list.front();
242 _count -= x.nr;
243 x.pr.set_value();
244 _wait_list.pop_front();
245 }
246 }

295 // semaphore cannot be moved with non-empty waiting list
296 SEASTAR_ASSERT(_wait_list.empty());
297 SEASTAR_ASSERT(other._wait_list.empty());
298 if (this != &other) {
299 exception_factory::operator=(other);

302 _used = std::move(other._used);
303 }

333 future<> wait(time_point timeout, size_t nr = 1) noexcept {
334 _used.use();
335 if (may_proceed(nr)) {
336 _count -= nr;
337 return make_ready_future<>();
338 }
339 if (_ex) {
340 return make_exception_future(_ex);
341 }
342 if (Clock::now() >= timeout) [[unlikely]] {
343 return make_exception_future(get_timeout_exception());
344 }
345 try {
346 entry& e = _wait_list.emplace_back(promise<>(), nr);
347 auto f = e.pr.get_future();
348 if (timeout != time_point::max()) {
349 e.timer.emplace(timeout);
350 abort_source& as = e.timer->abort_source();
351 _wait_list.make_back_abortable(as);
352 }
353 return f;
354 } catch (...) {
355 return make_exception_future(std::current_exception());
356 }
357 }
358
372 future<> wait(abort_source& as, size_t nr = 1) noexcept {
373 _used.use();
374 if (may_proceed(nr)) {
375 _count -= nr;
376 return make_ready_future<>();
377 }
378 if (_ex) {
379 return make_exception_future(_ex);
380 }
381 if (as.abort_requested()) [[unlikely]] {
382 return make_exception_future(get_aborted_exception());
383 }
384 try {
385 entry& e = _wait_list.emplace_back(promise<>(), nr);
386 // taking future here since make_back_abortable may expire the entry
387 auto f = e.pr.get_future();
388 _wait_list.make_back_abortable(as);
389 return f;
390 } catch (...) {
391 return make_exception_future(std::current_exception());
392 }
393 }

514void
515basic_semaphore<ExceptionFactory, Clock>::broken(std::exception_ptr xp) noexcept {

527template<typename ExceptionFactory = semaphore_default_exception_factory, typename Clock = typename timer<>::clock>
528class semaphore_units {
529 basic_semaphore<ExceptionFactory, Clock>* _sem;
530 size_t _n;
531
532 semaphore_units(basic_semaphore<ExceptionFactory, Clock>* sem, size_t n) noexcept : _sem(sem), _n(n) {}
533public:
534 semaphore_units() noexcept : semaphore_units(nullptr, 0) {}
535 semaphore_units(basic_semaphore<ExceptionFactory, Clock>& sem, size_t n) noexcept : semaphore_units(&sem, n) {}
536 semaphore_units(semaphore_units&& o) noexcept : _sem(o._sem), _n(std::exchange(o._n, 0)) {
537 }
538 semaphore_units& operator=(semaphore_units&& o) noexcept {
539 if (this != &o) {
540 return_all();
541 _sem = o._sem;
542 _n = std::exchange(o._n, 0);
543 }
544 return *this;
545 }
546 semaphore_units(const semaphore_units&) = delete;
547 ~semaphore_units() noexcept {
548 return_all();
549 }
557 size_t return_units(size_t units) {
558 if (units > _n) {
559 throw std::invalid_argument("Cannot take more units than those protected by the semaphore");
560 }
561 _n -= units;
562 _sem->signal(units);
563 return _n;
564 }
565
566 void return_all() noexcept {
567 if (_n) {
568 _sem->signal(_n);
569 _n = 0;
570 }
571 }
572
575 size_t release() noexcept {
576 return std::exchange(_n, 0);
577 }
578
586 semaphore_units split(size_t units) {
587 if (units > _n) {
588 throw std::invalid_argument("Cannot take more units than those protected by the semaphore");
589 }
590 _n -= units;
591 return semaphore_units(_sem, units);
592 }
593
598 void adopt(semaphore_units&& other) noexcept {
599 SEASTAR_ASSERT(other._sem == _sem);
600 _n += other.release();
601 }
602
604 size_t count() const noexcept {
605 return _n;
606 }
607
609 explicit operator bool() const noexcept {
610 return _n != 0;
611 }
612};
