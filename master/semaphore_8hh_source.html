318 future<> wait(time_point timeout, size_t nr = 1) noexcept {
319 _used.use();
320 if (may_proceed(nr)) {
321 _count -= nr;
322 return make_ready_future<>();
323 }
324 if (_ex) {
325 return make_exception_future(_ex);
326 }
327 if (Clock::now() >= timeout) [[unlikely]] {
328 return make_exception_future(get_timeout_exception());
329 }
330 try {
331 entry& e = _wait_list.emplace_back(promise<>(), nr);
332 auto f = e.pr.get_future();
333 if (timeout != time_point::max()) {
334 e.timer.emplace(timeout);
335 abort_source& as = e.timer->abort_source();
336 _wait_list.make_back_abortable(as);
337 }
338 return f;
339 } catch (...) {
340 return make_exception_future(std::current_exception());
341 }
342 }
343
357 future<> wait(abort_source& as, size_t nr = 1) noexcept {
358 _used.use();
359 if (may_proceed(nr)) {
360 _count -= nr;
361 return make_ready_future<>();
362 }
363 if (_ex) {
364 return make_exception_future(_ex);
365 }
366 if (as.abort_requested()) [[unlikely]] {
367 return make_exception_future(get_aborted_exception());
368 }
369 try {
370 entry& e = _wait_list.emplace_back(promise<>(), nr);
371 // taking future here since make_back_abortable may expire the entry
372 auto f = e.pr.get_future();
373 _wait_list.make_back_abortable(as);
374 return f;
375 } catch (...) {
376 return make_exception_future(std::current_exception());
377 }
378 }

405 void signal(size_t nr = 1) noexcept {
406 if (_ex) {
407 return;
408 }
409 _count += nr;
410 while (!_wait_list.empty() && has_available_units(_wait_list.front().nr)) {
411 auto& x = _wait_list.front();
412 _count -= x.nr;
413 x.pr.set_value();
414 _wait_list.pop_front();
415 }
416 }
417
419 //
425 void consume(size_t nr = 1) noexcept {
426 _used.use();
427 if (_ex) {
428 return;
429 }
430 _count -= nr;
431 }

443 bool try_wait(size_t nr = 1) noexcept {
444 _used.use();
445 if (may_proceed(nr)) {
446 _count -= nr;
447 return true;
448 } else {

517template<typename ExceptionFactory = semaphore_default_exception_factory, typename Clock = typename timer<>::clock>
518class semaphore_units {
519 basic_semaphore<ExceptionFactory, Clock>* _sem;
520 size_t _n;
521
522 semaphore_units(basic_semaphore<ExceptionFactory, Clock>* sem, size_t n) noexcept : _sem(sem), _n(n) {}
523public:
524 semaphore_units() noexcept : semaphore_units(nullptr, 0) {}
525 semaphore_units(basic_semaphore<ExceptionFactory, Clock>& sem, size_t n) noexcept : semaphore_units(&sem, n) {}
526 semaphore_units(semaphore_units&& o) noexcept : _sem(o._sem), _n(std::exchange(o._n, 0)) {
527 }
528 semaphore_units& operator=(semaphore_units&& o) noexcept {
529 if (this != &o) {
530 return_all();
531 _sem = o._sem;
532 _n = std::exchange(o._n, 0);
533 }
534 return *this;
535 }
536 semaphore_units(const semaphore_units&) = delete;
537 ~semaphore_units() noexcept {
538 return_all();
539 }
547 size_t return_units(size_t units) {
548 if (units > _n) {
549 throw std::invalid_argument("Cannot take more units than those protected by the semaphore");
550 }
551 _n -= units;
552 _sem->signal(units);
553 return _n;
554 }
555
556 void return_all() noexcept {
557 if (_n) {
558 _sem->signal(_n);
559 _n = 0;
560 }
561 }
562
565 size_t release() noexcept {
566 return std::exchange(_n, 0);
567 }
568
576 semaphore_units split(size_t units) {
577 if (units > _n) {
578 throw std::invalid_argument("Cannot take more units than those protected by the semaphore");
579 }
580 _n -= units;
581 return semaphore_units(_sem, units);
582 }
583
588 void adopt(semaphore_units&& other) noexcept {
589 SEASTAR_ASSERT(other._sem == _sem);
590 _n += other.release();
591 }
592
594 size_t count() const noexcept {
595 return _n;
596 }
597
599 explicit operator bool() const noexcept {
600 return _n != 0;
601 }
