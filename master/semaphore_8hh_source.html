183 struct expiry_handler {
184 basic_semaphore& sem;
185 void operator()(entry& e, const std::optional<std::exception_ptr>& ex) noexcept {
186 if (e.timer) {
187 e.pr.set_exception(sem.get_timeout_exception());
188 } else if (ex) {
189 e.pr.set_exception(*ex);
190 } else if (sem._ex) {
191 e.pr.set_exception(sem._ex);
192 } else {
193 e.pr.set_exception(sem.get_aborted_exception());
194 }
195 }

303 future<> wait(size_t nr = 1) noexcept {
304 return wait(time_point::max(), nr);

319 future<> wait(time_point timeout, size_t nr = 1) noexcept {
320 _used.use();
321 if (may_proceed(nr)) {
322 _count -= nr;
323 return make_ready_future<>();
324 }
325 if (_ex) {
326 return make_exception_future(_ex);
327 }
328 if (Clock::now() >= timeout) [[unlikely]] {
329 return make_exception_future(get_timeout_exception());
330 }
331 try {
332 entry& e = _wait_list.emplace_back(promise<>(), nr);
333 auto f = e.pr.get_future();
334 if (timeout != time_point::max()) {
335 e.timer.emplace(timeout);
336 abort_source& as = e.timer->abort_source();
337 _wait_list.make_back_abortable(as);
338 }
339 return f;
340 } catch (...) {
341 return make_exception_future(std::current_exception());
342 }
343 }
344
358 future<> wait(abort_source& as, size_t nr = 1) noexcept {
359 _used.use();
360 if (may_proceed(nr)) {
361 _count -= nr;
362 return make_ready_future<>();
363 }
364 if (_ex) {
365 return make_exception_future(_ex);
366 }
367 if (as.abort_requested()) [[unlikely]] {
368 return make_exception_future(get_aborted_exception());
369 }
370 try {
371 entry& e = _wait_list.emplace_back(promise<>(), nr);
372 // taking future here since make_back_abortable may expire the entry
373 auto f = e.pr.get_future();
374 _wait_list.make_back_abortable(as);
375 return f;
376 } catch (...) {
377 return make_exception_future(std::current_exception());
378 }
379 }

406 void signal(size_t nr = 1) noexcept {
407 if (_ex) {
408 return;
409 }
410 _count += nr;
411 while (!_wait_list.empty() && has_available_units(_wait_list.front().nr)) {
412 auto& x = _wait_list.front();
413 _count -= x.nr;
414 x.pr.set_value();
415 _wait_list.pop_front();
416 }
417 }
418
420 //
426 void consume(size_t nr = 1) noexcept {
427 _used.use();
428 if (_ex) {
429 return;
430 }
431 _count -= nr;
432 }

444 bool try_wait(size_t nr = 1) noexcept {
445 _used.use();
446 if (may_proceed(nr)) {
447 _count -= nr;
448 return true;
449 } else {
450 return false;
451 }

505void
506basic_semaphore<ExceptionFactory, Clock>::broken(std::exception_ptr xp) noexcept {
507 static_assert(std::is_nothrow_copy_constructible_v<std::exception_ptr>);
508 _ex = xp;

510 while (!_wait_list.empty()) {

518template<typename ExceptionFactory = semaphore_default_exception_factory, typename Clock = typename timer<>::clock>
519class semaphore_units {
520 basic_semaphore<ExceptionFactory, Clock>* _sem;
521 size_t _n;
522
523 semaphore_units(basic_semaphore<ExceptionFactory, Clock>* sem, size_t n) noexcept : _sem(sem), _n(n) {}
524public:
525 semaphore_units() noexcept : semaphore_units(nullptr, 0) {}
526 semaphore_units(basic_semaphore<ExceptionFactory, Clock>& sem, size_t n) noexcept : semaphore_units(&sem, n) {}
527 semaphore_units(semaphore_units&& o) noexcept : _sem(o._sem), _n(std::exchange(o._n, 0)) {
528 }
529 semaphore_units& operator=(semaphore_units&& o) noexcept {
530 if (this != &o) {
531 return_all();
532 _sem = o._sem;
533 _n = std::exchange(o._n, 0);
534 }
535 return *this;
536 }
537 semaphore_units(const semaphore_units&) = delete;
538 ~semaphore_units() noexcept {
539 return_all();
540 }
548 size_t return_units(size_t units) {
549 if (units > _n) {
550 throw std::invalid_argument("Cannot take more units than those protected by the semaphore");
551 }
552 _n -= units;
553 _sem->signal(units);
554 return _n;
555 }
557 void return_all() noexcept {
558 if (_n) {
559 _sem->signal(_n);
560 _n = 0;
561 }
562 }
566 size_t release() noexcept {
567 return std::exchange(_n, 0);
568 }
577 semaphore_units split(size_t units) {
578 if (units > _n) {
579 throw std::invalid_argument("Cannot take more units than those protected by the semaphore");
580 }
581 _n -= units;
582 return semaphore_units(_sem, units);
583 }
589 void adopt(semaphore_units&& other) noexcept {
590 SEASTAR_ASSERT(other._sem == _sem);
591 _n += other.release();
592 }
593
595 size_t count() const noexcept {
596 return _n;
597 }
598
600 explicit operator bool() const noexcept {
601 return _n != 0;
602 }
