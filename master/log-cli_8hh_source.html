110// \brief Parse associations from loggers to log-levels and write the resulting pairs to the output iterator.
111//
112// \throws \c std::runtime_error for an invalid log-level.
113//
114template <class OutputIter>
115void parse_logger_levels(const program_options::string_map& levels, OutputIter out) {
116 std::for_each(levels.begin(), levels.end(), [&out](auto&& pair) {
117 *out++ = std::make_pair(pair.first, parse_log_level(pair.second));
118 });
