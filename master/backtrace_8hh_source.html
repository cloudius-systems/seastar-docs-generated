60// Invokes func for each frame passing it as argument.
61// incremental=false is the default mode and simply calls ::backtrace once
62// and then calls func for each frame. If the ::backtrace call crashes,
63// func will not be called.
64// incremental=true will call ::backtrace in a loop, increasing the number of
65// frames to capture by one each time. This is useful for cases where
66// ::backtrace itself may crash, e.g., in a crash handler, in this case the
67// func will be called for each frame which didn't crash. See #2710. This
68// is much slower, however.
69template<typename Func>
70void backtrace(Func&& func, bool incremental = false) noexcept(noexcept(func(frame()))) {
71#ifdef HAVE_EXECINFO
72 constexpr size_t max_backtrace = 100;
73 void* buffer[max_backtrace];
74
75 if (incremental) {
76 for (size_t last_frame = 1; last_frame <= max_backtrace; ++last_frame) {
77 int n = ::backtrace(buffer, last_frame);
78 if (n < static_cast<int>(last_frame)) {
79 return;
80 }
81 auto ip = reinterpret_cast<uintptr_t>(buffer[last_frame-1]);
82 func(decorate(ip - 1));
83 }
84 } else {
85 int n = ::backtrace(buffer, max_backtrace);
86 for (int i = 0; i < n; ++i) {
87 auto ip = reinterpret_cast<uintptr_t>(buffer[i]);
88 func(decorate(ip - 1));
89 }
90 }
91#else
92// Not implemented yet
93#define SEASTAR_BACKTRACE_UNIMPLEMENTED
94#endif
95}

97// Represents a call stack of a single thread.
98class simple_backtrace {
99public:
100 using vector_type = boost::container::static_vector<frame, 64>;
101private:
102 vector_type _frames;
103 size_t _hash = 0;
104 char _delimeter = ' ';
105private:
106 size_t calculate_hash() const noexcept;
107public:
108 simple_backtrace(vector_type f) noexcept : _frames(std::move(f)), _hash(calculate_hash()) {}
109 simple_backtrace() noexcept = default;
110 [[deprecated]] simple_backtrace(vector_type f, char delimeter) : _frames(std::move(f)), _hash(calculate_hash()), _delimeter(delimeter) {}
111 [[deprecated]] simple_backtrace(char delimeter) : _delimeter(delimeter) {}
112
113 size_t hash() const noexcept { return _hash; }
114 char delimeter() const noexcept { return _delimeter; }
115
116 friend fmt::formatter<simple_backtrace>;
117
118 bool operator==(const simple_backtrace& o) const noexcept {
119 return _hash == o._hash && _frames == o._frames;
120 }
121
122 bool operator!=(const simple_backtrace& o) const noexcept {
123 return !(*this == o);
124 }
125};

150// Extended backtrace which consists of a backtrace of the currently running task
151// and information about the chain of tasks waiting for the current operation to complete.
152class tasktrace {
153public:
154 struct entry {
155 std::variant<shared_backtrace, task_entry> backtrace_or_task_entry;
156 std::source_location resume_point;
157
158 bool operator == (const entry &other) const {
159 return backtrace_or_task_entry == other.backtrace_or_task_entry
160 && resume_point.line() == other.resume_point.line()
161 && resume_point.column() == other.resume_point.column()
162 && std::string_view{ resume_point.file_name() } == std::string_view{ other.resume_point.file_name() };
163 }
164 };
165 using vector_type = boost::container::static_vector<entry, 16>;
166private:
167 simple_backtrace _main;
168 vector_type _prev;
169 scheduling_group _sg;
170 size_t _hash;
171public:
172 tasktrace() = default;
173 tasktrace(simple_backtrace main, vector_type prev, size_t prev_hash, scheduling_group sg) noexcept;
174 tasktrace(const tasktrace&) = default;
175 tasktrace& operator=(const tasktrace&) = default;
176 ~tasktrace();
177
178 size_t hash() const noexcept { return _hash; }
179 char delimeter() const noexcept { return _main.delimeter(); }
180
181 friend fmt::formatter<tasktrace>;
182
183 bool operator==(const tasktrace& o) const noexcept;
184
185 bool operator!=(const tasktrace& o) const noexcept {
186 return !(*this == o);
187 }
188};

272template <class Exc, typename... Args>
273std::exception_ptr make_backtraced_exception_ptr(Args&&... args) {
274 using exc_type = std::decay_t<Exc>;
275 static_assert(std::is_base_of_v<std::exception, exc_type>,
276 "throw_with_backtrace only works with exception types");
277 return std::make_exception_ptr<internal::backtraced<exc_type>>(Exc(std::forward<Args>(args)...));
278}
