49template<typename ExceptionFactory = default_timeout_exception_factory, typename Clock, typename Duration, typename... T>
50future<T...> with_timeout(std::chrono::time_point<Clock, Duration> timeout, future<T...> f) {
51 if (f.available()) {
52 return f;
53 }
54 auto pr = std::make_unique<promise<T...>>();
55 auto result = pr->get_future();
56 timer<Clock> timer([&pr = *pr] {
57 pr.set_exception(std::make_exception_ptr(ExceptionFactory::timeout()));
58 });
59 timer.arm(timeout);
60 // Future is returned indirectly.
61 (void)f.then_wrapped([pr = std::move(pr), timer = std::move(timer)] (auto&& f) mutable {
62 if (timer.cancel()) {
63 f.forward_to(std::move(*pr));
64 } else {
65 f.ignore_ready_future();
66 }
67 });
68 return result;
69}
