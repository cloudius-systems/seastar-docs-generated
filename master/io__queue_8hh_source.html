88 // _pending represents a reservation of tokens from the bucket.
89 //
90 // In the "dispatch timeline" defined by the growing bucket head of the group,

92 // to this queue.
93 //
94 // For example, if:
95 // _group._token_bucket.head == 300

98 // then the reservation is [200, 700), 100 tokens are ready to be dispatched by this queue,
99 // and another 400 tokens are going to be appear soon. (And after that, this queue

258class io_throttler {
259public:
260 using capacity_t = fair_queue_entry::capacity_t;
261 using clock_type = std::chrono::steady_clock;
262
263 /*
264 * tldr; The math
265 *
266 * Bw, Br -- write/read bandwidth (bytes per second)
267 * Ow, Or -- write/read iops (ops per second)
268 *
269 * xx_max -- their maximum values (configured)
270 *
271 * Throttling formula:
272 *
273 * Bw/Bw_max + Br/Br_max + Ow/Ow_max + Or/Or_max <= K
274 *
275 * where K is the scalar value <= 1.0 (also configured)
276 *
277 * Bandwidth is bytes time derivatite, iops is ops time derivative, i.e.
278 * Bx = d(bx)/dt, Ox = d(ox)/dt. Then the formula turns into
279 *
280 * d(bw/Bw_max + br/Br_max + ow/Ow_max + or/Or_max)/dt <= K
281 *
282 * Fair queue tickets are {w, s} weight-size pairs that are
283 *
284 * s = read_base_count * br, for reads
285 * Br_max/Bw_max * read_base_count * bw, for writes
286 *
287 * w = read_base_count, for reads
288 * Or_max/Ow_max * read_base_count, for writes
289 *
290 * Thus the formula turns into
291 *
292 * d(sum(w/W + s/S))/dr <= K
293 *
294 * where {w, s} is the ticket value if a request and sum summarizes the
295 * ticket values from all the requests seen so far, {W, S} is the ticket
296 * value that corresonds to a virtual summary of Or_max requests of
297 * Br_max size total.
298 */
299
300 /*
301 * The normalization results in a float of the 2^-30 seconds order of
302 * magnitude. Not to invent float point atomic arithmetics, the result
303 * is converted to an integer by multiplying by a factor that's large
304 * enough to turn these values into a non-zero integer.
305 *
306 * Also, the rates in bytes/sec when adjusted by io-queue according to
307 * multipliers become too large to be stored in 32-bit ticket value.
308 * Thus the rate resolution is applied. The t.bucket is configured with a
309 * time period for which the speeds from F (in above formula) are taken.
310 */
311
312 static constexpr float fixed_point_factor = float(1 << 24);
313 using rate_resolution = std::milli;
314 using token_bucket_t = internal::shared_token_bucket<capacity_t, rate_resolution, internal::capped_release::no>;
315
316private:
317
318 /*
319 * The dF/dt <= K limitation is managed by the modified token bucket
320 * algo where tokens are ticket.normalize(cost_capacity), the refill
321 * rate is K.
322 *
323 * The token bucket algo must have the limit on the number of tokens
324 * accumulated. Here it's configured so that it accumulates for the
325 * latency_goal duration.
326 *
327 * The replenish threshold is the minimal number of tokens to put back.
328 * It's reserved for future use to reduce the load on the replenish
329 * timestamp.
330 *
331 * The timestamp, in turn, is the time when the bucket was replenished
332 * last. Every time a shard tries to get tokens from bucket it first
333 * tries to convert the time that had passed since this timestamp
334 * into more tokens in the bucket.
335 */
336
337 token_bucket_t _token_bucket;
338 const capacity_t _per_tick_threshold;
339
340public:
341
342 // Convert internal capacity value back into the real token
343 static double capacity_tokens(capacity_t cap) noexcept {
344 return (double)cap / fixed_point_factor / token_bucket_t::rate_cast(std::chrono::seconds(1)).count();
345 }
346
347 // Convert floating-point tokens into the token bucket capacity
348 static capacity_t tokens_capacity(double tokens) noexcept {
349 return tokens * token_bucket_t::rate_cast(std::chrono::seconds(1)).count() * fixed_point_factor;
350 }
351
352 auto capacity_duration(capacity_t cap) const noexcept {
353 return _token_bucket.duration_for(cap);
354 }
