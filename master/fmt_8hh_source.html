24// Single point through which Seastar pulls in {fmt}.
25//
26// All Seastar code that needs {fmt} includes this header instead of the
27// individual <fmt/*.h> headers. When SEASTAR_IMPORT_FMT is defined (see the
28// Seastar_IMPORT_FMT option in CMakeLists.txt) this imports fmt as a C++20
29// module; otherwise it is a plain textual include of every fmt header Seastar
30// uses. Consumers must pick one per translation unit -- mixing `import fmt;`
31// with a textual <fmt/*.h> in the same TU is a redefinition error -- which is
32// exactly why every Seastar fmt include is funnelled through here.
33//
34// The fmt module has to be one built with FMT_ATTACH_TO_GLOBAL_MODULE, so that
35// its declarations stay traditionally mangled and interchangeable with what a
36// textual include yields. Seastar's own sources include fmt textually either
37// way, and its ABI mentions fmt types, so a translation unit importing a
38// module-attached fmt could not link against the library. The build arranges
39// this (see cmake/SeastarDependencies.cmake); it is worth knowing about when
40// compiling fmt's module interface unit by hand.
41//
42// It also defines SEASTAR_FMT_VERSION, which the version-dependent parts of
43// Seastar's public headers key off instead of fmt's own FMT_VERSION. The
44// build may define it (as an integer, MMmmpp, like FMT_VERSION) for setups
45// where fmt does not supply it; otherwise it is taken from fmt. Importing fmt
46// is such a setup: fmt's preprocessor macros are not part of the module, so
47// the version has to come from the build, derived from the fmt that Seastar
48// was configured against (Seastar does not bundle fmt, so it cannot be
49// hardcoded here).
50
51#ifdef SEASTAR_IMPORT_FMT
52
53#ifndef SEASTAR_FMT_VERSION
54#error "SEASTAR_IMPORT_FMT requires the build to define SEASTAR_FMT_VERSION (see Seastar_IMPORT_FMT)"
55#endif
56
57import fmt;
58
59#else

71#define SEASTAR_FMT_VERSION FMT_VERSION
72#endif
