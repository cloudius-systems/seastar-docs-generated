248// The state of an async() call. Only the derived class async_work depends on
249// the function that async() runs; everything else, including run_async(),
250// depends only on its return type, so it is compiled once for each return
251// type instead of once for every function passed to async().
252template <typename ReturnType>
253struct async_work_base {
254 thread_attributes attr;
255 promise<ReturnType> pr{};
256 thread th{};
257 explicit async_work_base(thread_attributes a) : attr(std::move(a)) {}
258 virtual ~async_work_base() = default;
259 // Calls the function, in the thread, and forwards its result to pr.
260 virtual void run() = 0;
261};
262
263template <typename ReturnType, typename Func, typename... Args>
264struct async_work final : async_work_base<ReturnType> {
265 Func func;
266 std::tuple<Args...> args;
267 async_work(thread_attributes a, Func&& f, Args&&... as)
268 : async_work_base<ReturnType>(std::move(a)), func(std::forward<Func>(f)), args(std::forward<Args>(as)...) {}
269 void run() override {
270 futurize<ReturnType>::apply(std::move(func), std::move(args)).forward_to(std::move(this->pr));
271 }
272};
273
274// Runs wp->run() in a new thread, and returns its result.
275template <typename ReturnType>
276futurize_t<ReturnType>
277run_async(std::unique_ptr<async_work_base<ReturnType>> wp) {
278 auto* w = wp.get();
279 auto ret = w->pr.get_future();
280 w->th = thread(std::move(w->attr), [w] {
281 w->run();
282 });
283 return w->th.join().then([ret = std::move(ret)] () mutable {
284 return std::move(ret);
285 }).finally([wp = std::move(wp)] {});
286}

314template <typename Func, typename... Args>
315inline
316futurize_t<std::invoke_result_t<Func, Args...>>
317async(thread_attributes attr, Func&& func, Args&&... args) noexcept {
318 using return_type = std::invoke_result_t<Func, Args...>;
319 try {
320 return internal::run_async(std::unique_ptr<internal::async_work_base<return_type>>(
321 new internal::async_work<return_type, Func, Args...>(std::move(attr), std::forward<Func>(func), std::forward<Args>(args)...)));
322 } catch (...) {
323 return futurize<return_type>::make_exception_future(std::current_exception());
324 }
325}
