268template <typename Func, typename... Args>
269inline
270futurize_t<std::invoke_result_t<Func, Args...>>
271async(thread_attributes attr, Func&& func, Args&&... args) noexcept {
272 using return_type = std::invoke_result_t<Func, Args...>;
273 struct work {
274 thread_attributes attr;
275 Func func;
276 std::tuple<Args...> args;
277 promise<return_type> pr{};
278 thread th{};
279 };
280
281 try {
282 auto wp = std::make_unique<work>(work{std::move(attr), std::forward<Func>(func), std::forward_as_tuple(std::forward<Args>(args)...)});
283 auto& w = *wp;
284 auto ret = w.pr.get_future();
285 w.th = thread(std::move(w.attr), [&w] {
286 futurize<return_type>::apply(std::move(w.func), std::move(w.args)).forward_to(std::move(w.pr));
287 });
288 return w.th.join().then([ret = std::move(ret)] () mutable {
289 return std::move(ret);
290 }).finally([wp = std::move(wp)] {});
291 } catch (...) {
292 return futurize<return_type>::make_exception_future(std::current_exception());
293 }
294}
