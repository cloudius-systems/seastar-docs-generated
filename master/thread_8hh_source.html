86// Internal class holding thread state. We can't hold this in
87// \c thread itself because \c thread is movable, and we want pointers

91 void operator()(char *ptr) const noexcept;
92 int valgrind_id;
93 stack_deleter(int valgrind_id);

97 stack_holder _stack;
98 noncopyable_function<void ()> _func;
99 jmp_buf_link _context;
100 promise<> _done;
101 bool _joined = false;

104 using all_thread_list = boost::intrusive::list<thread_context,
105 boost::intrusive::member_hook<thread_context, boost::intrusive::list_member_hook<>,

107 boost::intrusive::constant_time_size<false>>;

109 static thread_local all_thread_list _all_threads;
110private:
111 static void s_main(int lo, int hi); // all parameters MUST be 'int' for makecontext

113 void main();
114 stack_holder make_stack(size_t stack_size);
115 virtual void run_and_dispose() noexcept override; // from task class

119 void switch_in();

121 bool should_yield() const;

123 void yield();

241template <typename Func, typename... Args>
242inline
243futurize_t<std::invoke_result_t<Func, Args...>>
244async(thread_attributes attr, Func&& func, Args&&... args) noexcept {
245 using return_type = std::invoke_result_t<Func, Args...>;
246 struct work {
247 thread_attributes attr;
248 Func func;
249 std::tuple<Args...> args;
250 promise<return_type> pr{};
251 thread th{};
252 };
253
254 try {
255 auto wp = std::make_unique<work>(work{std::move(attr), std::forward<Func>(func), std::forward_as_tuple(std::forward<Args>(args)...)});
256 auto& w = *wp;
257 auto ret = w.pr.get_future();
258 w.th = thread(std::move(w.attr), [&w] {
259 futurize<return_type>::apply(std::move(w.func), std::move(w.args)).forward_to(std::move(w.pr));
260 });
261 return w.th.join().then([ret = std::move(ret)] () mutable {
262 return std::move(ret);
263 }).finally([wp = std::move(wp)] {});
264 } catch (...) {
265 return futurize<return_type>::make_exception_future(std::current_exception());
266 }
267}
