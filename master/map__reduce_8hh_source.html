93template <typename Iterator, typename Mapper, typename Reducer>
94requires requires (Iterator i, Mapper mapper, Reducer reduce) {
95 *i++;
96 { i != i } -> std::convertible_to<bool>;
97 mapper(*i);
98 reduce(futurize_invoke(mapper, *i).get());
99}
100inline
101auto
102map_reduce(Iterator begin, Iterator end, Mapper&& mapper, Reducer&& r)
103 -> typename reducer_traits<Reducer>::future_type
104{
105 struct state {
106 Mapper mapper;
107 Reducer reducer;
108 };
109 auto s = make_lw_shared(state{std::forward<Mapper>(mapper), std::forward<Reducer>(r)});
110 future<> ret = make_ready_future<>();
111 while (begin != end) {
112 ret = futurize_invoke(s->mapper, *begin++).then_wrapped([ret = std::move(ret), s] (auto f) mutable {
113 return ret.then_wrapped([f = std::move(f), s] (auto rf) mutable {
114 if (rf.failed()) {
115 f.ignore_ready_future();
116 return rf;
117 } else {
118 return futurize_invoke(s->reducer, std::move(f.get()));
119 }
120 });
121 });
122 }
123 return reducer_traits<Reducer, lw_shared_ptr<state>>::maybe_call_get(std::move(ret), s);
124}

168template <typename Iterator, typename Mapper, typename Initial, typename Reduce>
169requires requires (Iterator i, Mapper mapper, Initial initial, Reduce reduce) {
170 *i++;
171 { i != i} -> std::convertible_to<bool>;
172 mapper(*i);
173 requires is_future<decltype(mapper(*i))>::value;
174 { reduce(std::move(initial), mapper(*i).get()) } -> std::convertible_to<Initial>;
175}
176inline
177future<Initial>
178map_reduce(Iterator begin, Iterator end, Mapper&& mapper, Initial initial, Reduce reduce) {
179 struct state {
180 Mapper mapper;
181 Initial result;
182 Reduce reduce;
183 std::exception_ptr ex;
184 };
185 auto s = make_lw_shared(state{std::forward<Mapper>(mapper), std::move(initial), std::move(reduce), std::exception_ptr()});
186 future<> ret = make_ready_future<>();
187 while (begin != end) {
188 ret = futurize_invoke(s->mapper, *begin++).then_wrapped([s = s.get(), ret = std::move(ret)] (auto f) mutable {
189 if (s->ex) {
190 f.ignore_ready_future(); // We only report the first exception
191 } else if (f.failed()) {
192 s->ex = f.get_exception();
193 } else {
194 try {
195 s->result = s->reduce(std::move(s->result), f.get());
196 } catch (...) {
197 s->ex = std::current_exception();
198 }
199 }
200 return std::move(ret);
201 });
202 }
203 return ret.then([s] {
204 if (s->ex) {
205 return make_exception_future<Initial>(std::move(s->ex));
206 }
207 return make_ready_future<Initial>(std::move(s->result));
208 });
209}
