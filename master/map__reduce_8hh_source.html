42struct reducer_with_get_traits<T, Ptr, false> {

94template <typename Iterator, typename Mapper, typename Reducer>
95requires requires (Iterator i, Mapper mapper, Reducer reduce) {
96 *i++;
97 { i != i } -> std::convertible_to<bool>;
98 mapper(*i);
99 reduce(futurize_invoke(mapper, *i).get());
100}
101inline
102auto
103map_reduce(Iterator begin, Iterator end, Mapper&& mapper, Reducer&& r)
104 -> typename reducer_traits<Reducer>::future_type
105{
106 struct state {
107 Mapper mapper;
108 Reducer reducer;
109 };
110 auto s = make_lw_shared(state{std::forward<Mapper>(mapper), std::forward<Reducer>(r)});
111 future<> ret = make_ready_future<>();
112 while (begin != end) {
113 ret = futurize_invoke(s->mapper, *begin++).then_wrapped([ret = std::move(ret), s] (auto f) mutable {
114 return ret.then_wrapped([f = std::move(f), s] (auto rf) mutable {
115 if (rf.failed()) {
116 f.ignore_ready_future();
117 return rf;
118 } else {
119 return futurize_invoke(s->reducer, std::move(f.get()));
120 }
121 });
122 });
123 }
124 return reducer_traits<Reducer, lw_shared_ptr<state>>::maybe_call_get(std::move(ret), s);
125}

169template <typename Iterator, typename Mapper, typename Initial, typename Reduce>
170requires requires (Iterator i, Mapper mapper, Initial initial, Reduce reduce) {
171 *i++;
172 { i != i} -> std::convertible_to<bool>;
173 mapper(*i);
174 requires is_future<decltype(mapper(*i))>::value;
175 { reduce(std::move(initial), mapper(*i).get()) } -> std::convertible_to<Initial>;
176}
177inline
178future<Initial>
179map_reduce(Iterator begin, Iterator end, Mapper&& mapper, Initial initial, Reduce reduce) {
180 struct state {
181 Mapper mapper;
182 Initial result;
183 Reduce reduce;
184 std::exception_ptr ex;
185 };
186 auto s = make_lw_shared(state{std::forward<Mapper>(mapper), std::move(initial), std::move(reduce), std::exception_ptr()});
187 future<> ret = make_ready_future<>();
188 while (begin != end) {
189 ret = futurize_invoke(s->mapper, *begin++).then_wrapped([s = s.get(), ret = std::move(ret)] (auto f) mutable {
190 if (s->ex) {
191 f.ignore_ready_future(); // We only report the first exception
192 } else if (f.failed()) {
193 s->ex = f.get_exception();
194 } else {
195 try {
196 s->result = s->reduce(std::move(s->result), f.get());
197 } catch (...) {
198 s->ex = std::current_exception();
199 }
200 }
201 return std::move(ret);
202 });
203 }
204 return ret.then([s] {
205 if (s->ex) {
206 return make_exception_future<Initial>(std::move(s->ex));
207 }
208 return make_ready_future<Initial>(std::move(s->result));
209 });
210}
