31// lock / unlock semantics for rwlock, so it can be used with with_lock()

35template<typename Clock = typename timer<>::clock>
36class rwlock_for_read {
37public:
38 future<> lock() {
39 return static_cast<basic_rwlock<Clock>*>(this)->read_lock();
40 }
41 void unlock() {
42 static_cast<basic_rwlock<Clock>*>(this)->read_unlock();
43 }
44 friend class basic_rwlock<Clock>;
45};

47template<typename Clock = typename timer<>::clock>
48class rwlock_for_write {
49public:
50 future<> lock() {
51 return static_cast<basic_rwlock<Clock>*>(this)->write_lock();
52 }
53 void unlock() {

56 friend class basic_rwlock<Clock>;

71class basic_rwlock : private rwlock_for_read<Clock>, rwlock_for_write<Clock> {
72 using semaphore_type = basic_semaphore<semaphore_default_exception_factory, Clock>;
73
74 static constexpr size_t max_ops = semaphore_type::max_counter();
75
76 semaphore_type _sem;
77public:
78 basic_rwlock()

80 }
81
85 rwlock_for_read<Clock>& for_read() {
86 return *this;
87 }
88
92 rwlock_for_write<Clock>& for_write() {
93 return *this;
94 }

99 future<> read_lock(typename semaphore_type::time_point timeout = semaphore_type::time_point::max()) {
100 return _sem.wait(timeout);
101 }
102
103 future<> read_lock(abort_source& as) {
104 return _sem.wait(as);
105 }
106
110 void read_unlock() {
111 SEASTAR_ASSERT(_sem.current() < max_ops);
112 _sem.signal();
113 }
114
119 future<> write_lock(typename semaphore_type::time_point timeout = semaphore_type::time_point::max()) {
120 return _sem.wait(timeout, max_ops);
121 }
122
123 future<> write_lock(abort_source& as) {
124 return _sem.wait(as, max_ops);
125 }
126
130 void write_unlock() {
131 SEASTAR_ASSERT(_sem.current() == 0);
132 _sem.signal(max_ops);
133 }
134
136 bool try_read_lock() {
137 return _sem.try_wait();
138 }
139
141 bool try_write_lock() {
142 return _sem.try_wait(max_ops);
143 }

146
158 future<holder> hold_read_lock(typename semaphore_type::time_point timeout = semaphore_type::time_point::max()) {
159 return get_units(_sem, 1, timeout);
160 }
161
162 future<holder> hold_read_lock(abort_source& as) {
163 return get_units(_sem, 1, as);

169 return try_get_units(_sem, 1);
170 }
171
183 future<holder> hold_write_lock(typename semaphore_type::time_point timeout = semaphore_type::time_point::max()) {
184 return get_units(_sem, max_ops, timeout);
185 }
186

188 return get_units(_sem, max_ops, as);
189 }
190
193 std::optional<holder> try_hold_write_lock() noexcept {
194 return try_get_units(_sem, max_ops);
195 }
196
198 bool locked() const {
199 return _sem.available_units() != max_ops;
200 }
201
202 friend class rwlock_for_read<Clock>;
203 friend class rwlock_for_write<Clock>;
204};
