86template <typename L3>
87class arp_for : public arp_for_protocol {
88public:
89 using l2addr = ethernet_address;
90 using l3addr = typename L3::address_type;
91private:
92 static constexpr auto max_waiters = 512;
93 enum oper {
94 op_request = 1,
95 op_reply = 2,
96 };
97 struct arp_hdr {
98 uint16_t htype;
99 uint16_t ptype;
100 uint8_t hlen;
101 uint8_t plen;
102 uint16_t oper;
103 l2addr sender_hwaddr;
104 l3addr sender_paddr;
105 l2addr target_hwaddr;
106 l3addr target_paddr;
107
108 static arp_hdr read(const char* p) {
109 arp_hdr ah;
110 ah.htype = consume_be<uint16_t>(p);
111 ah.ptype = consume_be<uint16_t>(p);
112 ah.hlen = consume_be<uint8_t>(p);
113 ah.plen = consume_be<uint8_t>(p);
114 ah.oper = consume_be<uint16_t>(p);
115 ah.sender_hwaddr = l2addr::consume(p);
116 ah.sender_paddr = l3addr::consume(p);
117 ah.target_hwaddr = l2addr::consume(p);
118 ah.target_paddr = l3addr::consume(p);
119 return ah;
120 }
121 void write(char* p) const {
122 produce_be<uint16_t>(p, htype);
123 produce_be<uint16_t>(p, ptype);
124 produce_be<uint8_t>(p, hlen);
125 produce_be<uint8_t>(p, plen);
126 produce_be<uint16_t>(p, oper);
127 sender_hwaddr.produce(p);
128 sender_paddr.produce(p);
129 target_hwaddr.produce(p);
130 target_paddr.produce(p);
131 }
132 static constexpr size_t size() {
133 return 8 + 2 * (l2addr::size() + l3addr::size());
134 }
135 };
136 struct resolution {
137 std::vector<promise<l2addr>> _waiters;
138 timer<> _timeout_timer;
139 };
140private:
141 l3addr _l3self = L3::broadcast_address();
142 std::unordered_map<l3addr, l2addr> _table;
143 std::unordered_map<l3addr, resolution> _in_progress;
144private:
145 packet make_query_packet(l3addr paddr);
146 virtual future<> received(packet p) override;
147 future<> handle_request(arp_hdr* ah);
148 l2addr l2self() const noexcept { return _arp.l2self(); }
149 void send(l2addr to, packet p);
150public:
151 future<> send_query(const l3addr& paddr);
152 explicit arp_for(arp& a) : arp_for_protocol(a, L3::arp_protocol_type()) {
153 _table[L3::broadcast_address()] = ethernet::broadcast_address();
154 }
155 future<ethernet_address> lookup(const l3addr& addr);
156 void learn(l2addr l2, l3addr l3);
157 void run();
158 void set_self_addr(l3addr addr) {
159 _table.erase(_l3self);
160 _table[addr] = l2self();
161 _l3self = addr;
162 }
163 friend class arp;
164};

166template <typename L3>
167packet
168arp_for<L3>::make_query_packet(l3addr paddr) {
169 arp_hdr hdr;
170 hdr.htype = ethernet::arp_hardware_type();
171 hdr.ptype = L3::arp_protocol_type();
172 hdr.hlen = sizeof(l2addr);
173 hdr.plen = sizeof(l3addr);
174 hdr.oper = op_request;
175 hdr.sender_hwaddr = l2self();
176 hdr.sender_paddr = _l3self;
177 hdr.target_hwaddr = ethernet::broadcast_address();
178 hdr.target_paddr = paddr;
179 auto p = packet();
180 p.prepend_uninitialized_header(hdr.size());
181 hdr.write(p.get_header(0, hdr.size()));
182 return p;
183}

212template <typename L3>
213future<ethernet_address>
214arp_for<L3>::lookup(const l3addr& paddr) {
215 auto i = _table.find(paddr);
216 if (i != _table.end()) {
217 return make_ready_future<ethernet_address>(i->second);
218 }
219 auto j = _in_progress.find(paddr);
220 auto first_request = j == _in_progress.end();
221 auto& res = first_request ? _in_progress[paddr] : j->second;
222
223 if (first_request) {
224 res._timeout_timer.set_callback([paddr, this, &res] {
225 // FIXME: future is discarded
226 (void)send_query(paddr);
227 for (auto& w : res._waiters) {
228 w.set_exception(arp_timeout_error());
229 }
230 res._waiters.clear();
231 });
232 res._timeout_timer.arm_periodic(std::chrono::seconds(1));
233 // FIXME: future is discarded
234 (void)send_query(paddr);
235 }
236
237 if (res._waiters.size() >= max_waiters) {
238 return make_exception_future<ethernet_address>(arp_queue_full_error());
239 }
240
241 res._waiters.emplace_back();
242 return res._waiters.back().get_future();
243}

245template <typename L3>
246void
247arp_for<L3>::learn(l2addr hwaddr, l3addr paddr) {
248 _table[paddr] = hwaddr;
249 auto i = _in_progress.find(paddr);
250 if (i != _in_progress.end()) {
251 auto& res = i->second;
252 res._timeout_timer.cancel();
253 for (auto &&pr : res._waiters) {
254 pr.set_value(hwaddr);
255 }
256 _in_progress.erase(i);
257 }
258}

260template <typename L3>
261future<>
262arp_for<L3>::received(packet p) {
263 auto ah = p.get_header(0, arp_hdr::size());
264 if (!ah) {
265 return make_ready_future<>();
266 }
267 auto h = arp_hdr::read(ah);
268 if (h.hlen != sizeof(l2addr) || h.plen != sizeof(l3addr)) {
269 return make_ready_future<>();
270 }
271 switch (h.oper) {
272 case op_request:
273 return handle_request(&h);
274 case op_reply:
275 arp_learn(h.sender_hwaddr, h.sender_paddr);
276 return make_ready_future<>();
277 default:
278 return make_ready_future<>();
279 }
280}

282template <typename L3>
283future<>
284arp_for<L3>::handle_request(arp_hdr* ah) {
285 if (ah->target_paddr == _l3self
286 && _l3self != L3::broadcast_address()) {
287 ah->oper = op_reply;
288 ah->target_hwaddr = ah->sender_hwaddr;
289 ah->target_paddr = ah->sender_paddr;
290 ah->sender_hwaddr = l2self();
291 ah->sender_paddr = _l3self;
292 auto p = packet();
293 ah->write(p.prepend_uninitialized_header(ah->size()));
294 send(ah->target_hwaddr, std::move(p));
295 }
296 return make_ready_future<>();
297}
