148template<typename Func>

151template<typename Func,
152 bool = std::is_empty_v<return_value_t<Func>>>
153struct return_type_of {
154 using type = void;
155 static void set(std::promise<void>& p, return_value_t<Func>&&) {
156 p.set_value();
157 }
158};
159template<typename Func>
160struct return_type_of<Func, false> {
161 using return_tuple_t = typename futurize<std::invoke_result_t<Func>>::tuple_type;
162 using type = std::tuple_element_t<0, return_tuple_t>;
163 static void set(std::promise<type>& p, return_value_t<Func>&& t) {
164 p.set_value(std::move(t));
165 }
166};
167template <typename Func> using return_type_t = typename return_type_of<Func>::type;
168}

179template<std::invocable Func, typename T = internal::return_type_t<Func>>
180std::future<T> submit_to(instance& instance, unsigned shard, Func func) {
181 std::promise<T> pr;
182 auto fut = pr.get_future();
183 run_on(instance, shard, [pr = std::move(pr), func = std::move(func)] () mutable noexcept {
184 // std::future returned via std::promise above.
185 (void)func().then_wrapped([pr = std::move(pr)] (auto&& result) mutable {
186 try {
187 internal::return_type_of<Func>::set(pr, result.get());
188 } catch (...) {
189 pr.set_exception(std::current_exception());
190 }
191 });
192 });
193 return fut;
194}
