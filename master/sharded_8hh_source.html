398 template <typename Reducer, typename Func, typename... Args>
399 auto map_reduce(Reducer&& r, Func&& func, Args&&... args) -> typename reducer_traits<Reducer>::future_type
400 {
401 auto rng = std::views::iota(size_t(0), _instances.size());
402 return ::seastar::map_reduce(rng.begin(), rng.end(),
403 [this, func = std::forward<Func>(func), args = std::make_tuple(std::forward<Args>(args)...)] (unsigned c) mutable {
404 return smp::submit_to(c, [this, &func, args] () mutable {
405 return std::apply([this, &func] (Args&&... args) mutable {
406 auto inst = get_local_service();
407 return std::invoke(func, *inst, std::forward<Args>(args)...);
408 }, std::move(args));
409 });
410 }, std::forward<Reducer>(r));
411 }

485 template <typename Mapper, typename Future = futurize_t<std::invoke_result_t<Mapper,Service&>>, typename return_type = decltype(internal::untuple(std::declval<typename Future::tuple_type>()))>
486 future<std::vector<return_type>> map(Mapper mapper) {
487 return do_with(std::vector<return_type>(), std::move(mapper),
488 [this] (std::vector<return_type>& vec, Mapper& mapper) mutable {
489 vec.resize(_instances.size());
490 return parallel_for_each(std::views::iota(0u, _instances.size()), [this, &vec, &mapper] (unsigned c) {
491 return smp::submit_to(c, [this, &mapper] {
492 auto inst = get_local_service();
493 return mapper(*inst);
494 }).then([&vec, c] (auto&& res) {
495 vec[c] = std::move(res);
496 });
497 }).then([&vec] {
498 return make_ready_future<std::vector<return_type>>(std::move(vec));
499 });
500 });

605auto
606sharded_parameter<Func, Param...>::evaluate() const {
607 auto unwrap_params_and_invoke = [this] (const auto&... params) {

610 return std::apply(unwrap_params_and_invoke, _params);

613template <typename Service>
614template <typename... Args>
615future<>
616sharded<Service>::start(Args&&... args) noexcept {
617 try {
618 _instances.resize(smp::count);
619 return sharded_parallel_for_each(
620 [this, args = std::make_tuple(std::forward<Args>(args)...)] (unsigned c) mutable {
621 return smp::submit_to(c, [this, args] () mutable {
622 _instances[this_shard_id()].service = std::apply([this] (Args... args) {
623 return create_local_service(internal::unwrap_sharded_arg(std::forward<Args>(args))...);
624 }, args);
625 });
626 }).then_wrapped([this] (future<> f) {
627 try {
628 f.get();
629 return make_ready_future<>();
630 } catch (...) {
631 return this->stop().then([e = std::current_exception()] () mutable {
632 std::rethrow_exception(e);
633 });
634 }
635 });
636 } catch (...) {
637 return current_exception_as_future();
638 }
639}

669// Helper check if Service::stop exists
670
671template <typename Service>
672concept sharded_has_stop = requires {
673 // If a member names "stop" exists, try to call it, even if it doesn't
674 // have the correct signature. This is so that we don't ignore a function
675 // named stop() just because the signature is incorrect, and instead
676 // force the user to resolve the ambiguity.
677 { &Service::stop };
678};
679
680template <bool stop_exists>
681struct sharded_call_stop {
682 template <typename Service>
683 static future<> call(Service& instance);
684};
685
686template <>
687template <typename Service>
688future<> sharded_call_stop<true>::call(Service& instance) {
689 return instance.stop();
690}
691
692template <>
693template <typename Service>
694future<> sharded_call_stop<false>::call(Service&) {
695 return make_ready_future<>();
696}
697
698template <typename Service>
699future<>
700stop_sharded_instance(Service& instance) {
701 constexpr bool has_stop = internal::sharded_has_stop<Service>;
702 return internal::sharded_call_stop<has_stop>::call(instance);
703}

707template <typename Service>
708future<>
709sharded<Service>::stop() noexcept {
710 try {
711 return sharded_parallel_for_each([this] (unsigned c) mutable {
712 return smp::submit_to(c, [this] () mutable {
713 auto inst = _instances[this_shard_id()].service;
714 if (!inst) {
715 return make_ready_future<>();
716 }
717 return internal::stop_sharded_instance(*inst);
718 });
719 }).then_wrapped([this] (future<> fut) {
720 return sharded_parallel_for_each([this] (unsigned c) {
721 return smp::submit_to(c, [this] {
722 auto fut = _instances[this_shard_id()].track_deletion();
723 _instances[this_shard_id()].service = nullptr;
724 return fut;
725 });
726 }).finally([this, fut = std::move(fut)] () mutable {
727 _instances.clear();
728 _instances = std::vector<sharded<Service>::entry>();
729 return std::move(fut);
730 });
731 });
732 } catch (...) {
733 return current_exception_as_future();
734 }
735}
