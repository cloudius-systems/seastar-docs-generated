55// We can't keep this in any of the socket servers as instance members, because a connection can
56// outlive the socket server. To avoid having the whole socket_server tracked as a shared pointer,
57// we will have a conntrack structure.
58//
59// Right now this class is used by the posix_server_socket_impl, but it could be used by any other.
60class conntrack {
61 class load_balancer {
62 std::vector<unsigned> _cpu_load;
63 public:
64 load_balancer() : _cpu_load(size_t(smp::count), 0) {}
65 void closed_cpu(shard_id cpu) {
66 _cpu_load[cpu]--;
67 }
68 shard_id next_cpu() {
69 // FIXME: The naive algorithm will just round robin the connections around the shards.
70 // A more complex version can keep track of the amount of activity in each connection,
71 // and use that information.
72 auto min_el = std::min_element(_cpu_load.begin(), _cpu_load.end());
73 auto cpu = shard_id(std::distance(_cpu_load.begin(), min_el));
74 _cpu_load[cpu]++;
75 return cpu;
76 }
77 shard_id force_cpu(shard_id cpu) {
78 _cpu_load[cpu]++;
79 return cpu;
80 }
81 };
82
83 lw_shared_ptr<load_balancer> _lb;
84 void closed_cpu(shard_id cpu) {
85 _lb->closed_cpu(cpu);
86 }
87public:
88 class handle {
89 shard_id _host_cpu;
90 shard_id _target_cpu;
91 foreign_ptr<lw_shared_ptr<load_balancer>> _lb;
92 public:
93 handle() : _lb(nullptr) {}
94 handle(shard_id cpu, lw_shared_ptr<load_balancer> lb)
95 : _host_cpu(this_shard_id())
96 , _target_cpu(cpu)
97 , _lb(make_foreign(std::move(lb))) {}
98
99 handle(const handle&) = delete;
100 handle(handle&&) = default;
101 handle& operator=(const handle&) = delete;
102 handle& operator=(handle&&) = default;
103 ~handle() {
104 if (!_lb) {
105 return;
106 }
107 // FIXME: future is discarded
108 (void)smp::submit_to(_host_cpu, [cpu = _target_cpu, lb = std::move(_lb)] {
109 lb->closed_cpu(cpu);
110 });
111 }
112 shard_id cpu() {
113 return _target_cpu;
114 }
115 };
116 friend class handle;
117
118 conntrack() : _lb(make_lw_shared<load_balancer>()) {}
119 handle get_handle() {
120 return handle(_lb->next_cpu(), _lb);
121 }
122 handle get_handle(shard_id cpu) {
123 return handle(_lb->force_cpu(cpu), _lb);
124 }
125};
